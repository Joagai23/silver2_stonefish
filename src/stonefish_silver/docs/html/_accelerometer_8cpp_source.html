43void Accelerometer::InternalUpdate(Scalar dt)
44{
45 // Calculate transformation from global to imu frame
46 Transform accTrans = getSensorFrame();
47
48 // Get acceleration
49 Vector3 R = accTrans.getOrigin() - attach->getCGTransform().getOrigin();
50 Vector3 la = accTrans.getBasis().inverse() * (
51 attach->getLinearAcceleration()
52 + attach->getAngularAcceleration().cross(R)
53 + attach->getAngularVelocity().cross(attach->getAngularVelocity().cross(R))
54 - SimulationApp::getApp()->getSimulationManager()->getGravity()
55 );
56
57 // Record sample
58 Sample s{std::vector<Scalar>({la.x(), la.y(), la.z()})};
59 AddSampleToHistory(s);
60}

62void Accelerometer::setRange(Vector3 linearAccelerationMax)
63{
64 channels[0].rangeMin = -btClamped(linearAccelerationMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
65 channels[1].rangeMin = -btClamped(linearAccelerationMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
66 channels[2].rangeMin = -btClamped(linearAccelerationMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
67 channels[0].rangeMax = btClamped(linearAccelerationMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
68 channels[1].rangeMax = btClamped(linearAccelerationMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
69 channels[2].rangeMax = btClamped(linearAccelerationMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
70}
