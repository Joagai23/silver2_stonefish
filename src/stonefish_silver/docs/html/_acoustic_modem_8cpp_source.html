40void AcousticModem::addNode(AcousticModem* node)
41{
42 if(node->getDeviceId() == 0)
43 {
44 cError("Modem device ID=0 not allowed!");
45 return;
46 }
47
48 if(nodes.find(node->getDeviceId()) != nodes.end())
49 cError("Modem node with ID=%d already exists!", node->getDeviceId());
50 else
51 nodes[node->getDeviceId()] = node;
52}

54void AcousticModem::removeNode(uint64_t deviceId)
55{
56 if(deviceId == 0)
57 return;
58
59 std::map<uint64_t, AcousticModem*>::iterator it = nodes.find(deviceId);
60 if(it != nodes.end())
61 nodes.erase(it);
62}

87bool AcousticModem::mutualContact(uint64_t device1Id, uint64_t device2Id)
88{
89 AcousticModem* node1 = getNode(device1Id);
90 AcousticModem* node2 = getNode(device2Id);
91
92 if(node1 == nullptr || node2 == nullptr)
93 return false;
94
95 Vector3 pos1 = node1->getDeviceFrame().getOrigin();
96 Vector3 pos2 = node2->getDeviceFrame().getOrigin();
97 Vector3 dir = pos2-pos1;
98 Scalar distance = dir.length();
99
100 if(!node1->isReceptionPossible(dir, distance) || !node2->isReceptionPossible(-dir, distance))
101 return false;
102
103 if(node1->getOcclusionTest() || node2->getOcclusionTest())
104 {
105 btCollisionWorld::ClosestRayResultCallback closest(pos1, pos2);
106 closest.m_collisionFilterGroup = MASK_DYNAMIC;
107 closest.m_collisionFilterMask = MASK_STATIC | MASK_DYNAMIC | MASK_ANIMATED_COLLIDING;
108 SimulationApp::getApp()->getSimulationManager()->getDynamicsWorld()->rayTest(pos1, pos2, closest);
109 return !closest.hasHit();
110 }
111 else
112 return true;
113}

116AcousticModem::AcousticModem(std::string uniqueName, uint64_t deviceId, Scalar minVerticalFOVDeg, Scalar maxVerticalFOVDeg, Scalar operatingRange)
117 : Comm(uniqueName, deviceId)
118{
119 btClamp(maxVerticalFOVDeg, Scalar(0), Scalar(360));
120 btClamp(minVerticalFOVDeg, Scalar(0), maxVerticalFOVDeg);
121 minFov2 = btRadians(minVerticalFOVDeg/Scalar(2));
122 maxFov2 = btRadians(maxVerticalFOVDeg/Scalar(2));
123 range = operatingRange <= Scalar(0) ? Scalar(1000) : operatingRange;
124 position = V0();
125 frame = std::string("");
126 occlusion = true;
127 addNode(this);
128}

135bool AcousticModem::isReceptionPossible(Vector3 worldDir, Scalar distance)
136{
137 //Check if modems are close enough
138 if(distance > range) return false;
139
140 //Check if direction is in the FOV of the device
141 Vector3 dir = (getDeviceFrame().getBasis().inverse() * worldDir).normalized();
142 Scalar d = Vector3(dir.getX(), dir.getY(), Scalar(0)).safeNorm();
143 Scalar vAngle = M_PI_2; // When dir.z == 0.0
144 if(!btFuzzyZero(dir.getZ()))
145 vAngle = atan2(d, -dir.getZ());
146 return btFabs(vAngle) >= minFov2 && btFabs(vAngle) <= maxFov2;
147}

170void AcousticModem::SendMessage(const std::vector<uint8_t>& data)
171{
172 if(getConnectedId() < 0) // Not connected
173 return;
174 else if(getConnectedId() == 0) // Broadcast
175 {
176 std::vector<uint64_t> nodeIds = getNodeIds();
177 for(size_t i=0; i<nodeIds.size(); ++i)
178 {
179 if(nodeIds[i] != getDeviceId() && mutualContact(getDeviceId(), nodeIds[i]))
180 {
181 auto msg = std::make_shared<AcousticDataFrame>();
182 msg->timeStamp = SimulationApp::getApp()->getSimulationManager()->getSimulationTime(true);
183 msg->seq = txSeq++;
184 msg->source = getDeviceId();
185 msg->destination = nodeIds[i];
186 msg->data = data;
187 msg->txPosition = getDeviceFrame().getOrigin();
188 msg->travelled = Scalar(0);
189 txBuffer.push_back(msg);
190 }
191 }
192 }
193 else // Conneted to one receiver
194 {
195 if(!mutualContact(getDeviceId(), getConnectedId()))
196 return;
197
198 auto msg = std::make_shared<AcousticDataFrame>();
199 msg->timeStamp = SimulationApp::getApp()->getSimulationManager()->getSimulationTime(true);
200 msg->seq = txSeq++;
201 msg->source = getDeviceId();
202 msg->destination = getConnectedId();
203 msg->data = data;
204 msg->txPosition = getDeviceFrame().getOrigin();
205 msg->travelled = Scalar(0);
206 txBuffer.push_back(msg);
207 }
208}

210void AcousticModem::ProcessMessages()
211{
212 std::shared_ptr<AcousticDataFrame> msg;
213 std::string ack {"ACK"};
214 std::vector<uint8_t> ackData {ack.begin(), ack.end()};
215 std::string ping {"PING"};
216 std::vector<uint8_t> pingData {ping.begin(), ping.end()};
217
218 for(auto it = rxBuffer.begin(); it != rxBuffer.end(); ++it)
219 {
220 // Send "ACK" message back to sender
221 msg = std::static_pointer_cast<AcousticDataFrame>(*it);
222 if(msg->data != ackData)
223 {
224 //timestamp and sequence don't change
225 std::shared_ptr<AcousticDataFrame> ackMsg = std::make_shared<AcousticDataFrame>();
226 ackMsg->timeStamp = msg->timeStamp;
227 ackMsg->seq = msg->seq;
228 ackMsg->destination = msg->source;
229 ackMsg->source = getDeviceId();
230 ackMsg->data = ackData;
231 ackMsg->txPosition = getDeviceFrame().getOrigin();
232 ackMsg->travelled = msg->travelled;
233 txBuffer.push_back(ackMsg);
234 }
235 }
236 // Remove "ACK" and "PING" messages from the RX buffer
237 rxBuffer.erase(std::remove_if(rxBuffer.begin(), rxBuffer.end(),
238 [&ackData, &pingData](const std::shared_ptr<CommDataFrame>& msg) {
239 return (msg->data == ackData || msg->data == pingData);
240 }), rxBuffer.end());
241}

243void AcousticModem::InternalUpdate(Scalar dt)
244{
245 //Propagate messages already sent
246 for(auto mIt = propagating.begin(); mIt != propagating.end();)
247 {
248 AcousticModem* dest = getNode(mIt->first->destination);
249 Vector3 dO = dest->getDeviceFrame().getOrigin();
250 Vector3 sO = mIt->second;
251 Vector3 dir = dO - sO;
252 Scalar d = dir.length();
253
254 if(d <= SOUND_VELOCITY_WATER*dt) //Message reached?
255 {
256 mIt->first->travelled += d;
257 dest->MessageReceived(mIt->first);
258 mIt = propagating.erase(mIt);
259 }
260 else //Advance pulse
261 {
262 dir /= d; //Normalize direction
263 d = SOUND_VELOCITY_WATER * dt;
264 mIt->second += dir * d;
265 mIt->first->travelled += d;
266 ++mIt;
267 }
268 }
269
270 //Send first message from the tx buffer
271 if(txBuffer.size() > 0)
272 {
273 auto msg = std::static_pointer_cast<AcousticDataFrame>(txBuffer[0]);
274 txBuffer.pop_front();
275
276 // The message is sent or lost
277 if(mutualContact(msg->source, msg->destination))
278 {
279 propagating.insert(std::make_pair(msg, msg->txPosition));
280 }
281 }

294std::vector<Renderable> AcousticModem::Render()
295{
296 std::vector<Renderable> items(0);
297
298 //Fov indicator
299 Renderable item;
300 item.model = glMatrixFromTransform(getDeviceFrame());
301 item.type = RenderableType::SENSOR_LINES;
302 GLfloat iconSize = 0.25f;
303 int div = 24;
304 //Upper circle
305 if(minFov2 > Scalar(0))
306 {
307 GLfloat r = iconSize * glm::sin((GLfloat)minFov2);
308 GLfloat h = iconSize * glm::cos((GLfloat)minFov2);
309 for(int i=0; i<=div; ++i)
310 {
311 GLfloat angle = (GLfloat)i/(GLfloat)div * 2.f * M_PI;
312 item.points.push_back(glm::vec3(glm::cos(angle)*r, glm::sin(angle)*r, -h));
313 if(i > 0 && i < div)
314 item.points.push_back(item.points.back());
315 }
316 }
317 //Lower circle
318 if(maxFov2 < Scalar(M_PI))
319 {
320 GLfloat r = iconSize * glm::sin((GLfloat)maxFov2);
321 GLfloat h = iconSize * glm::cos((GLfloat)maxFov2);
322 for(int i=0; i<=div; ++i)
323 {
324 GLfloat angle = (GLfloat)i/(GLfloat)div * 2.f * M_PI;
325 item.points.push_back(glm::vec3(glm::cos(angle)*r, glm::sin(angle)*r, -h));
326 if(i > 0 && i < div)
327 item.points.push_back(item.points.back());
328 }
329 }
330 //4 bars
331 div = 16;
332 if(maxFov2 - minFov2 > Scalar(0))
333 {
334 for(int i=0; i<4; ++i)
335 {
336 GLfloat hangle = (GLfloat)(i * M_PI_2);
337 GLfloat x = iconSize * glm::cos(hangle);
338 GLfloat y = iconSize * glm::sin(hangle);
339 for(int h=0; h<=div; ++h)
340 {
341 GLfloat angle = (GLfloat)h/(GLfloat)div * (maxFov2-minFov2) + minFov2;
342 item.points.push_back(glm::vec3(glm::sin(angle)*x, glm::sin(angle)*y, -glm::cos(angle)*iconSize));
343 if(h == 0 && minFov2 > Scalar(0))
344 {
345 glm::vec3 v = item.points.back();
346 item.points.push_back(glm::vec3(0.f,0.f,0.f));
347 item.points.push_back(v);
348 }
349 else if(h == div && maxFov2 < Scalar(M_PI))
350 {
351 item.points.push_back(item.points.back());
352 item.points.push_back(glm::vec3(0.f,0.f,0.f));
353 }
354 else if(h > 0 && h < div)
355 item.points.push_back(item.points.back());
356 }
357 }
358 }
359 items.push_back(item);
360
361 //Axes
362 item.type = RenderableType::SENSOR_CS;
363 items.push_back(item);
364
365 //Connected nodes
366 item.type = RenderableType::SENSOR_LINES;
367 item.model = glm::mat4(1.f);
368 item.points.clear();
369 if(getConnectedId() == 0)
370 {
371 std::vector<uint64_t> nodeIds = getNodeIds();
372 for(size_t i=0; i<nodeIds.size(); ++i)
373 if(nodeIds[i] != getDeviceId())
374 {
375 Transform Tn = getNode(nodeIds[i])->getDeviceFrame();
376 item.points.push_back(glVectorFromVector(getDeviceFrame().getOrigin()));
377 item.points.push_back(glVectorFromVector(Tn.getOrigin()));
378 }
379 }
380 else if(getConnectedId() > 0)
381 {
382 AcousticModem* cNode = getNode(getConnectedId());
383 if(cNode != nullptr)
384 {
385 item.points.push_back(glVectorFromVector(getDeviceFrame().getOrigin()));
386 item.points.push_back(glVectorFromVector(cNode->getDeviceFrame().getOrigin()));
387 }
388 }
389 if(!item.points.empty())
390 items.push_back(item);
391
392#ifdef DEBUG
393 item.type = RenderableType::SENSOR_POINTS;
394 item.model = glm::mat4(1.f);
395 for( auto mIt = propagating.begin(); mIt != propagating.end(); ++mIt)
396 {
397 Vector3 mPos = mIt->second;
398 item.points.push_back(glm::vec3((GLfloat)mPos.getX(), (GLfloat)mPos.getY(), (GLfloat)mPos.getZ()));
399 }
400 items.push_back(item);
401#endif
402
403 return items;
404}
