97
100 FirstOrder(Scalar tau) : tau(tau)
101 {
102 }
103
105
109 Scalar Update(Scalar dt, Scalar sp) override
110 {
111 Scalar alpha = dt / tau;
112 Scalar output = alpha * sp + (1 - alpha) * lastOutput;
113 lastOutput = outputLimit > Scalar(0) ? btClamped(output, -outputLimit, outputLimit) : output;
114 return lastOutput;
115 }

119 {
120 return RotorDynamicsType::FIRST_ORDER;
121 }
122
123 private:

132
136 Yoerger(Scalar alpha, Scalar beta) : alpha(alpha), beta(beta)
137 {
138 }
139
141
145 Scalar Update(Scalar dt, Scalar sp) override
146 {
147 // state += dt*(beta*_cmd - alpha*state*std::abs(state));
148 Scalar output = lastOutput + dt * (beta * sp - (alpha * lastOutput * btFabs(lastOutput)));
149 lastOutput = outputLimit > Scalar(0) ? btClamped(output, -outputLimit, outputLimit) : output;
150 return lastOutput;
151 }

209 class MechanicalPI : public RotorDynamics
210 {
211 public:
213
219 MechanicalPI(Scalar J, Scalar Kp, Scalar Ki, Scalar iLim)
220 : J(J), Kp(Kp), Ki(Ki), iLim(iLim), iError(0), damping(0)
221 {
222 }
223
225
229 Scalar Update(Scalar dt, Scalar sp) override
230 {
231 Scalar error = sp - lastOutput;
232 Scalar tau = Kp * error + Ki * iError;
233 iError = btClamped(iError + error * dt, -iLim, iLim);
234
235 Scalar tauD = lastOutput > Scalar(0) ? damping : -damping;
236 Scalar output = lastOutput + (tau - tauD)/J * dt;
237 lastOutput = outputLimit > Scalar(0) ? btClamped(output, -outputLimit, outputLimit) : output;
238 return lastOutput;
239 }
240
242
245 void setDampingTorque(Scalar tau)
246 {
247 damping = btFabs(tau);
248 }
249
251 RotorDynamicsType getType()
252 {
253 return RotorDynamicsType::MECHANICAL_PI;
254 }
255
256 private:
257 Scalar J;
258 Scalar Kp;
259 Scalar Ki;
260 Scalar iLim;
261 Scalar iError;
262 Scalar damping;
263 };

320 class DeadbandThrust : public ThrustModel
321 {
322 public:
324
330 DeadbandThrust(Scalar ktn, Scalar ktp, Scalar dl, Scalar du) : ktn(ktn), ktp(ktp), dl(dl), du(du)
331 {
332 }
333
335
339 std::pair<Scalar, Scalar> Update(Scalar input) override
340 {
341 Scalar vv = input * btFabs(input);
342 Scalar thrust(0);
343 if (vv < dl)
344 {
345 thrust = ktn * (vv - dl);
346 }
347 else if (vv > du)
348 {
349 thrust = ktp * (vv - du);
350 }
351 return std::make_pair(thrust, Scalar(0));
352 }
353
355 ThrustModelType getType() override
356 {
357 return ThrustModelType::DEADBAND;
358 }

362 Scalar ktp;
363 Scalar dl;
364 Scalar du;

368 class InterpolatedThrust : public ThrustModel
369 {
370 public:
372
376 InterpolatedThrust(const std::vector<Scalar>& in, const std::vector<Scalar>& out)
377 : inputValues(in), outputValues(out)
378 {
379 if (inputValues.empty() || outputValues.empty())
380 throw std::runtime_error("Interpolated thrust model: input and output values must not be empty!");
381
382 if (inputValues.size() != outputValues.size())
383 throw std::runtime_error("Interpolated thrust model: input and output values must be same size!");
384 }
385
387
391 std::pair<Scalar, Scalar> Update(Scalar input) override
392 {
393 Scalar thrust(0);
394
395 // Ensure the input values are sorted
396 auto it = std::lower_bound(inputValues.begin(), inputValues.end(), input);
397
398 if (it == inputValues.begin()) // If the value is less than the smallest input value, return the first output value
399 {
400 thrust = outputValues.front();
401 }
402 else if (it == inputValues.end()) // If the value is greater than the largest input value, return the last output value
403 {
404 thrust = outputValues.back();
405 }
406 else
407 {
408 // Perform linear interpolation
409 auto idx = std::distance(inputValues.begin(), it);
410 Scalar x0 = inputValues[idx - 1];
411 Scalar x1 = inputValues[idx];
412 Scalar y0 = outputValues[idx - 1];
413 Scalar y1 = outputValues[idx];
414 thrust = y0 + (input - x0) * (y1 - y0) / (x1 - x0);
415 }
416 return std::make_pair(thrust, Scalar(0));
417 }
418
420 ThrustModelType getType() override
421 {
422 return ThrustModelType::LINTERP;
423 }
424
425 protected:
426 std::vector<Scalar> inputValues, outputValues;
427 };

430 class FDThrust : public ThrustModel
431 {
432 public:
434
441 FDThrust(Scalar D, Scalar ktp, Scalar ktn, Scalar kq, bool RH, Scalar rho)
442 : D(D), ktp(ktp), ktn(ktn), kq(kq), RH(RH), rho(rho)
443 {
444 // TODO: Find a better way of defining alpha and beta
445 alpha = -ktp;
446 beta = -kq;
447 }
448
450
454 std::pair<Scalar, Scalar> Update(Scalar input) override
455 {
456 bool backward = (RH && input < Scalar(0)) || (!RH && input > Scalar(0));
457
458 /*kt and kq depend on the advance ratio J
459 J = u/(omega*D), where:
460 u - ambient velocity [m/s]
461 n - propeller rotational rate [1/s]
462 D - propeller diameter [m] */
463 Scalar n = (backward ? Scalar(-1) : Scalar(1)) * btFabs(input)/(Scalar(2) * M_PI); // Accounts for propoller handedness
464
465 // Thrust is the force generated by pushing the liquid through the thruster.
466 Scalar kt0 = backward ? ktn : ktp; //In case of non-symmetrical thrusters the coefficient may be different
467 //kt(J) = kt0 + alpha * J --> approximated with linear function
468 Scalar thrust = rho * D*D*D * btFabs(n) * (D*kt0*n + alpha*u);
469
470 // Torque is the loading of propeller due to liquid resistance (reaction force).
471 Scalar kq0 = kq;
472 //kQ(J) = kQ0 + beta * J --> approximated with linear function
473 Scalar torque = (RH ? Scalar(-1) : Scalar(1)) * rho * D*D*D*D * btFabs(n) * (D*kq0*n + beta*u);
474
475 return std::make_pair(thrust, torque);
476 }
477
479
482 void setIncomingFluidVelocity(Scalar vel)
483 {
484 u = vel;
485 }
486
488 ThrustModelType getType() override
489 {
490 return ThrustModelType::FD;
491 }
492
493 protected:
494 Scalar D;
495 Scalar ktp;
496 Scalar ktn;
497 Scalar kq;
498 Scalar u;
499 bool RH;
500 Scalar rho;
501 Scalar alpha;
502 Scalar beta;
503 };
