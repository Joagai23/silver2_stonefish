36AnimatedEntity::AnimatedEntity(std::string uniqueName, Trajectory* traj) : MovingEntity(uniqueName, "", ""), tr(traj)
37{
38 if(traj == nullptr)
39 return;
40
41 T_CG2O = T_O2C = T_O2G = I4();
42
43 linearAcc.setZero();
44 angularAcc.setZero();
45
46 //Build rigid body
47 btEmptyShape* shape = new btEmptyShape();
48 BuildRigidBody(shape, false);
49 phyObjectId = graObjectId = -1;
50}

52AnimatedEntity::AnimatedEntity(std::string uniqueName, Trajectory* traj, Scalar sphereRadius, const Transform& origin, std::string material, std::string look, bool collides)
53 : MovingEntity(uniqueName, material, look), tr(traj)
54{
55 if(traj == nullptr)

59 T_CG2O = origin.inverse();

64
65 //Build graphical objects

69 phyObjectId = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->BuildObject(phyMesh);
70 graObjectId = phyObjectId;
71 delete phyMesh;
72 }

127AnimatedEntity::AnimatedEntity(std::string uniqueName, Trajectory* traj, std::string graphicsFilename, Scalar graphicsScale, const Transform& graphicsOrigin,
128 std::string physicsFilename, Scalar physicsScale, const Transform& physicsOrigin, std::string material, std::string look, bool collides)
129 : MovingEntity(uniqueName, material, look), tr(traj)
130{
131 if(traj == nullptr)
132 return;
133
134 //Load geometry from files
135 Mesh* graMesh = OpenGLContent::LoadMesh(graphicsFilename, graphicsScale, false);
136 Mesh* phyMesh;
137 T_O2G = graphicsOrigin;
138 T_CG2O = I4();
139
140 if(physicsFilename != "")
141 {
142 phyMesh = OpenGLContent::LoadMesh(physicsFilename, physicsScale, false);
143 T_O2C = physicsOrigin;
144 }
145 else
146 {
147 phyMesh = graMesh;
148 T_O2C = T_O2G;
149 }
150
151 //Build rigid body
152 btConvexHullShape* shape = new btConvexHullShape();
153 for(size_t i=0; i<phyMesh->getNumOfVertices(); ++i)
154 {
155 glm::vec3 pos = phyMesh->getVertexPos(i);
156 Vector3 v(pos.x, pos.y, pos.z);
157 shape->addPoint(v);
158 }
159 shape->optimizeConvexHull();
160 BuildRigidBody(shape, collides);
161
162 //Build graphical objects
163 if(SimulationApp::getApp()->hasGraphics())
164 {
165 phyObjectId = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->BuildObject(phyMesh);
166 if(graMesh != phyMesh)
167 graObjectId = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->BuildObject(graMesh);
168 else
169 graObjectId = phyObjectId;
170 }
171
172 //Delete mesh data
173 if(graMesh != phyMesh)
174 {
175 delete phyMesh;
176 delete graMesh;
177 }
178 else
179 delete phyMesh;
180}

257void AnimatedEntity::BuildRigidBody(btCollisionShape* shape, bool collides)
258{
259 btDefaultMotionState* motionState = new btDefaultMotionState(tr->getInterpolatedTransform());
260 shape->setMargin(0.0);
261
262 btRigidBody::btRigidBodyConstructionInfo rigidBodyCI(Scalar(0), motionState, shape, V0());
263 rigidBodyCI.m_friction = rigidBodyCI.m_rollingFriction = rigidBodyCI.m_restitution = Scalar(0); //not used
264 rigidBodyCI.m_linearDamping = rigidBodyCI.m_angularDamping = Scalar(0); //not used
265 rigidBodyCI.m_linearSleepingThreshold = rigidBodyCI.m_angularSleepingThreshold = Scalar(0); //not used
266 rigidBodyCI.m_additionalDamping = false;
267
268 rigidBody = new btRigidBody(rigidBodyCI);
269 rigidBody->setUserPointer(this);
270 rigidBody->setCollisionFlags(rigidBody->getCollisionFlags()
271 | btCollisionObject::CF_KINEMATIC_OBJECT
272 | btCollisionObject::CF_CUSTOM_MATERIAL_CALLBACK);
273 if(!collides) rigidBody->setCollisionFlags(rigidBody->getCollisionFlags() | btCollisionObject::CF_NO_CONTACT_RESPONSE);
274 rigidBody->setActivationState(DISABLE_DEACTIVATION);
275}

278{

280 {
281 if(rigidBody->getCollisionFlags() & btCollisionObject::CF_NO_CONTACT_RESPONSE) //No collisions
282 sm->getDynamicsWorld()->addRigidBody(rigidBody, MASK_ANIMATED_NONCOLLIDING, MASK_DYNAMIC);
283 else
284 sm->getDynamicsWorld()->addRigidBody(rigidBody, MASK_ANIMATED_COLLIDING, MASK_DYNAMIC); //Only collide with dynamic bodies

293void AnimatedEntity::Update(Scalar dt)
294{
295 if(tr == nullptr || rigidBody == nullptr)
296 return;
297
298 tr->Play(dt);
299 rigidBody->getMotionState()->setWorldTransform(tr->getInterpolatedTransform() * T_CG2O.inverse());
300 rigidBody->setLinearVelocity(tr->getInterpolatedLinearVelocity());
301 rigidBody->setAngularVelocity(tr->getInterpolatedAngularVelocity());
302 setLinearAcceleration(tr->getInterpolatedLinearAcceleration());
303}

305std::vector<Renderable> AnimatedEntity::Render()
306{
307 std::vector<Renderable> items(0);
308
309 if(rigidBody != nullptr && isRenderable())
310 {
311 Renderable item;
312 item.type = RenderableType::SOLID_CS;
313 item.model = glMatrixFromTransform(getOTransform());
314 items.push_back(item);
315
316 if(graObjectId >= 0)
317 {
318 item.type = RenderableType::SOLID;
319 item.cor = glVectorFromVector(getOTransform().getOrigin());
320 item.vel = glVectorFromVector(getLinearVelocity());
321 item.avel = glVectorFromVector(getAngularVelocity());
322 item.materialName = mat.name;
323 item.objectId = dm == DisplayMode::GRAPHICAL ? graObjectId : phyObjectId;
324 item.lookId = dm == DisplayMode::GRAPHICAL ? lookId : -1;
325 items.push_back(item);
326 }
327
328 std::vector<Renderable> trajectoryItems = tr->Render();
329 items.insert(items.begin(), trajectoryItems.begin(), trajectoryItems.end());
330 }
331
332 return items;
333}
