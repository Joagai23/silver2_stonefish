35Atmosphere::Atmosphere(std::string uniqueName, Fluid g) : ForcefieldEntity(uniqueName)
36{
37 ghost->setCollisionFlags(ghost->getCollisionFlags() | btCollisionObject::CF_STATIC_OBJECT);
38
39 Scalar size(100000);
40 Vector3 halfExtents = Vector3(size/Scalar(2), size/Scalar(2), size/Scalar(2));
41 ghost->setWorldTransform(Transform(Quaternion::getIdentity(), Vector3(0,0,-size/Scalar(2)))); //Above ocean surface and ground (z=0)
42 ghost->setCollisionShape(new btBoxShape(halfExtents));
43
44 gas = g;
45 wind = std::vector<VelocityField*>(0);
46 glAtmosphere = nullptr;
47}

82void Atmosphere::SetSunPosition(Scalar longitudeDeg, Scalar latitudeDeg, std::tm& utc)
83{
84 if(glAtmosphere == nullptr) return;
85
86 Scalar latitude = latitudeDeg/Scalar(180) * M_PI;
87 Scalar longitude = longitudeDeg/Scalar(180) * M_PI;
88 Scalar meridian(0); //GMT
89 Scalar decimalHours = (Scalar)utc.tm_hour + (Scalar)utc.tm_min/Scalar(60) + (Scalar)utc.tm_sec/Scalar(60);
90 std::tm utc0 = utc;
91 utc0.tm_mday = 1; //1
92 utc0.tm_mon = 0; //January
93 int J = JulianDay(utc) - JulianDay(utc0);
94 Scalar solarTime = decimalHours + Scalar(0.17)*btSin(Scalar(4)*M_PI*(J - 80)/Scalar(373)) - Scalar(0.129)*btSin(Scalar(2)*M_PI*(J - 8)/Scalar(355)) + Scalar(12)*(meridian - longitude)/M_PI;
95 Scalar declination = Scalar(0.4093)*btSin(Scalar(2)*M_PI*(J - 81)/Scalar(368));
96 Scalar elevation = btAsin(btSin(latitude)*btSin(declination) - btCos(latitude)*btCos(declination)*btCos(M_PI*solarTime/Scalar(12)));
97 Scalar azimuth = btAtan2(btSin(M_PI*solarTime/Scalar(12)), btCos(M_PI*solarTime/Scalar(12))*btSin(latitude) - btTan(declination)*btCos(latitude));
98
99 glAtmosphere->SetSunPosition((float)(azimuth/M_PI*Scalar(180)), (float)(elevation/M_PI*Scalar(180)));
100}

164void Atmosphere::ApplyFluidForces(btDynamicsWorld* world, btCollisionObject* co, bool recompute)
165{
166 Entity* ent;
167 btRigidBody* rb = btRigidBody::upcast(co);
168 btMultiBodyLinkCollider* mbl = btMultiBodyLinkCollider::upcast(co);
169
170 if(rb != 0)
171 {
172 if(rb->isStaticOrKinematicObject())
173 return;
174 else
175 ent = (Entity*)rb->getUserPointer();
176 }
177 else if(mbl != 0)
178 {
179 if(mbl->isStaticOrKinematicObject())
180 return;
181 else
182 ent = (Entity*)mbl->getUserPointer();
183 }
184 else
185 return;
186
187 if(ent->getType() == EntityType::SOLID)
188 {
189 if(recompute)
190 {
191 ((SolidEntity*)ent)->ComputeAerodynamicForces(this);
192 }
193
194 ((SolidEntity*)ent)->ApplyAerodynamicForces();
195 }
196}

198int Atmosphere::JulianDay(std::tm& tm)
199{
200 int m = tm.tm_mon + 1;
201 int y = tm.tm_year + 1900;
202 int d = tm.tm_mday;
203
204 Scalar X = (m + 9) / 12.0;
205 int A = 4716 + y + (int)trunc(X);
206 Scalar Y = 275 * m / 9.0;
207 Scalar V = 7 * A / 4.0;
208 Scalar B = 1729279.5 + 367 * y + trunc(Y) - trunc(V) + d;
209 Scalar Q = (A + 83) / 100.0;
210 Scalar W = 3 * (trunc(Q) + 1) / 4.0;
211 return B + 38 - (int)trunc(W);
212}
