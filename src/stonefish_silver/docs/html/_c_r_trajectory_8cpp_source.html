36void CRTrajectory::Interpolate()
37{
38 if(points.size() < 3)
39 PWLTrajectory::Interpolate();
40 else
41 {
42 //Find current path segment
43 auto it = std::find_if(points.begin(), points.end(),
44 [&](const auto& key){ return key.t >= playTime; });
45
46 Transform T0, T1, T2, T3;
47 Scalar t0, t1, t2, t3;
48
49 if(it <= points.begin()+1) //Beginning
50 {
51 T1 = points[0].T;
52 T2 = points[1].T;
53 T3 = points[2].T;
54 T0.setOrigin(T1.getOrigin()-(T2.getOrigin()-T1.getOrigin()));
55 T0.setRotation(T1.getRotation());
56 t1 = points[0].t;
57 t2 = points[1].t;
58 t3 = points[2].t;
59 t0 = t1-(t2-t1);
60 }
61 else if(it >= points.end()-1) //End
62 {
63 T0 = points[points.size()-3].T;
64 T1 = points[points.size()-2].T;
65 T2 = points.back().T;
66 T3.setOrigin(T2.getOrigin()+(T2.getOrigin()-T1.getOrigin()));
67 T3.setRotation(T2.getRotation());
68 t0 = points[points.size()-3].t;
69 t1 = points[points.size()-2].t;
70 t2 = points.back().t;
71 t3 = t2 + (t2-t1);
72 }
73 else
74 {
75 T0 = (it-2)->T;
76 T1 = (it-1)->T;
77 T2 = it->T;
78 T3 = (it+1)->T;
79 t0 = (it-2)->t;
80 t1 = (it-1)->t;
81 t2 = it->t;
82 t3 = (it+1)->t;
83 }
84
85 //Linear quantities
86 if(btFuzzyZero((T2.getOrigin()-T1.getOrigin()).safeNorm())) //Coinciding points
87 {
88 interpTrans.setOrigin(T2.getOrigin());
89 interpVel = V0();
90 }
91 else
92 {
93 interpTrans.setOrigin(catmullRom(T0.getOrigin(), T1.getOrigin(), T2.getOrigin(), T3.getOrigin(),
94 t0, t1, t2, t3, playTime));
95 interpVel = catmullRomDerivative(T0.getOrigin(), T1.getOrigin(), T2.getOrigin(), T3.getOrigin(),
96 t0, t1, t2, t3, playTime);
97 }
98 //Angular quantities
99 Vector3 dummy;
100 interpTrans.setRotation(slerp(T1.getRotation(), T2.getRotation(), (playTime-t1)/(t2-t1)));
101 calculateVelocityShortestPath(T1, T2, t2-t1, dummy, interpAngVel);
102
103 if(!forward)
104 {
105 interpVel = -interpVel;
106 interpAngVel = -interpAngVel;
107 }
108 }
109}

111void CRTrajectory::BuildGraphicalPath()
112{
113 PWLTrajectory::BuildGraphicalPath();
114
115 if(points.size() >= 3)
116 {
117 vis[1].points.clear();
118 for(size_t i=0; i<points.size()-1; ++i)
119 {
120 Vector3 P1 = points[i].T.getOrigin();
121 Vector3 P2 = points[i+1].T.getOrigin();
122 Scalar t1 = points[i].t;
123 Scalar t2 = points[i+1].t;
124 Scalar dist = (P2-P1).safeNorm();
125 if(btFuzzyZero(dist))
126 continue;
127
128 Vector3 P0, P3;
129 Scalar t0, t3;
130
131 if(i==0) //Beginning
132 {
133 P3 = points[2].T.getOrigin();
134 P0 = P1-(P2-P1);
135 t3 = points[2].t;
136 t0 = t1-(t2-t1);
137 }
138 else if(i==points.size()-2) //End
139 {
140 P0 = points[i-1].T.getOrigin();
141 P3 = P2+(P2-P1);
142 t0 = points[i-1].t;
143 t3 = t2+(t2-t1);
144 }
145 else //Middle
146 {
147 P0 = points[i-1].T.getOrigin();
148 P3 = points[i+2].T.getOrigin();
149 t0 = points[i-1].t;
150 t3 = points[i+2].t;
151 }
152
153 Scalar dt = (t2-t1)/Scalar(100.0);
154 for(Scalar t=t1; t<t2; t+=dt)
155 vis[1].points.push_back(glVectorFromVector(catmullRom(P0, P1, P2, P3, t0, t1, t2, t3, t)));
156 }
157 vis[1].points.push_back(glVectorFromVector(points.back().T.getOrigin()));
158 }
159}

161Vector3 CRTrajectory::catmullRom(Vector3 P0, Vector3 P1, Vector3 P2, Vector3 P3,
162 Scalar t0, Scalar t1, Scalar t2, Scalar t3, Scalar t)
163{
164 Vector3 A1 = (t1-t)/(t1-t0)*P0 + (t-t0)/(t1-t0)*P1;
165 Vector3 A2 = (t2-t)/(t2-t1)*P1 + (t-t1)/(t2-t1)*P2;
166 Vector3 A3 = (t3-t)/(t3-t2)*P2 + (t-t2)/(t3-t2)*P3;
167 Vector3 B1 = (t2-t)/(t2-t0)*A1 + (t-t0)/(t2-t0)*A2;
168 Vector3 B2 = (t3-t)/(t3-t1)*A2 + (t-t1)/(t3-t1)*A3;
169 Vector3 C = (t2-t)/(t2-t1)*B1 + (t-t1)/(t2-t1)*B2;
170 return C;
171}

173Vector3 CRTrajectory::catmullRomDerivative(Vector3 P0, Vector3 P1, Vector3 P2, Vector3 P3,
174 Scalar t0, Scalar t1, Scalar t2, Scalar t3, Scalar t)
175{
176 Vector3 A1 = (t1-t)/(t1-t0)*P0 + (t-t0)/(t1-t0)*P1;
177 Vector3 A2 = (t2-t)/(t2-t1)*P1 + (t-t1)/(t2-t1)*P2;
178 Vector3 A3 = (t3-t)/(t3-t2)*P2 + (t-t2)/(t3-t2)*P3;
179 Vector3 B1 = (t2-t)/(t2-t0)*A1 + (t-t0)/(t2-t0)*A2;
180 Vector3 B2 = (t3-t)/(t3-t1)*A2 + (t-t1)/(t3-t1)*A3;
181 Vector3 dA1 = Scalar(1)/(t1-t0)*(P1-P0);
182 Vector3 dA2 = Scalar(1)/(t2-t1)*(P2-P1);
183 Vector3 dA3 = Scalar(1)/(t3-t2)*(P3-P2);
184 Vector3 dB1 = Scalar(1)/(t2-t0)*(A2-A1) + (t2-t)/(t2-t0)*dA1 + (t-t0)/(t2-t0)*dA2;
185 Vector3 dB2 = Scalar(1)/(t3-t1)*(A3-A2) + (t3-t)/(t3-t1)*dA2 + (t-t1)/(t3-t1)*dA3;
186 Vector3 dC = Scalar(1)/(t2-t1)*(B2-B1) + (t2-t)/(t2-t1)*dB1 + (t-t1)/(t2-t1)*dB2;
187 return dC;
188}
