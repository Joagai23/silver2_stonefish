33Camera::Camera(std::string uniqueName, unsigned int resolutionX, unsigned int resolutionY, Scalar horizFOVDeg, Scalar frequency) : VisionSensor(uniqueName, frequency)
34{
35 fovH = horizFOVDeg <= Scalar(0) ? Scalar(90) : (horizFOVDeg > Scalar(360) ? Scalar(360) : horizFOVDeg);
36 resX = resolutionX > 0 ? (resolutionX + resolutionX % 2) : 2;
37 resY = resolutionY > 0 ? (resolutionY + resolutionY % 2) : 2;
38 setDisplayOnScreen(false, 0, 0, 1.f);
39}

72void Camera::UpdateTransform()
73{
74 Transform cameraTransform = getSensorFrame();
75 Vector3 eyePosition = cameraTransform.getOrigin(); //O
76 Vector3 direction = cameraTransform.getBasis().getColumn(2); //Z
77 Vector3 cameraUp = -cameraTransform.getBasis().getColumn(1); //-Y
78 SetupCamera(eyePosition, direction, cameraUp);
79}

81std::vector<Renderable> Camera::Render()
82{
83 std::vector<Renderable> items = Sensor::Render();
84 if(isRenderable())
85 {
86 Renderable item;
87 item.model = glMatrixFromTransform(getSensorFrame());
88 item.type = RenderableType::SENSOR_LINES;
89
90 //Create camera dummy
91 GLfloat iconSize = 0.5f;
92 GLfloat x = iconSize*tanf(fovH/360.f*M_PI);
93 GLfloat aspect = (GLfloat)resX/(GLfloat)resY;
94 GLfloat y = x/aspect;
95
96 item.points.push_back(glm::vec3(0,0,0));
97 item.points.push_back(glm::vec3(x, -y, iconSize));
98 item.points.push_back(glm::vec3(0,0,0));
99 item.points.push_back(glm::vec3(x, y, iconSize));
100 item.points.push_back(glm::vec3(0,0,0));
101 item.points.push_back(glm::vec3(-x, -y, iconSize));
102 item.points.push_back(glm::vec3(0,0,0));
103 item.points.push_back(glm::vec3(-x, y, iconSize));
104
105 item.points.push_back(glm::vec3(x, -y, iconSize));
106 item.points.push_back(glm::vec3(x, y, iconSize));
107 item.points.push_back(glm::vec3(x, y, iconSize));
108 item.points.push_back(glm::vec3(-x, y, iconSize));
109 item.points.push_back(glm::vec3(-x, y, iconSize));
110 item.points.push_back(glm::vec3(-x, -y, iconSize));
111 item.points.push_back(glm::vec3(-x, -y, iconSize));
112 item.points.push_back(glm::vec3(x, -y, iconSize));
113
114 item.points.push_back(glm::vec3(-0.5f*x, -y, iconSize));
115 item.points.push_back(glm::vec3(0.f, -1.5f*y, iconSize));
116 item.points.push_back(glm::vec3(0.f, -1.5f*y, iconSize));
117 item.points.push_back(glm::vec3(0.5f*x, -y, iconSize));
118
119 items.push_back(item);
120 }
121 return items;
122}
