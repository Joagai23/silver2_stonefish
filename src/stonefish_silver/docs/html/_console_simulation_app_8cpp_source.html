102//Static
103int ConsoleSimulationApp::RunSimulation(void* data)
104{
105 ConsoleSimulationApp& simApp = static_cast<ConsoleSimulationThreadData*>(data)->app;
106 SimulationManager* simManager = simApp.getSimulationManager();
107 simManager->setCallSimulationStepCompleted(simApp.timeStep_ == Scalar(0));
108
109 int maxThreads = std::max(omp_get_max_threads()/2, 1);
110 omp_set_num_threads(maxThreads);
111
112 while(simApp.getState() == SimulationState::RUNNING)
113 {
114 simApp.StepSimulation();
115 }
116
117 return 0;
118}
