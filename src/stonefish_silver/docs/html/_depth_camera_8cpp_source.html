36DepthCamera::DepthCamera(std::string uniqueName, unsigned int resolutionX, unsigned int resolutionY, Scalar hFOVDeg, Scalar minDepth, Scalar maxDepth, Scalar frequency)
37 : Camera(uniqueName, resolutionX, resolutionY, hFOVDeg, frequency)
38{
39 depthRange.x = minDepth < Scalar(0.01) ? 0.01f : (GLfloat)minDepth;
40 depthRange.y = maxDepth > Scalar(0.01) ? (GLfloat)maxDepth : 1.f;
41 noiseStdDev = 0.f;
42 newDataCallback = nullptr;

44 glCamera = nullptr;
45}

52void DepthCamera::setNoise(float depthStdDev)
53{
54 if(depthStdDev >= 0.f && depthStdDev != noiseStdDev)
55 {
56 noiseStdDev = depthStdDev;
57 if(glCamera != nullptr)
58 glCamera->setNoise(noiseStdDev);
59 }

82void DepthCamera::InitGraphics()
83{
84 glCamera = new OpenGLDepthCamera(glm::vec3(0,0,0), glm::vec3(0,0,1.f), glm::vec3(0,-1.f,0), 0, 0, resX, resY, (GLfloat)fovH, depthRange.x, depthRange.y, freq < Scalar(0));
85 glCamera->setNoise(noiseStdDev);
86 glCamera->setCamera(this);
87 UpdateTransform();
88 glCamera->UpdateTransform();
89 InternalUpdate(0);
90 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddView(glCamera);
91}

93void DepthCamera::SetupCamera(const Vector3& eye, const Vector3& dir, const Vector3& up)
94{
95 glm::vec3 eye_ = glm::vec3((GLfloat)eye.x(), (GLfloat)eye.y(), (GLfloat)eye.z());
96 glm::vec3 dir_ = glm::vec3((GLfloat)dir.x(), (GLfloat)dir.y(), (GLfloat)dir.z());
97 glm::vec3 up_ = glm::vec3((GLfloat)up.x(), (GLfloat)up.y(), (GLfloat)up.z());
98 glCamera->SetupCamera(eye_, dir_, up_);
99}
