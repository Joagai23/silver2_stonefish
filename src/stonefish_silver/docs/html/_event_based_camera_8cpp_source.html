83void EventBasedCamera::InitGraphics()
84{
85 glCamera = new OpenGLEventBasedCamera(glm::vec3(0,0,0), glm::vec3(0,0,1.f), glm::vec3(0,-1.f,0), 0, 0, resX, resY, (GLfloat)fovH,
86 depthRange, C, Tr, freq < Scalar(0));
87 glCamera->setNoise(sigmaC);
88 glCamera->setCamera(this);
89 UpdateTransform();
90 glCamera->UpdateTransform();
91 InternalUpdate(0);
92 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddView(glCamera);
93}

108void EventBasedCamera::NewDataReady(void* data, unsigned int index)
109{
110 lastEventCount = index;
111
112#ifdef DEBUG
113 if(lastEventCount > 0)
114 {
115 GLint* data_ = (GLint*)data;
116 int firstTime = INT32_MAX;
117 int lastTime = 0;
118 for(unsigned int i = 0; i < lastEventCount; ++i)
119 {
120 if(abs(data_[i*2+1]) > lastTime)
121 lastTime = abs(data_[i*2+1]);
122 if(abs(data_[i*2+1]) < firstTime)
123 firstTime = abs(data_[i*2+1]);
124 }
125 printf("First event time: %d, Last event time: %d\n", firstTime, lastTime);
126 }
127#endif
128
129 if(newDataCallback != nullptr)
130 {
131 imageData = (GLint*)data;
132 newDataCallback(this);
133 imageData = nullptr;
134 }
135}
