57int FeatherstoneRobot::getJoint(const std::string& name)
58{
59 if(dynamics == nullptr)
60 cCritical("Robot links not defined!");
61
62 for(unsigned int i=0; i<dynamics->getNumOfJoints(); ++i)
63 if(dynamics->getJointName(i) == name) return i;
64
65 return -1;
66}

96void FeatherstoneRobot::DefineLinks(SolidEntity* baseLink, std::vector<SolidEntity*> otherLinks, bool selfCollision)
97{
98 if(dynamics != nullptr)
99 cCritical("Robot cannot be redefined!");
100
101 links.push_back(baseLink);
102 detachedLinks = otherLinks;
103 dynamics = new FeatherstoneEntity(name + "_Dynamics", (unsigned short)detachedLinks.size() + 1, baseLink, fixed);
104 dynamics->setSelfCollision(selfCollision);
105}

107void FeatherstoneRobot::BuildKinematicStructure()
108{
109 cInfo("Building kinematic tree of robot '%s', consisting of %d links and %d joints.", getName().c_str(), detachedLinks.size()+1, jointsData.size());
110
111 //Sort joints
112 std::vector<JointData> sortedJoints;
113
114 //---Add joints connected to base
115 for(int i=(int)jointsData.size()-1; i>=0; --i)
116 {
117 if(jointsData[i].parent == links[0]->getName())
118 {
119 sortedJoints.push_back(jointsData[i]);
120 jointsData.erase(jointsData.begin() + i);
121 cInfo("Joint %ld: %s<-->%s", sortedJoints.size(),
122 sortedJoints.back().parent.c_str(),
123 sortedJoints.back().child.c_str());
124 }
125 }
126
127 //---Traverse through tree
128 size_t i0=0;
129 size_t i1=sortedJoints.size()-1;
130
131 while(jointsData.size() > 0)
132 {
133 for(size_t i=i0; i<=i1; ++i)
134 {
135 for(int h=(int)jointsData.size()-1; h>=0; --h)
136 {
137 if(jointsData[h].parent == sortedJoints[i].child)
138 {
139 sortedJoints.push_back(jointsData[h]);
140 jointsData.erase(jointsData.begin() + h);
141 cInfo("Joint %ld: %s<-->%s", sortedJoints.size(),
142 sortedJoints.back().parent.c_str(),
143 sortedJoints.back().child.c_str());
144 }
145 }
146 }
147 i0 = i1+1;
148 i1 = sortedJoints.size()-1;
149 }
150
151 jointsData = sortedJoints;
152
153 //Build kinematic tree
154 for(size_t i=0; i<jointsData.size(); ++i)
155 {
156 Transform linkTrans;
157
158 // Check if connected links are already part of the model (parallel mechanisms)
159 unsigned int parentId = UINT32_MAX;
160 unsigned int childId = UINT32_MAX;
161 for(size_t h=0; h<dynamics->getNumOfLinks(); ++h)
162 {
163 if(dynamics->getLink(h).solid->getName() == jointsData[i].parent)
164 parentId = h;
165 else if(dynamics->getLink(h).solid->getName() == jointsData[i].child)
166 childId = h;
167 }
168
169 if(parentId < dynamics->getNumOfLinks()
170 && childId < dynamics->getNumOfLinks()) // Kinematic loop
171 {
172 cCritical("Featherstone's algorithm does not support kinematic loops!");
173 }
174 else // Standard joint
175 {
176 if(parentId >= dynamics->getNumOfLinks())
177 cCritical("Parent link '%s' not yet joined with robot!", jointsData[i].parent.c_str());
178
179 // Find child link
180 for(size_t h=0; h<detachedLinks.size(); ++h)
181 if(detachedLinks[h]->getName() == jointsData[i].child)
182 childId = h;
183
184 if(childId >= detachedLinks.size())
185 {
186 cCritical("Child link '%s' doesn't exist!", jointsData[i].child.c_str());
187 }
188
189 // Add link
190 linkTrans = dynamics->getLinkTransform(parentId) * dynamics->getLink(parentId).solid->getCG2OTransform() * jointsData[i].origin;
191 dynamics->AddLink(detachedLinks[childId], linkTrans);
192 links.push_back(detachedLinks[childId]);
193 detachedLinks.erase(detachedLinks.begin()+childId);
194 childId = dynamics->getNumOfLinks()-1;
195 }
196
197 // Add joint
198 switch(jointsData[i].jtype)
199 {
200 case JointType::FIXED:
201 {
202 dynamics->AddFixedJoint(jointsData[i].name, parentId, childId, linkTrans.getOrigin());
203 }
204 break;
205
206 case JointType::REVOLUTE:
207 {
208 dynamics->AddRevoluteJoint(jointsData[i].name, parentId, childId, linkTrans.getOrigin(), linkTrans.getBasis() * jointsData[i].axis);
209 dynamics->AddJointLimit(dynamics->getNumOfJoints()-1, jointsData[i].posLim.first, jointsData[i].posLim.second);
210
211 if(jointsData[i].damping > Scalar(0))
212 {
213 dynamics->setJointDamping(dynamics->getNumOfJoints()-1, 0.0, jointsData[i].damping);
214 }
215 }
216 break;
217
218 case JointType::PRISMATIC:
219 {
220 dynamics->AddPrismaticJoint(jointsData[i].name, parentId, childId, linkTrans.getBasis() * jointsData[i].axis);
221 dynamics->AddJointLimit(dynamics->getNumOfJoints()-1, jointsData[i].posLim.first, jointsData[i].posLim.second);
222
223 if(jointsData[i].damping > Scalar(0))
224 {
225 dynamics->setJointDamping(dynamics->getNumOfJoints()-1, 0.0, jointsData[i].damping);
226 }
227 }
228 break;
229
230 default:
231 break;
232 }
233 }

250void FeatherstoneRobot::AddJointSensor(JointSensor* s, const std::string& monitoredJointName)
251{
252 int jointId = getJoint(monitoredJointName);
253 if(jointId > -1)
254 {
255 s->AttachToJoint(dynamics, jointId);
256 sensors.push_back(s);
257 }
258 else
259 cCritical("Joint '%s' doesn't exist. Sensor '%s' cannot be attached!", monitoredJointName.c_str(), s->getName().c_str());
260}

274void FeatherstoneRobot::AddLinkActuator(LinkActuator* a, const std::string& actuatedLinkName, const Transform& origin)
275{
276 int linkId = getLinkIndex(actuatedLinkName);
277 if(linkId < -1)
278 {
279 cCritical("Link '%s' doesn't exist. Actuator '%s' cannot be attached!", actuatedLinkName.c_str(), a->getName().c_str());
280 return;
281 }
282 if(a->getType() == ActuatorType::SUCTION_CUP) // Special case
283 {
284 static_cast<SuctionCup*>(a)->AttachToLink(getDynamics(), linkId);
285 }
286 else
287 {
288 a->AttachToSolid(getLink(actuatedLinkName), origin);
289 }
290 actuators.push_back(a);
291}
