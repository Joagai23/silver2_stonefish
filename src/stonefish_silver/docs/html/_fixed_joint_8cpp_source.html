37FixedJoint::FixedJoint(std::string uniqueName, SolidEntity* solid)
38 : Joint(uniqueName, false)
39{

51FixedJoint::FixedJoint(std::string uniqueName, SolidEntity* solidA, SolidEntity* solidB)
52 : Joint(uniqueName, false)
53{
54 btRigidBody* bodyA = solidA->rigidBody;
55 btRigidBody* bodyB = solidB->rigidBody;
56 Transform frameInA = Transform::getIdentity();
57 Transform frameInB = solidB->getCGTransform().inverse() * solidA->getCGTransform();
58
59 btFixedConstraint* fixed = new btFixedConstraint(*bodyA, *bodyB, frameInA, frameInB);
60 setConstraint(fixed);
61
62 jSolidA = solidA;
63 jSolidB = solidB;
64
65 cInfo("Fixed joint created between '%s' and '%s'.", jSolidA->getName().c_str(), jSolidB->getName().c_str());
66}

68FixedJoint::FixedJoint(std::string uniqueName, SolidEntity* solid, FeatherstoneEntity* fe, int linkId)
69 : Joint(uniqueName, false)
70{
71 Transform linkTransform = fe->getLinkTransform(linkId+1);
72 Transform solidTransform = solid->getCGTransform();
73 Vector3 pivotInA = V0();
74 Vector3 pivotInB = solidTransform.inverse() * linkTransform.getOrigin();
75 Matrix3 frameInA = Matrix3::getIdentity();
76 Matrix3 frameInB = solidTransform.getBasis().inverse() * linkTransform.getBasis();
77
78 btMultiBodyFixedConstraint* fixed = new btMultiBodyFixedConstraint(fe->getMultiBody(), linkId, solid->rigidBody, pivotInA, pivotInB, frameInA, frameInB);
79 fixed->setMaxAppliedImpulse(BT_LARGE_FLOAT);
80 setConstraint(fixed);
81
82 jSolidA = fe->getLink(linkId+1).solid;
83 jSolidB = solid;
84
85 cInfo("Fixed joint created between '%s' and '%s'.", jSolidA->getName().c_str(), jSolidB->getName().c_str());
86}

112void FixedJoint::UpdateDefinition()
113{
114 if(constraint != nullptr)
115 {
116 delete constraint;
117 if(jSolidA == nullptr)
118 {
119 setConstraint(new btFixedConstraint(*jSolidB->getRigidBody(), Transform::getIdentity()));
120 }
121 else
122 {
123 Transform frameInA = Transform::getIdentity();
124 Transform frameInB = jSolidB->getCGTransform().inverse() * jSolidA->getCGTransform();
125 setConstraint(new btFixedConstraint(*jSolidA->getRigidBody(), *jSolidB->getRigidBody(), frameInA, frameInB));
126 }
127 }
128 else if(mbConstraint != nullptr)
129 {
130 Transform linkATransform = jSolidA->getCGTransform();
131 Transform linkBTransform = jSolidB->getCGTransform();
132 Vector3 pivotInB = linkBTransform.inverse() * linkATransform.getOrigin();
133 Matrix3 frameInB = linkBTransform.getBasis().inverse() * linkATransform.getBasis();
134
135 btMultiBodyFixedConstraint* fix = static_cast<btMultiBodyFixedConstraint*>(mbConstraint);
136 fix->setPivotInB(pivotInB);
137 fix->setFrameInB(frameInB);
138 }
139}
