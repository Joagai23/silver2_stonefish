60void ForceTorque::InternalUpdate(Scalar dt)
61{
62 if(j != NULL && attach != NULL)
63 {
64 Vector3 force(j->getFeedback(0), j->getFeedback(1), j->getFeedback(2));
65 Vector3 torque(j->getFeedback(3), j->getFeedback(4), j->getFeedback(5));
66
67 if(j->isMultibodyJoint())
68 {
69 force /= dt;
70 torque /= dt;
71 }
72
73 lastFrame = attach->getOTransform() * o2s;
74 Matrix3 toSensor = lastFrame.getBasis().inverse();
75 force = toSensor * force;
76 torque = toSensor * torque;
77
78 Sample s{std::vector<Scalar>({force.getX(), force.getY(), force.getZ(), torque.getX(), torque.getY(), torque.getZ()})};
79 AddSampleToHistory(s);
80 }
81 else
82 {
83 Vector3 force, torque;
84 unsigned int childId = fe->getJointFeedback(jId, force, torque);
85 lastFrame = fe->getLink(childId).solid->getCG2OTransform() * o2s;
86 Matrix3 toSensor = lastFrame.getBasis().inverse();
87 force = toSensor * force;
88 torque = toSensor * torque;
89 lastFrame = fe->getLink(childId).solid->getCGTransform() * lastFrame; //From local to global
90
91 Sample s{std::vector<Scalar>({force.getX(), force.getY(), force.getZ(), torque.getX(), torque.getY(), torque.getZ()})};
92 AddSampleToHistory(s);
93 }
94}

96void ForceTorque::setRange(const Vector3& forceMax, const Vector3& torqueMax)
97{
98 channels[0].rangeMin = -btClamped(forceMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
99 channels[1].rangeMin = -btClamped(forceMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
100 channels[2].rangeMin = -btClamped(forceMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
101 channels[0].rangeMax = btClamped(forceMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
102 channels[1].rangeMax = btClamped(forceMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
103 channels[2].rangeMax = btClamped(forceMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
104
105 channels[3].rangeMin = -btClamped(torqueMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
106 channels[4].rangeMin = -btClamped(torqueMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
107 channels[5].rangeMin = -btClamped(torqueMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
108 channels[3].rangeMax = btClamped(torqueMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
109 channels[4].rangeMax = btClamped(torqueMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
110 channels[5].rangeMax = btClamped(torqueMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
111}
