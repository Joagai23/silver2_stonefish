46void GPS::InternalUpdate(Scalar dt)
47{
48 //get sensor frame in world
49 Transform gpsTrans = getSensorFrame();
50
51 //GPS not updating underwater
52 Ocean* liq = SimulationApp::getApp()->getSimulationManager()->getOcean();
53 if(liq != nullptr && liq->IsInsideFluid(gpsTrans.getOrigin()))
54 {
55 Sample s{std::vector<Scalar>({BT_LARGE_FLOAT, BT_LARGE_FLOAT, Scalar(0), Scalar(0)})};
56 AddSampleToHistory(s);
57 }
58 else
59 {
60 Vector3 gpsPos = gpsTrans.getOrigin();
61
62 //add noise
63 if(!btFuzzyZero(nedStdDev))
64 {
65 gpsPos.setX(gpsPos.x() + noise(randomGenerator));
66 gpsPos.setY(gpsPos.y() + noise(randomGenerator));
67 }
68
69 //convert NED to geodetic coordinates
70 double latitude;
71 double longitude;
72 double height;
73 SimulationApp::getApp()->getSimulationManager()->getNED()->Ned2Geodetic(gpsPos.x(), gpsPos.y(), 0.0, latitude, longitude, height);
74
75 //record sample
76 Sample s{std::vector<Scalar>({latitude, longitude, gpsPos.x(), gpsPos.y()})};
77 AddSampleToHistory(s);
78 }
79}

81void GPS::setNoise(Scalar nedDev)
82{
83 nedStdDev = btClamped(nedDev, Scalar(0), Scalar(BT_LARGE_FLOAT));
84 noise = std::normal_distribution<Scalar>(Scalar(0), nedStdDev);
85}
