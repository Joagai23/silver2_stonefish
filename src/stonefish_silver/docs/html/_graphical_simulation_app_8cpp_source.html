182void GraphicalSimulationApp::Init()
183{
184 //General initialization
185 SimulationApp::Init();
186 //Window initialization + loading thread
187 loading = true;
188 InitializeSDL();
189
190 //Continue initialization with console visible
191 cInfo("Initializing rendering pipeline:");
192 cInfo("Loading GUI...");
193 gui = new IMGUI(windowW, windowH);
194 InitializeGUI(); //Initialize non-standard graphical elements
195 glPipeline = new OpenGLPipeline(rSettings, hSettings);
196 ShowHUD();
197
198 cInfo("Initializing simulation:");
199 InitializeSimulation();
200
201 cInfo("Ready for running...");
202 SDL_Delay(1000);
203
204 //Close loading console - exit loading thread
205 loading = false;
206 int status = 0;
207 SDL_WaitThread(loadingThread, &status);
208 SDL_GL_MakeCurrent(window, glMainContext);
209
210 //Create performance counters
211 glGenQueries(2, timeQuery);
212
213 state_ = SimulationState::STOPPED;
214}

216void GraphicalSimulationApp::InitializeSDL()
217{
218 SDL_Init(SDL_INIT_VIDEO | SDL_INIT_EVENTS | SDL_INIT_JOYSTICK);
219
220 //Create OpenGL contexts
221 SDL_GL_SetAttribute(SDL_GL_CONTEXT_PROFILE_MASK, SDL_GL_CONTEXT_PROFILE_CORE);
222 SDL_GL_SetAttribute(SDL_GL_CONTEXT_MAJOR_VERSION, 4);
223 SDL_GL_SetAttribute(SDL_GL_CONTEXT_MINOR_VERSION, 3);
224 SDL_GL_SetAttribute(SDL_GL_DOUBLEBUFFER, 1);
225 SDL_GL_SetAttribute(SDL_GL_DEPTH_SIZE, 24);
226 SDL_GL_SetAttribute(SDL_GL_STENCIL_SIZE, 8);
227 SDL_GL_SetAttribute(SDL_GL_RED_SIZE, 8);
228 SDL_GL_SetAttribute(SDL_GL_GREEN_SIZE, 8);
229 SDL_GL_SetAttribute(SDL_GL_BLUE_SIZE, 8);
230 SDL_GL_SetAttribute(SDL_GL_SHARE_WITH_CURRENT_CONTEXT, 1);
231
232 //Create window
233 window = SDL_CreateWindow(getName().c_str(),
234 SDL_WINDOWPOS_CENTERED,
235 SDL_WINDOWPOS_CENTERED,
236 windowW,
237 windowH,
238 SDL_WINDOW_OPENGL | SDL_WINDOW_SHOWN// | SDL_WINDOW_ALLOW_HIGHDPI
239 );
240
241 //Set window icon
242 uint32_t rmask, gmask, bmask, amask;
243#if SDL_BYTEORDER == SDL_BIG_ENDIAN
244 int shift = (icon_image.bytes_per_pixel == 3) ? 8 : 0;
245 rmask = 0xff000000 >> shift;
246 gmask = 0x00ff0000 >> shift;
247 bmask = 0x0000ff00 >> shift;
248 amask = 0x000000ff >> shift;
249#else
250 rmask = 0x000000ff;
251 gmask = 0x0000ff00;
252 bmask = 0x00ff0000;
253 amask = (icon_image.bytes_per_pixel == (3)) ? 0 : 0xff000000;
254#endif
255
256 SDL_Surface* icon = SDL_CreateRGBSurfaceFrom((void*)icon_image.pixel_data, icon_image.width, icon_image.height,
257 icon_image.bytes_per_pixel*8, icon_image.bytes_per_pixel*icon_image.width, rmask, gmask, bmask, amask);
258
259 SDL_SetWindowIcon(window, icon);
260 SDL_FreeSurface(icon);
261
262 //Create OpenGL contexts
263 glLoadingContext = SDL_GL_CreateContext(window);
264 if(glLoadingContext == nullptr)
265 cCritical("SDL2: %s", SDL_GetError());
266
267 glMainContext = SDL_GL_CreateContext(window);
268 if(glMainContext == nullptr)
269 cCritical("SDL2: %s", SDL_GetError());
270
271 //Disable vertical synchronization
272 if(!rSettings.verticalSync)
273 {
274 if(SDL_GL_SetSwapInterval(0) == -1)
275 cError("SDL2: %s", SDL_GetError());
276 }
277
278 //Initialize OpenGL function handlers
279 int version = gladLoadGL((GLADloadfunc) SDL_GL_GetProcAddress);
280 int vmajor = GLAD_VERSION_MAJOR(version);
281 int vminor = GLAD_VERSION_MINOR(version);
282 if(vmajor < 4 || (vmajor == 4 && vminor < 3))
283 cCritical("This program requires support for OpenGL 4.3, however OpenGL %d.%d was detected! Exiting...", vmajor, vminor);
284
285 //Initialize OpenGL pipeline
286 cInfo("Window created. OpenGL %d.%d contexts created.", vmajor, vminor);
287 OpenGLState::Init();
288 GLSLShader::Init();
289
290 //Initialize console output
291 std::vector<ConsoleMessage> textLines = console_->getLines();
292 delete console_;
293 console_ = new OpenGLConsole();
294 for(size_t i=0; i<textLines.size(); ++i)
295 console_->AppendMessage(textLines[i]);
296 ((OpenGLConsole*)console_)->Init(windowW, windowH);
297
298 //Create loading thread
299 GraphicalSimulationThreadData* data = new GraphicalSimulationThreadData{*this};
300 loadingThread = SDL_CreateThread(GraphicalSimulationApp::RenderLoadingScreen, "loadingThread", data);
301
302 //Look for joysticks
303 int jcount = SDL_NumJoysticks();
304
305 if(jcount > 0)
306 {
307 joystick = SDL_JoystickOpen(0);
308 joystickButtons = new bool[SDL_JoystickNumButtons(joystick)];
309 memset(joystickButtons, 0, SDL_JoystickNumButtons(joystick));
310 joystickAxes = new int16_t[SDL_JoystickNumAxes(joystick)];
311 memset(joystickAxes, 0, SDL_JoystickNumAxes(joystick) * sizeof(int16_t));
312 joystickHats = new uint8_t[SDL_JoystickNumHats(joystick)];
313 memset(joystickHats, 0, SDL_JoystickNumHats(joystick));
314 cInfo("Joystick %s connected (%d axes, %d hats, %d buttons)", SDL_JoystickName(joystick),
315 SDL_JoystickNumAxes(joystick),
316 SDL_JoystickNumHats(joystick),
317 SDL_JoystickNumButtons(joystick));
318 }
319}

338void GraphicalSimulationApp::KeyDown(SDL_Event *event)
339{
340 GLfloat moveStep = 0.1f;
341 if(event->key.keysym.mod & KMOD_SHIFT)
342 moveStep = 1.f;
343
344 switch (event->key.keysym.sym)
345 {
346 case SDLK_ESCAPE:
347 Quit();
348 break;
349
350 case SDLK_h:
351 displayHUD = !displayHUD;
352 break;
353
354 case SDLK_k:
355 displayKeymap = !displayKeymap;
356 break;
357
358 case SDLK_p:
359 displayPerformance = !displayPerformance;
360 break;
361
362 case SDLK_c:
363 displayConsole = !displayConsole;
364 ((OpenGLConsole*)console_)->ResetScroll();
365 break;
366
367 case SDLK_w: //Forward
368 {
369 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
370 if(trackball->isEnabled())
371 trackball->MoveCenter(trackball->GetLookingDirection() * moveStep);
372 }
373 break;
374
375 case SDLK_s: //Backward
376 {
377 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
378 if(trackball->isEnabled())
379 trackball->MoveCenter(-trackball->GetLookingDirection() * moveStep);
380 }
381 break;
382
383 case SDLK_a: //Left
384 {
385 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
386 if(trackball->isEnabled())
387 {
388 glm::vec3 axis = glm::cross(trackball->GetLookingDirection(), trackball->GetUpDirection());
389 trackball->MoveCenter(-axis * moveStep);
390 }
391 }
392 break;
393
394 case SDLK_d: //Right
395 {
396 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
397 if(trackball->isEnabled())
398 {
399 glm::vec3 axis = glm::cross(trackball->GetLookingDirection(), trackball->GetUpDirection());
400 trackball->MoveCenter(axis * moveStep);
401 }
402 }
403 break;
404
405 case SDLK_q: //Up
406 {
407 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
408 if(trackball->isEnabled())
409 trackball->MoveCenter(glm::vec3(0.f, 0.f, -moveStep));
410 }
411 break;
412
413 case SDLK_z: //Down
414 {
415 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
416 if(trackball->isEnabled())
417 trackball->MoveCenter(glm::vec3(0.f, 0.f, moveStep));
418 }
419 break;
420
421 default:
422 break;
423 }
424}

458void GraphicalSimulationApp::LoopInternal()
459{
460 SDL_Event event;
461 SDL_FlushEvents(SDL_FINGERDOWN, SDL_MULTIGESTURE);
462
463 while(SDL_PollEvent(&event))
464 {
465 switch(event.type)
466 {
467 case SDL_WINDOWEVENT:
468 WindowEvent(&event);
469 break;
470
471 case SDL_TEXTINPUT:
472 break;
473
474 case SDL_KEYDOWN:
475 {
476 gui->KeyDown(event.key.keysym.sym);
477 KeyDown(&event);
478 break;
479 }
480
481 case SDL_KEYUP:
482 {
483 gui->KeyUp(event.key.keysym.sym);
484 KeyUp(&event);
485 break;
486 }
487
488 case SDL_MOUSEBUTTONDOWN:
489 {
490 gui->MouseDown(event.button.x, event.button.y, event.button.button == SDL_BUTTON_LEFT);
491 mouseWasDown = event;
492 }
493 break;
494
495 case SDL_MOUSEBUTTONUP:
496 {
497 //GUI
498 gui->MouseUp(event.button.x, event.button.y, event.button.button == SDL_BUTTON_LEFT);
499
500 //Trackball
501 if(event.button.button == SDL_BUTTON_RIGHT || event.button.button == SDL_BUTTON_MIDDLE)
502 {
503 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
504 if(trackball->isEnabled())
505 trackball->MouseUp();
506 }
507
508 //Pass
509 MouseUp(&event);
510 }
511 break;
512
513 case SDL_MOUSEMOTION:
514 {
515 //GUI
516 gui->MouseMove(event.motion.x, event.motion.y);
517
518 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
519 if(trackball->isEnabled())
520 {
521 GLfloat xPos = (GLfloat)(event.motion.x-getWindowWidth()/2.f)/(GLfloat)(getWindowHeight()/2.f);
522 GLfloat yPos = -(GLfloat)(event.motion.y-getWindowHeight()/2.f)/(GLfloat)(getWindowHeight()/2.f);
523 trackball->MouseMove(xPos, yPos);
524 }
525
526 //Pass
527 MouseMove(&event);
528 }
529 break;
530
531 case SDL_MOUSEWHEEL:
532 {
533 if(displayConsole) //GUI
534 ((OpenGLConsole*)console_)->Scroll((GLfloat)-event.wheel.y);
535 else
536 {
537 //Trackball
538 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
539 if(trackball->isEnabled())
540 trackball->MouseScroll(event.wheel.y * -1.f);
541
542 //Pass
543 MouseScroll(&event);
544 }
545 }
546 break;
547
548 case SDL_JOYBUTTONDOWN:
549 joystickButtons[event.jbutton.button] = true;
550 JoystickDown(&event);
551 break;
552
553 case SDL_JOYBUTTONUP:
554 joystickButtons[event.jbutton.button] = false;
555 JoystickUp(&event);
556 break;
557
558 case SDL_QUIT:
559 {
560 if(state_ == SimulationState::RUNNING)
561 StopSimulation();
562
563 Quit();
564 }
565 break;
566 }
567 }
568
569 if(joystick != nullptr)
570 {
571 for(int i=0; i<SDL_JoystickNumAxes(joystick); i++)
572 joystickAxes[i] = SDL_JoystickGetAxis(joystick, i);
573
574 for(int i=0; i<SDL_JoystickNumHats(joystick); i++)
575 joystickHats[i] = SDL_JoystickGetHat(joystick, i);
576 }
577
578 ProcessInputs();
579 RenderLoop();
580
581 //workaround for checking if IMGUI is being manipulated
582 if(mouseWasDown.type == SDL_MOUSEBUTTONDOWN && !gui->isAnyActive())
583 {
584 OpenGLTrackball* trackball = getSimulationManager()->getTrackball();
585
586 if(trackball->isEnabled())
587 {
588 if(mouseWasDown.button.button == SDL_BUTTON_LEFT)
589 {
590 glm::vec3 eye = trackball->GetEyePosition();
591 glm::vec3 ray = trackball->Ray(mouseWasDown.button.x, mouseWasDown.button.y);
592 selectedEntity = getSimulationManager()->PickEntity(Vector3(eye.x, eye.y, eye.z), Vector3(ray.x, ray.y, ray.z));
593 }
594 else //RIGHT OR MIDDLE
595 {
596 GLfloat xPos = (GLfloat)(mouseWasDown.motion.x-getWindowWidth()/2.f)/(GLfloat)(getWindowHeight()/2.f);
597 GLfloat yPos = -(GLfloat)(mouseWasDown.motion.y-getWindowHeight()/2.f)/(GLfloat)(getWindowHeight()/2.f);
598 trackball->MouseDown(xPos, yPos, mouseWasDown.button.button == SDL_BUTTON_MIDDLE);
599 }
600 }
601
602 //Pass
603 MouseDown(&event);
604 }
605 mouseWasDown.type = SDL_LASTEVENT;
606
607 // Update FPS
608 uint64_t elapsedTime = GetTimeInMicroseconds() - startTime_;
609 startTime_ = GetTimeInMicroseconds();
610 double dt = std::min(elapsedTime/1000000.0 + 1e-6, 1.0); //in s, secure against large values at the beginning
611 constexpr double f = 1.0/60.0;
612 fps_ = f*(1.0/dt) + (1.0-f)*fps_;
613}

615void GraphicalSimulationApp::RenderLoop()
616{
617 //Do some updates
618 if(state_ != SimulationState::RUNNING)
619 {
620 getSimulationManager()->UpdateDrawingQueue();
621 }
622
623 //Rendering
624 glBeginQuery(GL_TIME_ELAPSED, timeQuery[timeQueryPingpong]);
625 glPipeline->Render(getSimulationManager());
626 glPipeline->DrawDisplay();
627
628 //GUI & Console
629 if(displayConsole)
630 {
631 gui->GenerateBackground();
632 ((OpenGLConsole*)console_)->Render(true);
633 }
634 else
635 {
636 if(displayHUD) //Draw immediate mode GUI
637 {
638 gui->GenerateBackground();
639 gui->Begin();
640 DoHUD();
641 gui->End();
642 }
643 else //Just draw logo in the corner
644 {
645 gui->Begin();
646 gui->End();
647 }
648 }
649 glEndQuery(GL_TIME_ELAPSED);
650
651 //Update drawing time
652 uint64_t drawTime;
653 glGetQueryObjectui64v(timeQuery[1-timeQueryPingpong], GL_QUERY_RESULT, &drawTime);
654 timeQueryPingpong = 1-timeQueryPingpong;
655 double dt = std::min(drawTime/1000000.0, 1000.0); //in ms
656 double f = 1.0/60.0;
657 drawingTime = f*dt + (1.0-f)*drawingTime;
658
659 //Update maximum drawing time
660 if(maxCounter >= 60)
661 {
662 maxDrawingTime = drawingTime;
663 maxCounter = 0;
664 }
665 else
666 {
667 maxDrawingTime = std::max(maxDrawingTime, dt);
668 ++maxCounter;
669 }
670
671 //glFinish(); //Ensure that the frame was fully rendered
672 SDL_GL_SwapWindow(window);
673}

1054int GraphicalSimulationApp::RenderLoadingScreen(void* data)
1055{
1056 //Get application
1057 GraphicalSimulationApp& app = static_cast<GraphicalSimulationThreadData*>(data)->app;
1058
1059 //Make drawing in this thread possible
1060 SDL_GL_MakeCurrent(app.window, app.glLoadingContext);
1061
1062 //Render loading screen
1063 glClearColor(0.2f, 0.2f, 0.2f, 0.0f);
1064 glScissor(0, 0, app.windowW, app.windowH);
1065 glViewport(0, 0, app.windowW, app.windowH);
1066 glDisable(GL_DEPTH_TEST);
1067 glDisable(GL_CULL_FACE);
1068 glEnable(GL_BLEND);
1069 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
1070
1071 GLuint vao;
1072 glGenVertexArrays(1, &vao);
1073 glBindVertexArray(vao);
1074 glEnableVertexAttribArray(0);
1075
1076 while(app.loading)
1077 {
1078 glClear(GL_COLOR_BUFFER_BIT);
1079
1080 //Lock to prevent adding lines to the console while rendering
1081 SDL_LockMutex(app.console_->getLinesMutex());
1082 static_cast<OpenGLConsole*>(app.console_)->Render(false);
1083 SDL_UnlockMutex(app.console_->getLinesMutex());
1084
1085 SDL_GL_SwapWindow(app.window);
1086 }
1087
1088 glBindVertexArray(0);
1089 glDeleteVertexArrays(1, &vao);
1090
1091 //Detach thread from GL context
1092 SDL_GL_MakeCurrent(app.window, nullptr);
1093 return 0;
1094}

1096int GraphicalSimulationApp::RunSimulation(void* data)
1097{
1098 GraphicalSimulationApp& simApp = static_cast<GraphicalSimulationThreadData*>(data)->app;
1099 SimulationManager* simManager = simApp.getSimulationManager();
1100
1101 simManager->setCallSimulationStepCompleted(simApp.timeStep_ == Scalar(0));
1102
1103 int maxThreads = std::max(omp_get_max_threads()/2, 1);
1104 omp_set_num_threads(maxThreads);
1105

1107 {
1108 simApp.StepSimulation();

1110
