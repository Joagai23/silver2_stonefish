34Gyroscope::Gyroscope(std::string uniqueName, Scalar frequency, int historyLength) : LinkSensor(uniqueName, frequency, historyLength)
35{
36 channels.push_back(SensorChannel("Angular velocity X", QuantityType::ANGULAR_VELOCITY));
37 channels.push_back(SensorChannel("Angular velocity Y", QuantityType::ANGULAR_VELOCITY));
38 channels.push_back(SensorChannel("Angular velocity Z", QuantityType::ANGULAR_VELOCITY));
39 bias = V0();
40}

42void Gyroscope::InternalUpdate(Scalar dt)
43{
44 //calculate transformation from global to gyro frame
45 Matrix3 toGyroFrame = getSensorFrame().getBasis().inverse();
46
47 //get angular velocity
48 Vector3 omega = toGyroFrame * attach->getAngularVelocity();
49
50 //add bias error
51 omega += bias;
52
53 //record sample
54 Sample s{std::vector<Scalar>({omega.x(), omega.y(), omega.z()})};
55 AddSampleToHistory(s);
56}

58void Gyroscope::setRange(Vector3 angularVelocityMax)
59{
60 channels[0].rangeMin = -btClamped(angularVelocityMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
61 channels[1].rangeMin = -btClamped(angularVelocityMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
62 channels[2].rangeMin = -btClamped(angularVelocityMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
63 channels[0].rangeMax = btClamped(angularVelocityMax.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT));
64 channels[1].rangeMax = btClamped(angularVelocityMax.getY(), Scalar(0), Scalar(BT_LARGE_FLOAT));
65 channels[2].rangeMax = btClamped(angularVelocityMax.getZ(), Scalar(0), Scalar(BT_LARGE_FLOAT));
66}

68void Gyroscope::setNoise(Vector3 angularVelocityStdDev, Vector3 angularVelocityBias)
69{
70 channels[0].setStdDev(btClamped(angularVelocityStdDev.getX(), Scalar(0), Scalar(BT_LARGE_FLOAT)));
