75void INS::InternalUpdate(Scalar dt)
76{
77 Scalar now = SimulationApp::getApp()->getSimulationManager()->getSimulationTime(true);
78
79 //--- internal sensors
80 //get sensor frame in world
81 Transform imuTrans = getSensorFrame();
82 Vector3 R = imuTrans.getOrigin() - attach->getCGTransform().getOrigin();
83 //get angular velocity
84 Vector3 av = imuTrans.getBasis().inverse() * attach->getAngularVelocity();
85 //get acceleration
86 Vector3 acc = imuTrans.getBasis().inverse() * (
87 attach->getLinearAcceleration()
88 + attach->getAngularAcceleration().cross(R)
89 + attach->getAngularVelocity().cross(attach->getAngularVelocity().cross(R))
90 ); // NO GRAVITY
91 //get angular acceleration
92 Vector3 aacc = imuTrans.getBasis().inverse() * attach->getAngularAcceleration();
93
94 //noise
95 if(imuNoise)
96 {
97 av.setX(av.x() + avNoiseX(randomGenerator));
98 av.setY(av.y() + avNoiseY(randomGenerator));
99 av.setZ(av.z() + avNoiseZ(randomGenerator));
100 acc.setX(acc.x() + accNoiseX(randomGenerator));
101 acc.setY(acc.y() + accNoiseY(randomGenerator));
102 acc.setZ(acc.z() + accNoiseZ(randomGenerator));
103 }
104
105 //Predict (implicit Euler)
106 velocity += acc * dt; //In body frame (accumulated velocity)
107
108 //--- external sensors
109 DVL* dvl;
110 if(dvlName != "" && (dvl = (DVL*)SimulationApp::getApp()->getSimulationManager()->getSensor(dvlName)) != nullptr) //Correct velocities
111 {
112 Sample s = dvl->getLastSample();
113 Scalar ts = s.getTimestamp();
114 if(ts >= 0.0 && now-ts <= dt) //Is DVL valid?
115 {
116 Transform dvlTrans = dvl->getSensorFrame();
117 Vector3 avv = imuTrans.getBasis() * av; // Includes noise
118 velocity = imuTrans.getBasis().inverse()* (
119 dvlTrans.getBasis() * Vector3(s.getValue(0), s.getValue(1), s.getValue(2)) // Pure linear velocity component
120 + avv.cross(imuTrans.getOrigin() - dvlTrans.getOrigin()) // Angular velocity component
121 );
122 altitude = s.getValue(3);
123 }
124 }
125
126 Vector3 dp = velocity * dt; //In body frame
127 ned += imuTrans.getBasis() * dp; //In NED frame
128
129 GPS* gps;
130 if(gpsName != "" && (gps = (GPS*)SimulationApp::getApp()->getSimulationManager()->getSensor(gpsName)) != nullptr) //Correct global position
131 {
132 Sample s = gps->getLastSample();
133 Scalar ts = s.getTimestamp();
134 if(ts >= 0.0 && now-ts <= dt && s.getValue(0) <= Scalar(90) && s.getValue(1) <= Scalar(180)) //Is GPS valid?
135 {
136 Vector3 trans = imuTrans.getOrigin() - gps->getSensorFrame().getOrigin();
137 ned.setX(trans.x() + s.getValue(2));
138 ned.setY(trans.y() + s.getValue(3));
139 }
140 }
141
142 Pressure* press;
143 if(pressName != "" && (press = (Pressure*)SimulationApp::getApp()->getSimulationManager()->getSensor(pressName)) != nullptr) //Correct depth
144 {
145 Sample s = press->getLastSample();
146 Scalar ts = s.getTimestamp();
147 if(ts >= 0.0 && now-ts <= dt) //Is pressure valid?
148 {
149 Ocean* liq = SimulationApp::getApp()->getSimulationManager()->getOcean();
150 if(liq != NULL)
151 {
152 Scalar rho = liq->getLiquid().density;
153 Scalar g = SimulationApp::getApp()->getSimulationManager()->getGravity().z();
154 Scalar depth = s.getValue(0)/(rho*g);
155 ned.setZ(depth + imuTrans.getOrigin().z() - press->getSensorFrame().getOrigin().z() ); //Depth at INS frame
156 }
157 }
158 }
159
160 //transformed output
161 Vector3 nedo = ned + (imuTrans.getBasis() * out.getOrigin());
162 Vector3 velo = out.getBasis().inverse() * (velocity + av.cross(out.getOrigin()));
163 Vector3 avo = out.getBasis().inverse() * av;
164 Vector3 acco = out.getBasis().inverse() * (acc + aacc.cross(out.getOrigin()) + av.cross(av.cross(out.getOrigin())));
165
166 //compute geodetic position
167 Scalar height;
168 SimulationApp::getApp()->getSimulationManager()->getNED()->Ned2Geodetic(nedo.x(), nedo.y(), nedo.z(), latitude, longitude, height);
169
170 //get angles
171 Scalar yaw, pitch, roll;
172 (imuTrans * out).getBasis().getEulerYPR(yaw, pitch, roll);
173
174 //record sample
175 Sample s{std::vector<Scalar>(
176 {nedo.x(), nedo.y(), nedo.z(), altitude, latitude, longitude,
177 velo.x(), velo.y(), velo.z(), roll, pitch, yaw,
178 avo.x(), avo.y(), avo.z(), acco.x(), acco.y(), acco.z()}
179 )};
180 AddSampleToHistory(s); //Adds noise.....:(
181}

203void INS::setRange(Vector3 angularVelocityMax, Vector3 linearAccelerationMax)
204{
205 channels[12].rangeMin = -btClamped(angularVelocityMax.x(), Scalar(0), Scalar(BT_LARGE_FLOAT));
206 channels[13].rangeMin = -btClamped(angularVelocityMax.y(), Scalar(0), Scalar(BT_LARGE_FLOAT));
207 channels[14].rangeMin = -btClamped(angularVelocityMax.z(), Scalar(0), Scalar(BT_LARGE_FLOAT));
208 channels[12].rangeMax = btClamped(angularVelocityMax.x(), Scalar(0), Scalar(BT_LARGE_FLOAT));
209 channels[13].rangeMax = btClamped(angularVelocityMax.y(), Scalar(0), Scalar(BT_LARGE_FLOAT));
210 channels[14].rangeMax = btClamped(angularVelocityMax.z(), Scalar(0), Scalar(BT_LARGE_FLOAT));
211 channels[15].rangeMin = -btClamped(linearAccelerationMax.x(), Scalar(0), Scalar(BT_LARGE_FLOAT));
212 channels[16].rangeMin = -btClamped(linearAccelerationMax.y(), Scalar(0), Scalar(BT_LARGE_FLOAT));
213 channels[17].rangeMin = -btClamped(linearAccelerationMax.z(), Scalar(0), Scalar(BT_LARGE_FLOAT));
214 channels[15].rangeMax = btClamped(linearAccelerationMax.x(), Scalar(0), Scalar(BT_LARGE_FLOAT));
215 channels[16].rangeMax = btClamped(linearAccelerationMax.y(), Scalar(0), Scalar(BT_LARGE_FLOAT));
216 channels[17].rangeMax = btClamped(linearAccelerationMax.z(), Scalar(0), Scalar(BT_LARGE_FLOAT));

219void INS::setNoise(Vector3 angularVelocityStdDev, Vector3 linearAccelerationStdDev)
220{
221 avNoiseX = std::normal_distribution<Scalar>(Scalar(0), btClamped(angularVelocityStdDev.x(), Scalar(0), BT_LARGE_FLOAT));
222 avNoiseY = std::normal_distribution<Scalar>(Scalar(0), btClamped(angularVelocityStdDev.y(), Scalar(0), BT_LARGE_FLOAT));
223 avNoiseZ = std::normal_distribution<Scalar>(Scalar(0), btClamped(angularVelocityStdDev.z(), Scalar(0), BT_LARGE_FLOAT));
224
225 accNoiseX = std::normal_distribution<Scalar>(Scalar(0), btClamped(linearAccelerationStdDev.x(), Scalar(0), BT_LARGE_FLOAT));
226 accNoiseY = std::normal_distribution<Scalar>(Scalar(0), btClamped(linearAccelerationStdDev.y(), Scalar(0), BT_LARGE_FLOAT));
227 accNoiseZ = std::normal_distribution<Scalar>(Scalar(0), btClamped(linearAccelerationStdDev.z(), Scalar(0), BT_LARGE_FLOAT));
228
229 imuNoise = true;
230}

237std::vector<Renderable> INS::Render()
238{
239 std::vector<Renderable> items = LinkSensor::Render();
240 if(isRenderable())
241 {
242 Renderable item;
243 item.type = RenderableType::SENSOR_CS;
244 item.model = glMatrixFromTransform(getSensorFrame() * out);
245 items.push_back(item);
246
247 item.type = RenderableType::SENSOR_LINES;
248 item.model = glMatrixFromTransform(getSensorFrame());
249 item.points.push_back(glm::vec3(0.f));
250 item.points.push_back(glVectorFromVector(out.getOrigin()));
251 items.push_back(item);
252 }
253 return items;
254}
