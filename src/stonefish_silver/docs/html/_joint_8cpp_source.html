76Scalar Joint::getFeedback(unsigned int dof)
77{
78 if(dof > 5)
79 return Scalar(0);
80
81 if(constraint != nullptr)
82 {
83 btJointFeedback* fb = constraint->getJointFeedback();
84 if(dof < 3)
85 return fb->m_appliedForceBodyA[dof];
86 else
87 return fb->m_appliedTorqueBodyA[dof-3];
88 }
89 else if(mbConstraint != nullptr)
90 {
91 return mbConstraint->getAppliedImpulse(dof);
92 }
93 else
94 return Scalar(0);
95}

107void Joint::AddToSimulation(SimulationManager* sm)
108{
109 if(constraint != nullptr)
110 {
111 //Force feedback
112 btJointFeedback* fb = new btJointFeedback();
113 constraint->enableFeedback(true);
114 constraint->setJointFeedback(fb);
115
116 //Breaking
117 constraint->setBreakingImpulseThreshold(BT_LARGE_FLOAT);
118
119 //Solver setup
120 Scalar erp, stopErp;
121 sm->getJointErp(erp, stopErp);
122
123 if(constraint->getConstraintType() == D6_SPRING_2_CONSTRAINT_TYPE)
124 {
125 for(int i=0; i<6; ++i) // Go through all axes
126 {
127 constraint->setParam(BT_CONSTRAINT_ERP, erp, i);
128 constraint->setParam(BT_CONSTRAINT_STOP_ERP, stopErp, i);
129 constraint->setParam(BT_CONSTRAINT_CFM, 0.0, i);
130 constraint->setParam(BT_CONSTRAINT_STOP_CFM, 0.0, i);
131 }
132 }
133 else // Use default axis
134 {
135 constraint->setParam(BT_CONSTRAINT_ERP, erp);
136 constraint->setParam(BT_CONSTRAINT_STOP_ERP, stopErp); //Avoid explosion by softening the joint limits
137 constraint->setParam(BT_CONSTRAINT_CFM, 0.0);
138 constraint->setParam(BT_CONSTRAINT_STOP_CFM, 0.0);
139 }
140
141 //Add joint to dynamics world
142 sm->getDynamicsWorld()->addConstraint(constraint, !collisionEnabled);
143 }
144 else if(mbConstraint != nullptr)
145 {
146 Scalar erp, stopErp;
147 sm->getJointErp(erp, stopErp);
148 mbConstraint->setErp(erp);
149 sm->getDynamicsWorld()->addMultiBodyConstraint(mbConstraint);
150 if(!collisionEnabled)
151 SimulationApp::getApp()->getSimulationManager()->DisableCollision(jSolidA, jSolidB);
152 }
153}

155void Joint::RemoveFromSimulation(SimulationManager* sm)
156{
157 if(constraint != nullptr)
158 {
159 delete constraint->getJointFeedback();
160 sm->getDynamicsWorld()->removeConstraint(constraint);
161 }
162 else if(mbConstraint != nullptr)
163 {
164 sm->getDynamicsWorld()->removeMultiBodyConstraint(mbConstraint);
165 if(!collisionEnabled)
166 SimulationApp::getApp()->getSimulationManager()->EnableCollision(jSolidA, jSolidB);
167 }
168}
