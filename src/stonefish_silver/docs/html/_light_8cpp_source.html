40Light::Light(std::string uniqueName, Scalar radius, Color color, Scalar lum)
41 : LinkActuator(uniqueName), attach2(nullptr), attach3(nullptr), c(color), coneAngle(0), glLight(nullptr)
42{
43 if(!SimulationApp::getApp()->hasGraphics())
44 cCritical("Not possible to use lights in console simulation! Use graphical simulation if possible.");
45
46 R = radius < Scalar(0.01) ? Scalar(0.01) : radius;
47 Fi = lum < Scalar(0) ? Scalar(0) : lum;
48}

61Transform Light::getActuatorFrame() const
62{
63 if(attach != nullptr)
64 return attach->getOTransform() * o2a; //Solid
65 else if(attach2 != nullptr)
66 return attach2->getTransform() * o2a; //Static
67 else if(attach3 != nullptr)
68 return attach3->getOTransform() * o2a; //Animated
69 else
70 return o2a;
71}

121void Light::InitGraphics()
122{
123 if(coneAngle > Scalar(0)) //Spot light
124 glLight = new OpenGLSpotLight(glm::vec3(0.f), glm::vec3(0.f,0.f,-1.f), (GLfloat)R, (GLfloat)coneAngle, c.rgb, (GLfloat)Fi);
125 else //Omnidirectional light
126 glLight = new OpenGLPointLight(glm::vec3(0.f), (GLfloat)R, c.rgb, (GLfloat)Fi);
127
128 UpdateTransform();
129 glLight->UpdateTransform();
130 glLight->SwitchOn();
131 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddLight(glLight);
132}

138void Light::UpdateTransform()
139{
140 Transform lightTransform = getActuatorFrame();
141 Vector3 pos = lightTransform.getOrigin();
142 glm::vec3 glPos((GLfloat)pos.x(), (GLfloat)pos.y(), (GLfloat)pos.z());
143 glLight->UpdatePosition(glPos);
144
145 if(coneAngle > Scalar(0))
146 {
147 Vector3 dir = lightTransform.getBasis().getColumn(2);
148 glm::vec3 glDir((GLfloat)dir.x(), (GLfloat)dir.y(), (GLfloat)dir.z());
149 ((OpenGLSpotLight*)glLight)->UpdateDirection(glDir);
150 }
151}

153std::vector<Renderable> Light::Render()
154{
155 std::vector<Renderable> items(0);
156
157 Renderable item;
158 item.model = glMatrixFromTransform(getActuatorFrame());
159 item.type = RenderableType::ACTUATOR_LINES;
160
161 GLfloat iconSize = 1.f;
162 unsigned int div = 24;
163
164 if(coneAngle > Scalar(0))
165 {
166 GLfloat r = iconSize * tanf((GLfloat)coneAngle/360.f*M_PI);
167
168 for(unsigned int i=0; i<div; ++i)
169 {
170 GLfloat angle1 = (GLfloat)i/(GLfloat)div * 2.f * M_PI;
171 GLfloat angle2 = (GLfloat)(i+1)/(GLfloat)div * 2.f * M_PI;
172 item.points.push_back(glm::vec3(r * cosf(angle1), r * sinf(angle1), iconSize));
173 item.points.push_back(glm::vec3(r * cosf(angle2), r * sinf(angle2), iconSize));
174 }
175
176 item.points.push_back(glm::vec3(0,0,0));
177 item.points.push_back(glm::vec3(r, 0, iconSize));
178 item.points.push_back(glm::vec3(0,0,0));
179 item.points.push_back(glm::vec3(-r, 0, iconSize));
180 item.points.push_back(glm::vec3(0,0,0));
181 item.points.push_back(glm::vec3(0, r, iconSize));
182 item.points.push_back(glm::vec3(0,0,0));
183 item.points.push_back(glm::vec3(0, -r, iconSize));
184 }
185 else
186 {
187 for(unsigned int i=0; i<div; ++i)
188 {
189 GLfloat angle1 = (GLfloat)i/(GLfloat)div * 2.f * M_PI;
190 GLfloat angle2 = (GLfloat)(i+1)/(GLfloat)div * 2.f * M_PI;
191 item.points.push_back(glm::vec3(0.5f * iconSize * cosf(angle1), 0.5f * iconSize * sinf(angle1), 0));
192 item.points.push_back(glm::vec3(0.5f * iconSize * cosf(angle2), 0.5f * iconSize * sinf(angle2), 0));
193 item.points.push_back(glm::vec3(0.5f * iconSize * cosf(angle1), 0, 0.5f * iconSize * sinf(angle1)));
194 item.points.push_back(glm::vec3(0.5f * iconSize * cosf(angle2), 0, 0.5f * iconSize * sinf(angle2)));
195 item.points.push_back(glm::vec3(0, 0.5f * iconSize * cosf(angle1), 0.5f * iconSize * sinf(angle1)));
196 item.points.push_back(glm::vec3(0, 0.5f * iconSize * cosf(angle2), 0.5f * iconSize * sinf(angle2)));
197 }
198 }
199
200 items.push_back(item);
201
202 return items;
203}
