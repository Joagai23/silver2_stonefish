68 struct MaterialPair
69 {
70 int mat1Id;
71 int mat2Id;
72
73 bool operator==(const MaterialPair& rhs) const
74 {
75 return (mat1Id == rhs.mat1Id && mat2Id == rhs.mat2Id)||(mat1Id == rhs.mat2Id && mat2Id == rhs.mat1Id);
76 }
77 };
78
80 struct MaterialPairHash
81 {
82 std::size_t operator()(const MaterialPair& mp) const
83 {
84 if(mp.mat1Id <= mp.mat2Id)
85 return std::hash<int>()(mp.mat1Id) ^ (std::hash<int>()(mp.mat2Id) << 16);
86 else
87 return std::hash<int>()(mp.mat2Id) ^ (std::hash<int>()(mp.mat1Id) << 16);
88 }
89 };
