51void Motor::setIntensity(Scalar tau)
52{
53 torque = tau;
54 if(limits.second > limits.first) // Limitted
55 torque = tau < limits.first ? limits.first : (tau > limits.second ? limits.second : tau);
56 else
57 torque = tau;
58 ResetWatchdog();
59}

66Scalar Motor::getAngle() const
67{
68 if(j != nullptr && j->getType() == JointType::REVOLUTE)
69 {
70 return ((RevoluteJoint*)j)->getAngle();
71 }
72 else if(fe != nullptr)
73 {
74 Scalar angle;
75 btMultibodyLink::eFeatherstoneJointType jt = btMultibodyLink::eInvalid;
76 fe->getJointPosition(jId, angle, jt);
77
78 if(jt == btMultibodyLink::eRevolute)
79 return angle;
80 else
81 return Scalar(0);
82 }
83 else
84 return Scalar(0);
85}

87Scalar Motor::getAngularVelocity() const
88{

93 else if(fe != nullptr)
94 {
95 Scalar angularV;
96 btMultibodyLink::eFeatherstoneJointType jt = btMultibodyLink::eInvalid;
97 fe->getJointVelocity(jId, angularV, jt);
98
99 if(jt == btMultibodyLink::eRevolute)
100 return angularV;
101 else
102 return Scalar(0);
103 }
104 else
105 return Scalar(0);

108void Motor::Update(Scalar dt)
109{
110 Actuator::Update(dt);
111
112 if(j != nullptr && j->getType() == JointType::REVOLUTE)
113 ((RevoluteJoint*)j)->ApplyTorque(torque);
114 else if(fe != nullptr)
115 fe->DriveJoint(jId, torque);
116}
