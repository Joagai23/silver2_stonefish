36Multibeam2::Multibeam2(std::string uniqueName, unsigned int horizontalRes, unsigned int verticalRes, Scalar horizontalFOVDeg, Scalar verticalFOVDeg,
37 Scalar minRange, Scalar maxRange, Scalar frequency) : Camera(uniqueName, horizontalRes, verticalRes, horizontalFOVDeg, frequency)
38{
39 fovV = verticalFOVDeg > Scalar(0) ? verticalFOVDeg : Scalar(90);
40 range.x = minRange < Scalar(0.01) ? 0.01f : (GLfloat)minRange;
41 range.y = maxRange > Scalar(0.01) ? (GLfloat)maxRange : 1.f;
42 newDataCallback = NULL;

46 rangeData = new GLfloat[resX*resY]; // Buffer for storing final data
47 memset(rangeData, 0, resX*resY*sizeof(GLfloat));
48}

95void Multibeam2::InitGraphics()
96{
97 if(fovH <= Scalar(MULTIBEAM_MAX_SINGLE_FOV))
98 {
99 CamData cd;
100 cd.cam = NULL;
101 cd.fovH = (GLfloat)fovH;
102 cd.width = resX;
103 cd.dataOffset = 0;
104 cameras.push_back(cd);
105 }
106 else
107 {
108 //Find number of cameras needed not to exceed max FOV for a single camera
109 int nCam = (int)ceil(fovH/Scalar(MULTIBEAM_MAX_SINGLE_FOV));
110 Scalar fovH1 = fovH/Scalar(nCam);
111
112 //Check if resolution in X can be divided evenly for this number of cameras
113 int resmod = resX % nCam;
114 if(resmod == 0)
115 {
116 for(int i=0; i<nCam; ++i)
117 {
118 CamData cd;
119 cd.cam = NULL;
120 cd.fovH = (GLfloat)fovH1;
121 cd.width = resX/nCam;
122 cd.dataOffset = 0;
123 cameras.push_back(cd);
124 }
125 }
126 else
127 {
128 //Correct FOV and resolution of the cameras
129 int resX1 = resX/nCam;
130 int resXc = resX1 + resmod;
131 Scalar fovHc = fovH/(Scalar(1) + (nCam-1)*Scalar(resX1)/Scalar(resXc));
132 fovH1 = fovHc * Scalar(resX1)/Scalar(resXc);
133
134 CamData cd;
135 cd.cam = NULL;
136 cd.dataOffset = 0;
137
138 cd.fovH = (GLfloat)fovHc;
139 cd.width = resXc;
140 cameras.push_back(cd);
141
142 cd.fovH = (GLfloat)fovH1;
143 cd.width = resX1;
144 for(int i=0; i<nCam-1; ++i) cameras.push_back(cd);
145 }
146 }
147
148 //Create depth cameras
149 GLint accResX = 0;
150 for(size_t i=0; i<cameras.size(); ++i)
151 {
152 cameras[i].cam = new OpenGLDepthCamera(glm::vec3(0,0,0), glm::vec3(0,0,1.f), glm::vec3(0,-1.f,0),
153 accResX, 0, cameras[i].width, resY, cameras[i].fovH, range.x, range.y, true, (GLfloat)fovV);
154 cameras[i].cam->setCamera(this, (unsigned int)i);
155 cameras[i].dataOffset = accResX*resY;
156 accResX += cameras[i].width;
157 }
158
159 //Update camera transformations
160 UpdateTransform();
161
162 for(size_t i=0; i<cameras.size(); ++i)
163 {
164 cameras[i].cam->UpdateTransform();
165 cameras[i].cam->Update();
166 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddView(cameras[i].cam);
167 }
168}

176void Multibeam2::UpdateTransform()
177{
178 Transform mbTransform = getSensorFrame();
179 Vector3 eyePosition = mbTransform.getOrigin(); //O
180 Vector3 direction = mbTransform.getBasis().getColumn(2); //Z
181 Vector3 cameraUp = -mbTransform.getBasis().getColumn(1); //-Y
182 Scalar accFov(0);
183 Scalar offset = fovH/Scalar(360)*M_PI;
184
185 for(size_t i=0; i<cameras.size(); ++i)
186 {
187 //Calculate i-th camera transform
188 Scalar halfFov = cameras[i].fovH/Scalar(360)*M_PI;
189 Vector3 dir = direction.rotate(cameraUp, offset - accFov - halfFov);
190 accFov += Scalar(2)*halfFov;
191
192 //Setup camera transformation
193 SetupCamera(i, eyePosition, dir, cameraUp);
194 }
195}

214void Multibeam2::NewDataReady(void* data, unsigned int index)
215{
216 if(index >= cameras.size())
217 return;
218
219 memcpy(getImageDataPointer(index), data, cameras[index].width * resY * sizeof(GLfloat));
220 dataCounter += index;
221 int lastIndex = (int)cameras.size()-1;
222 int nSum = lastIndex*(lastIndex+1)/2;
223
224 if(dataCounter == nSum) //Check if data received from all cameras
225 {
226 dataCounter = 0;
227
228 //Copy (rearange) data from temp to final sonar image
229 for(size_t i=0; i<cameras.size(); ++i)
230 {
231 size_t xoffset = cameras[i].dataOffset/resY;
232
233 for(size_t h=0; h<resY; ++h)
234 {
235 memcpy(&rangeData[xoffset + h*resX],
236 &imageData[cameras[i].dataOffset + h*cameras[i].width],
237 sizeof(GLfloat)*cameras[i].width);
238 }
239 }
240
241 //Call callback
242 if(newDataCallback != NULL)
243 newDataCallback(this);
244 }
245}

247std::vector<Renderable> Multibeam2::Render()
248{
249 std::vector<Renderable> items = Sensor::Render();
250 if(isRenderable())
251 {
252 Renderable item;
253 item.model = glMatrixFromTransform(getSensorFrame());
254 item.type = RenderableType::SENSOR_LINES;
255
256 unsigned int div = (unsigned int)ceil(fovH/5.0);
257 GLfloat iconSize = 0.5f;
258 GLfloat cosFovV2 = cosf(fovV/360.f*M_PI);
259 GLfloat sinFovV2 = sinf(fovV/360.f*M_PI);
260 GLfloat r = iconSize/cosFovV2;
261 GLfloat thetaDiv = fovH/180.f * M_PI/(GLfloat)div;
262 GLfloat offset = -fovH/360.f * M_PI;
263 GLfloat y = sinFovV2 * r;
264
265 for(unsigned int i=0; i<div; ++i)
266 {
267 GLfloat theta1 = i*thetaDiv + offset;
268 GLfloat theta2 = theta1 + thetaDiv;
269 GLfloat d1 = cosf(theta1) * r;
270 GLfloat d2 = cosf(theta2) * r;
271 GLfloat z1 = cosFovV2 * d1;
272 GLfloat z2 = cosFovV2 * d2;
273 GLfloat x1 = sinf(theta1) * r;
274 GLfloat x2 = sinf(theta2) * r;
275
276 item.points.push_back(glm::vec3(x1,y,z1));
277 item.points.push_back(glm::vec3(x2,y,z2));
278 item.points.push_back(glm::vec3(x1,-y,z1));
279 item.points.push_back(glm::vec3(x2,-y,z2));
280
281 if(i == 0) //End 1
282 {
283 item.points.push_back(glm::vec3(x1,y,z1));
284 item.points.push_back(glm::vec3(x1,-y,z1));
285 item.points.push_back(glm::vec3(x1,y,z1));
286 item.points.push_back(glm::vec3(0,0,0));
287 item.points.push_back(glm::vec3(x1,-y,z1));
288 item.points.push_back(glm::vec3(0,0,0));
289 }
290 else if(i == div-1) //End 2
291 {
292 item.points.push_back(glm::vec3(x2,y,z2));
293 item.points.push_back(glm::vec3(x2,-y,z2));
294 item.points.push_back(glm::vec3(x2,y,z2));
295 item.points.push_back(glm::vec3(0,0,0));
296 item.points.push_back(glm::vec3(x2,-y,z2));
297 item.points.push_back(glm::vec3(0,0,0));
298 }
299 }
300
301 items.push_back(item);
302 }
303 return items;
304}
