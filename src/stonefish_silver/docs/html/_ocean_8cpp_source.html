41Ocean::Ocean(std::string uniqueName, Scalar waves, Fluid l) : ForcefieldEntity(uniqueName)
42{
43 ghost->setCollisionFlags(ghost->getCollisionFlags() | btCollisionObject::CF_STATIC_OBJECT);
44 oceanState = waves > Scalar(2.0) ? Scalar(2.0) : waves;
45
46 Scalar size(100000);
47 depth = size;
48 Vector3 halfExtents = Vector3(size/Scalar(2), size/Scalar(2), size/Scalar(2));
49 ghost->setWorldTransform(Transform(Quaternion::getIdentity(), Vector3(0, 0, size/Scalar(2) - oceanState*Scalar(3)))); //Move ocean influence zone a bit up to account for waves
50 ghost->setCollisionShape(new btBoxShape(halfExtents));
51
52 currents = std::vector<VelocityField*>(0);
53 currentsEnabled = false;
54
55 liquid = l;
56 wavesDebug.type = RenderableType::HYDRO_POINTS;
57 wavesDebug.model = glm::mat4(1.f);
58 waterType = Scalar(0.0);
59 glOcean = nullptr;
60}

147float Ocean::GetDepth(const glm::vec3& point)
148{
149 if(hasWaves()) //Geometric waves
150 {
151 GLfloat waveHeight = glOcean->ComputeWaveHeight(point.x, point.y);
152 glm::vec3 wavePoint(point.x, point.y, waveHeight);
153#ifdef DEBUG_WAVES
154 wavesDebug.points.push_back(wavePoint);
155#endif
156 return point.z - waveHeight;
157 }
158 else //Flat surface
159 {
160 glm::vec3 wavePoint(point.x, point.y, 0.f);
161#ifdef DEBUG_WAVES
162 wavesDebug.points.push_back(wavePoint);
163#endif
164 return point.z;
165 }
166}

173Scalar Ocean::GetPressure(const Vector3& point)
174{
175 Scalar g = SimulationApp::getApp()->getSimulationManager()->getGravity().getZ();
176 Scalar d = GetDepth(point);
177 Scalar pressure = d > Scalar(0) ? d*liquid.density*g : Scalar(0);
178 return pressure;
179}

217void Ocean::ApplyFluidForces(btDynamicsWorld* world, btCollisionObject* co, bool recompute)
218{
219 Entity* ent;
220 btRigidBody* rb = btRigidBody::upcast(co);
221 btMultiBodyLinkCollider* mbl = btMultiBodyLinkCollider::upcast(co);
222
223 if(rb != 0)
224 {
225 if(rb->isStaticOrKinematicObject())
226 return;
227 else
228 ent = (Entity*)rb->getUserPointer();
229 }
230 else if(mbl != 0)
231 {
232 if(mbl->isStaticOrKinematicObject())
233 return;
234 else
235 ent = (Entity*)mbl->getUserPointer();
236 }
237 else
238 return;
239
240 HydrodynamicsSettings settings;
241
242 if(ent->getType() == EntityType::SOLID)
243 {
244 if(recompute)
245 {
246 settings.dampingForces = true;
247 settings.reallisticBuoyancy = true;
248 ((SolidEntity*)ent)->ComputeHydrodynamicForces(settings, this);
249 }
250
251 ((SolidEntity*)ent)->ApplyHydrodynamicForces();
252 }

270std::vector<Renderable> Ocean::Render(const std::vector<Actuator*>& act)
271{
272 std::vector<Renderable> items(0);
273
274 //Update currents data
275 glOceanCurrentsUBOData.gravity = glm::vec3(0.f,0.f,9.81f);
276 glOceanCurrentsUBOData.numCurrents = 0;
277
278 if(currentsEnabled)
279 {
280 for(size_t i=0; i<currents.size(); ++i)
281 if(currents[i]->isEnabled())
282 {
283 std::vector<Renderable> citems = currents[i]->Render(glOceanCurrentsUBOData.currents[glOceanCurrentsUBOData.numCurrents]);
284 items.insert(items.end(), citems.begin(), citems.end());
285 ++glOceanCurrentsUBOData.numCurrents;
286 }
287 }
288
289 for(size_t i=0; i<act.size(); ++i)
290 if(act[i]->getType() == ActuatorType::THRUSTER)
291 {
292 Thruster* th = (Thruster*)act[i];
293 Transform thFrame = th->getActuatorFrame();
294 Vector3 thPos = thFrame.getOrigin();
295 Vector3 thDir = -thFrame.getBasis().getColumn(0);
296 Scalar R = th->getPropellerDiameter()/Scalar(2);
297 Scalar vel = (th->isPropellerRight() ? Scalar(0.1) : Scalar(-0.1)) * th->getThrust();
298 glOceanCurrentsUBOData.currents[glOceanCurrentsUBOData.numCurrents].posR = glm::vec4((GLfloat)thPos.getX(),
299 (GLfloat)thPos.getY(),
300 (GLfloat)thPos.getZ(), (GLfloat)R);
301 glOceanCurrentsUBOData.currents[glOceanCurrentsUBOData.numCurrents].dirV = glm::vec4((GLfloat)thDir.getX(),
302 (GLfloat)thDir.getY(),
303 (GLfloat)thDir.getZ(),
304 (GLfloat)vel);
305 glOceanCurrentsUBOData.currents[glOceanCurrentsUBOData.numCurrents].params = glm::vec3(0.f);
306 glOceanCurrentsUBOData.currents[glOceanCurrentsUBOData.numCurrents].type = 10;
307 ++glOceanCurrentsUBOData.numCurrents;
308 }
309
310 if(wavesDebug.points.size() > 0)
311 {
312 items.push_back(wavesDebug);
313 wavesDebug.points.clear();
314 }
315
316 return items;
317}
