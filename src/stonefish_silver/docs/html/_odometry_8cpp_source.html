53void Odometry::InternalUpdate(Scalar dt)
54{
55 //Calculate transformation from global to imu frame
56 Transform odomTrans = getSensorFrame();
57
58 Vector3 pos = odomTrans.getOrigin();
59 Vector3 v = odomTrans.getBasis().inverse() * attach->getLinearVelocityInLocalPoint(odomTrans.getOrigin() - attach->getCGTransform().getOrigin());
60
61 Quaternion orn = odomTrans.getRotation();
62 Scalar angle = orn.getAngle() + ornNoise(randomGenerator);
63 orn = Quaternion(orn.getAxis(), angle);
64
65 Vector3 av = odomTrans.getBasis().inverse() * attach->getAngularVelocity();
66
67 //Record sample
68 Sample s{std::vector<Scalar>({pos.x(), pos.y(), pos.z(), v.x(), v.y(), v.z(), orn.x(), orn.y(), orn.z(), orn.w(), av.x(), av.y(), av.z()})};
69 AddSampleToHistory(s);
70}

72void Odometry::setNoise(Scalar positionStdDev, Scalar velocityStdDev, Scalar angleStdDev, Scalar angularVelocityStdDev)
73{
74 channels[0].setStdDev(btClamped(positionStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
75 channels[1].setStdDev(btClamped(positionStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
76 channels[2].setStdDev(btClamped(positionStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
77 channels[3].setStdDev(btClamped(velocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
78 channels[4].setStdDev(btClamped(velocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
79 channels[5].setStdDev(btClamped(velocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
80 channels[10].setStdDev(btClamped(angularVelocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
81 channels[11].setStdDev(btClamped(angularVelocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
82 channels[12].setStdDev(btClamped(angularVelocityStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT)));
83 ornStdDev = btClamped(angleStdDev, Scalar(0), Scalar(BT_LARGE_FLOAT));
84 ornNoise = std::normal_distribution<Scalar>(Scalar(0), ornStdDev);
85}
