47OpenGLAtmosphere::OpenGLAtmosphere(const std::string& modelFilename, RenderQuality shadow)
48{
49 //Prepare for rendering
50 for(unsigned short i=0; i<AtmosphereTextures::TEXTURE_COUNT; ++i) textures[i] = 0;
51 skySunShaders[0] = nullptr;
52 skySunShaders[1] = nullptr;
53
54 //Init shadow baking
55 sunShadowmapShader = nullptr;
56 sunShadowmapArray = 0;
57 sunDepthSampler = 0;
58 sunShadowSampler = 0;
59 sunShadowmapSplits = 4;
60 sunShadowmapSize = 4096;
61 sunShadowFBO = 0;
62 sunSkyUBO = 0;
63 sunDirection = glm::vec3(0,0,1.f);
64 sunModelView = glm::mat4x4(0);
65 sunShadowFrustum = nullptr;
66 sunShadowCPM = nullptr;
67
68 //Set shadow quality
69 switch(shadow)
70 {
71 case RenderQuality::DISABLED:
72 sunShadowmapSize = 0;
73 sunShadowmapSplits = 0;
74 break;
75
76 case RenderQuality::LOW:
77 sunShadowmapSize = 1024;
78 sunShadowmapSplits = 4;
79 break;
80
81 case RenderQuality::MEDIUM:
82 sunShadowmapSize = 2048;
83 sunShadowmapSplits = 4;
84 break;
85
86 case RenderQuality::HIGH:
87 sunShadowmapSize = 4096;
88 sunShadowmapSplits = 4;
89 break;
90 }
91
92 //Load API and Data
93 memset(&sunSkyUBOData, 0, sizeof(SunSkyUBO));
94 LoadAtmosphereData(modelFilename);
95
96 //Set initial coditions
97 SetSunPosition(0.0, 45.0);
98 setAirTemperature(20.0);
99 setAirHumidity(0.5);
100
101 //Permanently bind atmosphere textures
102 OpenGLState::BindTexture(TEX_ATM_TRANSMITTANCE, GL_TEXTURE_2D, textures[AtmosphereTextures::TRANSMITTANCE]);
103 OpenGLState::BindTexture(TEX_ATM_SCATTERING, GL_TEXTURE_3D, textures[AtmosphereTextures::SCATTERING]);
104 OpenGLState::BindTexture(TEX_ATM_IRRADIANCE, GL_TEXTURE_2D, textures[AtmosphereTextures::IRRADIANCE]);
105
106 //Build rendering shaders
107 std::vector<GLuint> compiledShaders;
108 compiledShaders.push_back(atmosphereAPI);
109 std::vector<GLSLSource> sources;
110 sources.push_back(GLSLSource(GL_VERTEX_SHADER, "atmSkySun.vert"));
111 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "atmSkySun.frag"));
112
113 //Sky rendering
114 skySunShaders[0] = new GLSLShader(sources, compiledShaders);
115 skySunShaders[0]->AddUniform("transmittance_texture", ParameterType::INT);
116 skySunShaders[0]->AddUniform("scattering_texture", ParameterType::INT);
117 skySunShaders[0]->AddUniform("single_mie_scattering_texture", ParameterType::INT);
118 skySunShaders[0]->AddUniform("eyePos", ParameterType::VEC3);
119 skySunShaders[0]->AddUniform("sunDir", ParameterType::VEC3);
120 skySunShaders[0]->AddUniform("invView", ParameterType::MAT4);
121 skySunShaders[0]->AddUniform("invProj", ParameterType::MAT4);
122 skySunShaders[0]->AddUniform("whitePoint", ParameterType::VEC3);
123 skySunShaders[0]->AddUniform("cosSunSize", ParameterType::FLOAT);
124 skySunShaders[0]->AddUniform("bottomRadius", ParameterType::FLOAT);
125 skySunShaders[0]->AddUniform("groundAlbedo", ParameterType::FLOAT);
126
127 sources.pop_back();
128 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "atmSkySunTemp.frag"));
129 skySunShaders[1] = new GLSLShader(sources, compiledShaders);
130 skySunShaders[1]->AddUniform("transmittance_texture", ParameterType::INT);
131 skySunShaders[1]->AddUniform("scattering_texture", ParameterType::INT);
132 skySunShaders[1]->AddUniform("single_mie_scattering_texture", ParameterType::INT);
133 skySunShaders[1]->AddUniform("eyePos", ParameterType::VEC3);
134 skySunShaders[1]->AddUniform("sunDir", ParameterType::VEC3);
135 skySunShaders[1]->AddUniform("invView", ParameterType::MAT4);
136 skySunShaders[1]->AddUniform("invProj", ParameterType::MAT4);
137 skySunShaders[1]->AddUniform("whitePoint", ParameterType::VEC3);
138 skySunShaders[1]->AddUniform("cosSunSize", ParameterType::FLOAT);
139 skySunShaders[1]->AddUniform("bottomRadius", ParameterType::FLOAT);
140 skySunShaders[1]->AddUniform("groundAlbedo", ParameterType::FLOAT);
141 skySunShaders[1]->AddUniform("skyEmissivity", ParameterType::FLOAT);
142 skySunShaders[1]->AddUniform("airTemperature", ParameterType::FLOAT);
143
144 //Initialize shadows
145 sunShadowFrustum = new ViewFrustum[sunShadowmapSplits];
146 sunShadowCPM = new glm::mat4x4[sunShadowmapSplits];
147
148 //Create sun & sky UBO
149 glGenBuffers(1, &sunSkyUBO);
150 glBindBuffer(GL_UNIFORM_BUFFER, sunSkyUBO);
151 glBufferData(GL_UNIFORM_BUFFER, sizeof(SunSkyUBO), NULL, GL_STATIC_DRAW);
152 glBindBuffer(GL_UNIFORM_BUFFER, 0);
153 glBindBufferRange(GL_UNIFORM_BUFFER, UBO_SUNSKY, sunSkyUBO, 0, sizeof(SunSkyUBO));
154
155 if(shadow > RenderQuality::DISABLED)
156 {
157 sunShadowmapShader = new GLSLShader("sunCSM.frag");
158 sunShadowmapShader->AddUniform("shadowmapArray", ParameterType::INT);
159 sunShadowmapShader->AddUniform("shadowmapLayer", ParameterType::FLOAT);
160
161 //Generate shadowmap array
162 glGenTextures(1, &sunShadowmapArray);
163 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D_ARRAY, sunShadowmapArray);
164 glTexImage3D(GL_TEXTURE_2D_ARRAY, 0, GL_DEPTH_COMPONENT32F, sunShadowmapSize, sunShadowmapSize, sunShadowmapSplits, 0, GL_DEPTH_COMPONENT, GL_FLOAT, NULL);
165 OpenGLState::UnbindTexture(TEX_BASE);
166
167 //Generate samplers
168 glGenSamplers(1, &sunDepthSampler);
169 glSamplerParameteri(sunDepthSampler, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
170 glSamplerParameteri(sunDepthSampler, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
171 glSamplerParameteri(sunDepthSampler, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
172 glSamplerParameteri(sunDepthSampler, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
173
174 glGenSamplers(1, &sunShadowSampler);
175 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
176 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
177 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
178 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
179 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_COMPARE_MODE, GL_COMPARE_REF_TO_TEXTURE);
180 glSamplerParameteri(sunShadowSampler, GL_TEXTURE_COMPARE_FUNC, GL_LEQUAL);
181
182 //Create shadowmap framebuffer
183 glGenFramebuffers(1, &sunShadowFBO);
184 OpenGLState::BindFramebuffer(sunShadowFBO);
185 glFramebufferTextureLayer(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT, sunShadowmapArray, 0, 0);
186 glDrawBuffer(GL_NONE);
187 glReadBuffer(GL_NONE);
188
189 GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
190 if(status != GL_FRAMEBUFFER_COMPLETE)
191 cError("Sun shadow FBO initialization failed!");
192
193 OpenGLState::BindFramebuffer(0);
194
195 //Permanently bind shadow textures and samplers
196 OpenGLState::BindTexture(TEX_SUN_SHADOW, GL_TEXTURE_2D_ARRAY, sunShadowmapArray);
197 glBindSampler(TEX_SUN_SHADOW, sunShadowSampler);
198
199 OpenGLState::BindTexture(TEX_SUN_DEPTH, GL_TEXTURE_2D_ARRAY, sunShadowmapArray);
200 glBindSampler(TEX_SUN_DEPTH, sunDepthSampler);
201 }
202}

204OpenGLAtmosphere::~OpenGLAtmosphere()
205{
206 for(unsigned short i=0; i< AtmosphereTextures::TEXTURE_COUNT; ++i)
207 if(textures[i] != 0) glDeleteTextures(1, &textures[i]);
208
209 if(skySunShaders[0] != nullptr) delete skySunShaders[0];
210 if(skySunShaders[1] != nullptr) delete skySunShaders[1];
211 //if(atmosphereAPI > 0) glDeleteShader(atmosphereAPI);
212
213 if(sunShadowmapArray != 0) glDeleteTextures(1, &sunShadowmapArray);
214
215 delete [] sunShadowFrustum;
216 delete [] sunShadowCPM;
217 glDeleteBuffers(1, &sunSkyUBO);
218
219 if(sunShadowmapSize > 0)
220 {
221 glDeleteSamplers(1, &sunDepthSampler);
222 glDeleteSamplers(1, &sunShadowSampler);
223 glDeleteFramebuffers(1, &sunShadowFBO);
224 delete sunShadowmapShader;
225 }

228void OpenGLAtmosphere::SetSunPosition(float azimuthDeg, float elevationDeg)
229{
230 azimuthDeg = azimuthDeg < -180.f ? -180.f : (azimuthDeg > 180.f ? 180.f : azimuthDeg);
231 elevationDeg = elevationDeg < -90.f ? -90.f : (elevationDeg > 90.f ? 90.f : elevationDeg);
232
233 sunAzimuth = azimuthDeg;
234 sunElevation = elevationDeg;
235
236 //Calculate sun dir vector
237 float sunAzimuthAngle = -sunAzimuth/180.f*M_PI + M_PI;
238 float sunZenithAngle = (90.f - sunElevation)/180.f*M_PI;
239 sunDirection = glm::normalize(glm::rotate(glm::vec3(cos(sunAzimuthAngle) * sin(sunZenithAngle), sin(sunAzimuthAngle) * sin(sunZenithAngle), cos(sunZenithAngle)), (float)M_PI, glm::vec3(0,1.f,0)));
240
241 //Build sun modelview matrix
242 glm::vec3 up(0,0,-1.f);
243 glm::vec3 right = glm::cross(sunDirection, up);
244 right = glm::normalize(right);
245 up = glm::normalize(glm::cross(right, sunDirection));
246 sunModelView = glm::lookAt(glm::vec3(0,0,0), -sunDirection, up) * glm::mat4(1.f);
247}

287void OpenGLAtmosphere::LoadAtmosphereData(const std::string& filename)
288{
289 //Load parameters used to generate atmosphere data
290 glm::uvec3 transmittanceSize;
291 glm::uvec3 irradianceSize;
292 glm::uvec3 scatteringSize;
293 GLfloat lengthUnitInMeters;
294 GLfloat bottomRadius;
295#ifdef EMBEDDED_RESOURCES
296 ResourceHandle rh(filename);
297 std::istringstream dataString(rh.string());
298 std::istream& data(dataString);
299#else
300 std::ifstream dataFile(filename, std::ios::binary | std::ios::in);
301 std::istream& data(dataFile);
302#endif
303 data.read((char*)&transmittanceSize, sizeof(transmittanceSize));
304 data.read((char*)&irradianceSize, sizeof(irradianceSize));
305 data.read((char*)&scatteringSize, sizeof(scatteringSize));
306 data.read((char*)&lengthUnitInMeters, sizeof(lengthUnitInMeters));
307 data.read((char*)&bottomRadius, sizeof(bottomRadius));
308 data.read((char*)&sunSkyUBOData.whitePoint, sizeof(sunSkyUBOData.whitePoint));
309 data.read((char*)&nPrecomputedWavelengths, sizeof(nPrecomputedWavelengths));
310 data.read((char*)&nScatteringOrders, sizeof(nScatteringOrders));
311
312 sunSkyUBOData.atmLengthUnitInMeters = lengthUnitInMeters;
313 sunSkyUBOData.planetRadiusInUnits = bottomRadius/lengthUnitInMeters;
314
315 //Load atmosphere textures
316 glm::vec4* pixels;
317 pixels = new glm::vec4[transmittanceSize.x * transmittanceSize.y];
318 data.read((char*)pixels, sizeof(glm::vec4) * transmittanceSize.x * transmittanceSize.y);
319 textures[AtmosphereTextures::TRANSMITTANCE] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, transmittanceSize, GL_RGBA32F, GL_RGBA, GL_FLOAT, pixels, FilteringMode::BILINEAR, false);
320 delete [] pixels;
321 pixels = new glm::vec4[irradianceSize.x * irradianceSize.y];
322 data.read((char*)pixels, sizeof(glm::vec4) * irradianceSize.x * irradianceSize.y);
323 textures[AtmosphereTextures::IRRADIANCE] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, irradianceSize, GL_RGBA32F, GL_RGBA, GL_FLOAT, pixels, FilteringMode::BILINEAR, false);
324 delete [] pixels;
325 pixels = new glm::vec4[scatteringSize.x * scatteringSize.y * scatteringSize.z];
326 data.read((char*)pixels, sizeof(glm::vec4) * scatteringSize.x * scatteringSize.y * scatteringSize.z);
327 textures[AtmosphereTextures::SCATTERING] = OpenGLContent::GenerateTexture(GL_TEXTURE_3D, scatteringSize, GL_RGBA16F, GL_RGBA, GL_FLOAT, pixels, FilteringMode::BILINEAR, false);
328 delete [] pixels;
329#ifndef EMBEDDED_RESOURCES
330 dataFile.close();
331#endif
332 cInfo("Loaded precomputed atmosphere model (%u wavelengths, %u scattering orders)",
333 nPrecomputedWavelengths, nScatteringOrders);
334}

337{
338 glm::vec3 eyePos = view->GetEyePosition()/sunSkyUBOData.atmLengthUnitInMeters;

350 skySunShaders[0]->SetUniform("scattering_texture", TEX_ATM_SCATTERING);
351 skySunShaders[0]->SetUniform("single_mie_scattering_texture", TEX_ATM_SCATTERING);
352 skySunShaders[0]->SetUniform("eyePos", eyePos);
353 skySunShaders[0]->SetUniform("sunDir", GetSunDirection());
354 skySunShaders[0]->SetUniform("invProj", invProjectionMatrix);
355 skySunShaders[0]->SetUniform("invView", invViewMatrix);
356 skySunShaders[0]->SetUniform("whitePoint", sunSkyUBOData.whitePoint);

391void OpenGLAtmosphere::BakeShadowmaps(OpenGLPipeline* pipe, OpenGLView* view)
392{
393 if(sunShadowmapSize == 0)
394 return;
395
396 //Pre-set splits
397 for(unsigned int i = 0; i < sunShadowmapSplits; ++i)
398 {
399 sunShadowFrustum[i].fov = view->GetFOVY() + 0.2f; //avoid artifacts in the borders
400 GLint* viewport = view->GetViewport();
401 sunShadowFrustum[i].ratio = (GLfloat)viewport[2]/(GLfloat)viewport[3];
402 delete [] viewport;
403 }
404
405 //Compute the z-distances for each split as seen in camera space
406 UpdateSplitDist(view->GetNearClip(), view->GetFarClip());
407
408 //Render maps
409 glCullFace(GL_FRONT); //GL_FRONT -> no shadow acne but problems with filtering
410 glDisable(GL_DEPTH_CLAMP);
411
412 OpenGLState::BindFramebuffer(sunShadowFBO);
413 OpenGLState::Viewport(0, 0, sunShadowmapSize, sunShadowmapSize);
414
415 glm::vec3 camPos = view->GetEyePosition();
416 glm::vec3 camDir = view->GetLookingDirection();
417 glm::vec3 camUp = view->GetUpDirection();
418
419 // for all shadow splits
420 for(unsigned int i = 0; i < sunShadowmapSplits; ++i)
421 {
422 //Compute the camera frustum slice boundary points in world space
423 UpdateFrustumCorners(sunShadowFrustum[i], camPos, camDir, camUp);
424
425 //Adjust the view frustum of the light, so that it encloses the camera frustum slice fully.
426 //note that this function sets the projection matrix as it sees best fit
427 glm::mat4 cp = BuildCropProjMatrix(sunShadowFrustum[i]);
428 sunShadowCPM[i] = cp * sunModelView;
429
430 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->SetProjectionMatrix(cp);
431 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->SetViewMatrix(sunModelView);
432 //Draw current depth map
433 glFramebufferTextureLayer(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT, sunShadowmapArray, 0, i);
434 glClear(GL_DEPTH_BUFFER_BIT);
435 pipe->DrawObjects();
436 }
437
438 OpenGLState::BindFramebuffer(0);
439
440 glEnable(GL_DEPTH_CLAMP);
441 glCullFace(GL_BACK);
442}

444void OpenGLAtmosphere::UpdateSkyEmissivity()
445{
446 //Input data
447 GLfloat T = airTemperature + 273.15f;
448 GLfloat RH = 100.f * airHumidity;
449 GLfloat N = 0.f; // Cloud cover in tenths (clear sky)
450
451 //Calculate dewpoint temperature
452 GLfloat Td = T - ((100.f - RH)/5.f); // Simplified formula
453
454 //Calculate sky emmissivity
455 GLfloat eps = 0.787 + 0.767 * logf(Td/273.15f) + 0.0224*N + 0.0035*N*N + 0.00028*N*N*N;
456
457 //Update the UBO
458 sunSkyUBOData.skyEmissivity = eps;
459 sunSkyUBOData.airTemperature = airTemperature;
460}

462//Computes the near and far distances for every frustum slice in camera eye space
463void OpenGLAtmosphere::UpdateSplitDist(GLfloat nd, GLfloat fd)
464{
465 fd = fd > 250.f ? 250.f : fd; //Limit camera frustum to max 250.0 m (to avoid shadow blocks)
466 nd = 0.1f;
467 GLfloat lambda = 0.95f;
468 GLfloat ratio = fd/nd;
469 sunShadowFrustum[0].near = nd;
470
471 for(unsigned int i = 1; i < sunShadowmapSplits; ++i)
472 {
473 GLfloat si = i / (GLfloat)sunShadowmapSplits;
474
475 sunShadowFrustum[i].near = lambda*(nd*powf(ratio, si)) + (1.f-lambda)*(nd + (fd - nd)*si);
476 sunShadowFrustum[i-1].far = sunShadowFrustum[i].near * 1.05f;
477 }
478
479 sunShadowFrustum[sunShadowmapSplits-1].far = fd;
480 //for(int i=0; i<sunShadowmapSplits; i++) printf("Frustum%d Near: %f Far: %f\n", i, sunShadowFrustum[i].near, sunShadowFrustum[i].far);
481}

483//Computes the 8 corner points of the current view frustum
484void OpenGLAtmosphere::UpdateFrustumCorners(ViewFrustum &f, glm::vec3 center, glm::vec3 dir, glm::vec3 up)
485{
486 glm::vec3 right = glm::cross(dir, up);
487 right = glm::normalize(right);
488 //up = glm::normalize(glm::cross(right, dir));
489
490 glm::vec3 fc = center + dir * f.far;
491 glm::vec3 nc = center + dir * f.near;
492
493 // these heights and widths are half the heights and widths of
494 // the near and far plane rectangles
495 GLfloat nearHeight = tan(f.fov/2.0f) * f.near;
496 GLfloat nearWidth = nearHeight * f.ratio;
497 GLfloat farHeight = tan(f.fov/2.0f) * f.far;
498 GLfloat farWidth = farHeight * f.ratio;
499
500 f.corners[0] = nc - up * nearHeight - right * nearWidth;
501 f.corners[1] = nc + up * nearHeight - right * nearWidth;
502 f.corners[2] = nc + up * nearHeight + right * nearWidth;
503 f.corners[3] = nc - up * nearHeight + right * nearWidth;
504 f.corners[4] = fc - up * farHeight - right * farWidth;
505 f.corners[5] = fc + up * farHeight - right * farWidth;
506 f.corners[6] = fc + up * farHeight + right * farWidth;
507 f.corners[7] = fc - up * farHeight + right * farWidth;
508}

510// this function builds a projection matrix for rendering from the shadow's POV.
511// First, it computes the appropriate z-range and sets an orthogonal projection.
512// Then, it translates and scales it, so that it exactly captures the bounding box
513// of the current frustum slice
514glm::mat4 OpenGLAtmosphere::BuildCropProjMatrix(ViewFrustum &f)
515{
516 GLfloat maxX = -1000.0f;
517 GLfloat maxY = -1000.0f;
518 GLfloat maxZ;
519 GLfloat minX = 1000.0f;
520 GLfloat minY = 1000.0f;
521 GLfloat minZ;
522 glm::vec4 transf;
523
524 //Find the z-range of the current frustum as seen from the light in order to increase precision
525 glm::mat4 shad_mv = sunModelView;
526
527 //Note: only the z-component is needed and thus the multiplication can be simplified
528 //transf.z = shad_modelview[2] * f.point[0].x + shad_modelview[6] * f.point[0].y + shad_modelview[10] * f.point[0].z + shad_modelview[14]
529 transf = shad_mv * glm::vec4(f.corners[0], 1.0f);
530 minZ = transf.z;
531 maxZ = transf.z;
532
533 for(int i = 1; i <8 ; i++) {
534 transf = shad_mv * glm::vec4(f.corners[i], 1.0f);
535 if(transf.z > maxZ) maxZ = transf.z;
536 if(transf.z < minZ) minZ = transf.z;
537 }
538
539 //Make sure all relevant shadow casters are included - use object bounding boxes!
540 Vector3 aabbMin;
541 Vector3 aabbMax;
542 SimulationApp::getApp()->getSimulationManager()->getWorldAABB(aabbMin, aabbMax);
543
544 transf = shad_mv * glm::vec4(aabbMin.x(), aabbMin.y(), aabbMin.z(), 1.f);
545 if(transf.z > maxZ) maxZ = transf.z;
546 if(transf.z < minZ) minZ = transf.z;
547
548 transf = shad_mv * glm::vec4(aabbMax.x(), aabbMax.y(), aabbMax.z(), 1.f);
549 if(transf.z > maxZ) maxZ = transf.z;
550 if(transf.z < minZ) minZ = transf.z;
551
552 //Set the projection matrix with the new z-bounds
553 //Note: there is inversion because the light looks at the neg z axis
554 glm::mat4 shad_proj = glm::ortho(-1.f, 1.f, -1.f, 1.f, -maxZ, -minZ);
555 glm::mat4 shad_mvp = shad_proj * shad_mv;
556
557 //Find the extends of the frustum slice as projected in light's homogeneous coordinates
558 for(int i = 0; i < 8; i++) {
559 transf = shad_mvp * glm::vec4(f.corners[i], 1.0f);
560
561 transf.x /= transf.w;
562 transf.y /= transf.w;
563
564 if(transf.x > maxX) maxX = transf.x;
565 if(transf.x < minX) minX = transf.x;
566 if(transf.y > maxY) maxY = transf.y;
567 if(transf.y < minY) minY = transf.y;
568 }
569
570 //Build crop matrix
571 glm::mat4 shad_crop = glm::mat4(1.f);
572 GLfloat scaleX = 2.0f/(maxX - minX);
573 GLfloat scaleY = 2.0f/(maxY - minY);
574 GLfloat offsetX = -0.5f*(maxX + minX)*scaleX;
575 GLfloat offsetY = -0.5f*(maxY + minY)*scaleY;
576 shad_crop[0][0] = scaleX;
577 shad_crop[1][1] = scaleY;
578 shad_crop[3][0] = offsetX;
579 shad_crop[3][1] = offsetY;
580
581 //Build crop projection matrix
582 glm::mat4 shad_crop_proj = shad_crop * shad_proj;
583
584 return shad_crop_proj;
585}

587void OpenGLAtmosphere::SetupMaterialShaders()
588{
589 //Calculate shadow splits
590 glm::mat4 bias(0.5f, 0.f, 0.f, 0.f,
591 0.f, 0.5f, 0.f, 0.f,
592 0.f, 0.f, 0.5f, 0.f,
593 0.5f, 0.5f, 0.5f, 1.f);
594 GLfloat frustumFar[4];
595 GLfloat frustumNear[4];
596
597 //For every inactive split
598 for(unsigned int i = sunShadowmapSplits; i<4; ++i)
599 {
600 frustumNear[i] = 0;
601 frustumFar[i] = 0;
602 sunSkyUBOData.sunClipSpace[i] = glm::mat4();
603 }
604
605 //For every active split
606 for(unsigned int i = 0; i < sunShadowmapSplits; ++i)
607 {
608 frustumNear[i] = sunShadowFrustum[i].near;
609 frustumFar[i] = sunShadowFrustum[i].far;
610 sunSkyUBOData.sunClipSpace[i] = (bias * sunShadowCPM[i]); // compute a matrix that transforms from world space to light clip space
611 }
612
613 sunSkyUBOData.sunFrustumNear = glm::vec4(frustumNear[0], frustumNear[1], frustumNear[2], frustumNear[3]);
614 sunSkyUBOData.sunFrustumFar = glm::vec4(frustumFar[0], frustumFar[1], frustumFar[2], frustumFar[3]);
615 sunSkyUBOData.sunDirection = GetSunDirection();
616
617 glBindBuffer(GL_UNIFORM_BUFFER, sunSkyUBO);
618 glBufferSubData(GL_UNIFORM_BUFFER, 0, sizeof(SunSkyUBO), &sunSkyUBOData);
619 glBindBuffer(GL_UNIFORM_BUFFER, 0);
620}

623{
624 //Texture setup
625 OpenGLState::BindTexture(TEX_SUN_SHADOW, GL_TEXTURE_2D_ARRAY, sunShadowmapArray);

637 }
638 OpenGLState::UseProgram(0);
639 OpenGLState::UnbindTexture(TEX_SUN_SHADOW);
