52OpenGLConsole::~OpenGLConsole()
53{

55 if(logoTexture > 0) glDeleteTextures(1, &logoTexture);

60void OpenGLConsole::Init(int w, int h)
61{
62 windowW = w;
63 windowH = h;
64 OpenGLPrinter::SetWindowSize(windowW, windowH);
65
66 if(logoTexture > 0) //Check if not already initialized
67 return;
68 //Load logo texture
69 logoTexture = OpenGLContent::LoadInternalTexture("logo_64.png", false, true);
70
71 glGenVertexArrays(1, &consoleVAO);
72 OpenGLState::BindVertexArray(consoleVAO);
73 glEnableVertexAttribArray(0);
74 OpenGLState::BindVertexArray(0);
75
76 texQuadShader = new GLSLShader("texQuad.frag","texQuad.vert");
77 texQuadShader->AddUniform("rect", ParameterType::VEC4);
78 texQuadShader->AddUniform("tex", ParameterType::INT);
79 texQuadShader->AddUniform("color", ParameterType::VEC4);
80
81 printer = new OpenGLPrinter(GetShaderPath() + std::string(STANDARD_FONT_NAME), STANDARD_FONT_SIZE);
82}

95void OpenGLConsole::Render(bool overlay)
96{
97 if(logoTexture == 0)
98 return;
99
100 int64_t now = GetTimeInMicroseconds();
101 GLfloat dt = (lastTime-now)/1000000.f;
102 lastTime = now;
103
104 if(lines.size() == 0)
105 return;
106
107 //Calculate visible lines range
108 long int maxVisibleLines = (long int)floorf((GLfloat)windowH/(GLfloat)(STANDARD_FONT_SIZE + 5)) + 1;
109 long int linesCount = lines.size();
110 long int visibleLines = maxVisibleLines;
111 long int scrolledLines = 0;
112
113 GLfloat logoSize = 64.f;
114 GLfloat logoMargin = 10.f;
115 glm::vec4 info(1.f, 1.f, 1.f, 1.f);
116 glm::vec4 warning(1.f, 1.f, 0.f, 1.f);
117 glm::vec4 error(1.f, 0.f, 0.f, 1.f);
118
119 if(linesCount < maxVisibleLines) //there is enough space to display all lines
120 {
121 scrollOffset = 0.f;
122 scrollVelocity = 0.f;
123 visibleLines = linesCount;
124 }
125 else //there is more lines than can appear on screen at once
126 {
127 scrolledLines = (long int)floorf(-scrollOffset /(GLfloat)(STANDARD_FONT_SIZE + 5));
128 scrolledLines = scrolledLines < 0 ? 0 : scrolledLines;
129 if(linesCount < scrolledLines + visibleLines)
130 visibleLines = linesCount - scrolledLines;
131
132 if(scrollVelocity != 0.f) //velocity damping (momentum effect)
133 {
134 scrollOffset += scrollVelocity * dt;
135 GLfloat dampingAcc = scrollVelocity * 1.f;
136 scrollVelocity += dampingAcc * dt;
137 }
138
139 //springy effect
140 if(scrollOffset > 0.f) //the list is scrolled up too much
141 {
142 GLfloat newVelocity = scrollVelocity - scrollOffset * 10.f * dt;
143 if(newVelocity > 0.f)
144 scrollVelocity = scrollOffset * 2.f;
145 else
146 scrollVelocity = newVelocity;
147 }
148 else if(visibleLines < (maxVisibleLines - 1)) //the list is scrolled down too much
149 {
150 GLfloat newVelocity = scrollVelocity + (1.f - (GLfloat)visibleLines/(GLfloat)(maxVisibleLines - 2)) * (GLfloat)windowH * 10.f * dt;
151 if(newVelocity < 0.f)
152 scrollVelocity = -((1.f - (GLfloat)visibleLines/(GLfloat)(maxVisibleLines - 1)) * (GLfloat)windowH) * 2.f;
153 else
154 scrollVelocity = newVelocity;
155 }
156 }
157
158 //Setup viewport and ortho
159 if(overlay)
160 {
161 glScissor(0, 0, windowW, windowH);
162 OpenGLState::Viewport(0, 0, windowW, windowH);
163 OpenGLState::DisableDepthTest();
164 OpenGLState::DisableCullFace();
165 OpenGLState::EnableBlend();
166 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
167
168 texQuadShader->Use();
169 texQuadShader->SetUniform("tex", 0);
170 texQuadShader->SetUniform("color", glm::vec4(0.3f,0.3f,0.3f,1.f));
171 texQuadShader->SetUniform("rect", glm::vec4(0, 0, 1.f, 1.f));
172
173 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D, ((GraphicalSimulationApp*)SimulationApp::getApp())->getGUI()->getTranslucentTexture());
174 OpenGLState::BindVertexArray(consoleVAO);
175 glDrawArrays(GL_TRIANGLES, 0, 6);
176
177 texQuadShader->SetUniform("color", glm::vec4(1.f,1.f,1.f,1.f));
178 texQuadShader->SetUniform("rect", glm::vec4((windowW - logoSize - logoMargin)/(GLfloat)windowW, 1.f - (logoMargin+logoSize)/(GLfloat)windowH, logoSize/(GLfloat)windowW, logoSize/(GLfloat)windowH));
179
180 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D, logoTexture);
181 glDrawArrays(GL_TRIANGLES, 0, 6);
182 OpenGLState::UseProgram(0);
183 OpenGLState::UnbindTexture(TEX_BASE);
184
185 //Text rendering
186 for(long int i = scrolledLines; i < scrolledLines + visibleLines; i++)
187 {
188 ConsoleMessage* msg = &lines[linesCount-1-i];
189 glm::vec4 color;
190 switch(msg->type)
191 {
192 case MessageType::INFO:
193 color = info;
194 break;
195 case MessageType::WARNING:
196 color = warning;
197 break;
198 case MessageType::ERROR:
199 case MessageType::CRITICAL:
200 color = error;
201 break;
202 }
203 printer->Print(msg->text.c_str(), color, 10.f, scrollOffset + 10.f + i * (STANDARD_FONT_SIZE + 5), STANDARD_FONT_SIZE);
204 }
205
206 OpenGLState::BindVertexArray(0);
207 OpenGLState::DisableBlend();
208 OpenGLState::EnableDepthTest();
209 OpenGLState::EnableCullFace();
210 }
211 else //During loading of resources (displaying in second thread -> no VAO sharing)
212 {
213 glUseProgram(texQuadShader->getProgramHandle());
214 texQuadShader->SetUniform("tex", 0);
215 texQuadShader->SetUniform("color", glm::vec4(1.f,1.f,1.f,1.f));
216 texQuadShader->SetUniform("rect", glm::vec4((windowW - logoSize - logoMargin)/(GLfloat)windowW, 1.f - (logoMargin+logoSize)/(GLfloat)windowH, logoSize/(GLfloat)windowW, logoSize/(GLfloat)windowH));
217
218 glActiveTexture(GL_TEXTURE0);
219 glBindTexture(GL_TEXTURE_2D, logoTexture);
220 glDrawArrays(GL_TRIANGLES, 0, 6);
221 glBindTexture(GL_TEXTURE_2D, 0);
222 glUseProgram(0);
223
224 //Text rendering
225 for(long int i = scrolledLines; i < scrolledLines + visibleLines; i++)
226 {
227 ConsoleMessage* msg = &lines[linesCount-1-i];
228 glm::vec4 color;
229 switch(msg->type)
230 {
231 case MessageType::INFO:
232 color = info;
233 break;
234 case MessageType::WARNING:
235 color = warning;
236 break;
237 case MessageType::ERROR:
238 case MessageType::CRITICAL:
239 color = error;
240 break;
241 }
242 printer->Print(msg->text.c_str(), color, 10.f, scrollOffset + 10.f + i * (STANDARD_FONT_SIZE + 5), STANDARD_FONT_SIZE, true);
243 }
244 }
245

247}
