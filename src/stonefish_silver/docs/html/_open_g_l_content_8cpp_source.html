488 if(csBuf[0] != 0) glDeleteBuffers(2, csBuf);
489 if(lightsUBO != 0) glDeleteBuffers(1, &lightsUBO);

924void OpenGLContent::UseLook(const Look& look, bool texturable, const glm::mat4& M)
925{
926 bool waves = false;
927 Ocean* ocean = SimulationApp::getApp()->getSimulationManager()->getOcean();
928 if(ocean != NULL && ocean->hasWaves()) waves = true;
929
930 texturable = texturable && (look.albedoTexture > 0 || look.normalMap > 0 || look.temperatureMap > 0);
931 int shaderMode = 0; // Above water/on surface
932 if(mode == DrawingMode::UNDERWATER)
933 shaderMode = (waves ? 2 : 1);
934 else if(mode == DrawingMode::TEMPERATURE)
935 shaderMode = 3;
936
937 bool updateMaterial = (look.name != currentLookName)
938 || (currentTexturable != texturable)
939 || (currentShaderMode != shaderMode);
940 currentLookName = look.name;
941 currentTexturable = texturable;
942 currentShaderMode = shaderMode;
943
944 size_t shaderId = (currentTexturable ? 4 : 0) + (size_t)currentShaderMode;
945 GLSLShader* shader = materialShaders[look.type == LookType::SIMPLE ? 0 : 1].shaders[shaderId];
946 shader->Use();
947 shader->SetUniform("MVP", viewProjection*M);
948 shader->SetUniform("M", M);
949 shader->SetUniform("N", glm::mat3(glm::transpose(glm::inverse(M))));
950 shader->SetUniform("MV", glm::mat3(glm::transpose(glm::inverse(view*M))));
951 shader->SetUniform("FC", FC);
952 shader->SetUniform("eyePos", eyePos);
953 shader->SetUniform("viewDir", viewDir);
954
955 if(updateMaterial)
956 {
957 switch(look.type)
958 {
959 default:
960 case LookType::SIMPLE: //Blinn-Phong
961 {
962 shader->SetUniform("specularStrength", look.params[0]);
963 shader->SetUniform("shininess", look.params[1]);
964 shader->SetUniform("reflectivity", look.reflectivity);
965 shader->SetUniform("color", glm::vec4(look.color.rgb, 1.f));
966 }
967 break;
968
969 case LookType::PHYSICAL: //Cook-Torrance
970 {
971 shader->SetUniform("roughness", look.params[0]);
972 shader->SetUniform("metallic", look.params[1]);
973 shader->SetUniform("reflectivity", look.reflectivity);
974 shader->SetUniform("color", glm::vec4(look.color.rgb, 1.f));
975 }
976 break;
977 }
978
979 if(currentTexturable)
980 {
981 if(look.albedoTexture > 0)
982 {
983 shader->SetUniform("enableAlbedoTex", true);
984 OpenGLState::BindTexture(TEX_MAT_ALBEDO, GL_TEXTURE_2D, look.albedoTexture);
985 }
986 else
987 {
988 shader->SetUniform("enableAlbedoTex", false);
989 OpenGLState::UnbindTexture(TEX_MAT_ALBEDO);
990 }
991
992 if(look.normalMap > 0)
993 {
994 shader->SetUniform("enableNormalTex", true);
995 OpenGLState::BindTexture(TEX_MAT_NORMAL, GL_TEXTURE_2D, look.normalMap);
996 }
997 else
998 {
999 shader->SetUniform("enableNormalTex", false);
1000 OpenGLState::UnbindTexture(TEX_MAT_NORMAL);
1001 }
1002
1003 if(shaderMode == 3)
1004 {
1005 if(look.temperatureMap > 0)
1006 {
1007 OpenGLState::BindTexture(TEX_MAT_TEMPERATURE, GL_TEXTURE_2D, look.temperatureMap);
1008 shader->SetUniform("enableTemperatureTex", true);
1009 shader->SetUniform("texTemperature", TEX_MAT_TEMPERATURE);
1010 }
1011 else
1012 {
1013 shader->SetUniform("enableTemperatureTex", false);
1014 OpenGLState::UnbindTexture(TEX_MAT_TEMPERATURE);
1015 }
1016 shader->SetUniform("temperatureRange", look.temperatureRange);
1017 }
1018 }
1019 else
1020 {
1021 if(shaderMode == 3)
1022 shader->SetUniform("temperature", look.temperatureRange.x);
1023 }
1024 }
1025
1026 if(mode == DrawingMode::UNDERWATER)
1027 {
1028 shader->SetUniform("cWater", ocean->getOpenGLOcean()->getLightAttenuation());
1029 shader->SetUniform("bWater", ocean->getOpenGLOcean()->getLightScattering());
1030 if(waves)
1031 {
1032 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D_ARRAY, ocean->getOpenGLOcean()->getWaveTexture());
1033 shader->SetUniform("texWaveFFT", TEX_POSTPROCESS1);
1034 shader->SetUniform("gridSizes", ocean->getOpenGLOcean()->getWaveGridSizes());
1035 }
1036 }
1037}

1039unsigned int OpenGLContent::BuildObject(Mesh* mesh)
1040{
1041 Object obj;
1042
1043 glGenVertexArrays(1, &obj.vao);
1044 glGenBuffers(1, &obj.vboVertex);
1045 glGenBuffers(1, &obj.vboIndex);
1046 obj.faceCount = (GLsizei)mesh->faces.size();
1047 obj.texturable = false;
1048
1049 OpenGLState::BindVertexArray(obj.vao);
1050 glEnableVertexAttribArray(0); //Position
1051 glEnableVertexAttribArray(1); //Normal
1052 if(mesh->isTexturable())
1053 {
1054 glEnableVertexAttribArray(2); //UV
1055 glEnableVertexAttribArray(3); //Tangent
1056 obj.texturable = true;
1057 }
1058
1059 glBindBuffer(GL_ARRAY_BUFFER, obj.vboVertex);
1060 glBufferData(GL_ARRAY_BUFFER, mesh->getVertexSize() * mesh->getNumOfVertices(), mesh->getVertexDataPointer(), GL_STATIC_DRAW);
1061 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, mesh->getVertexSize(), 0);
1062 glVertexAttribPointer(1, 3, GL_FLOAT, GL_TRUE, mesh->getVertexSize(), (void*)sizeof(glm::vec3));
1063 if(mesh->isTexturable())
1064 {
1065 glVertexAttribPointer(2, 2, GL_FLOAT, GL_FALSE, mesh->getVertexSize(), (void*)(sizeof(glm::vec3)*2));
1066 glVertexAttribPointer(3, 3, GL_FLOAT, GL_TRUE, mesh->getVertexSize(), (void*)(sizeof(glm::vec3)*2 + sizeof(glm::vec2)));
1067 }
1068 glBindBuffer(GL_ARRAY_BUFFER, 0);
1069
1070 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, obj.vboIndex);
1071 glBufferData(GL_ELEMENT_ARRAY_BUFFER, sizeof(Face) * mesh->faces.size(), &mesh->faces[0].vertexID[0], GL_STATIC_DRAW);
1072 OpenGLState::BindVertexArray(0);
1073
1074 objects.push_back(obj);
1075 return (unsigned int)objects.size()-1;
1076}

1172GLuint OpenGLContent::LoadTexture(const std::string& filename, bool srgb, bool alpha, GLfloat anisotropy, bool internal)
1173{
1174 int width, height, channels;
1175 int reqChannels = alpha ? 4 : 3;
1176 GLuint texture;
1177
1178 // Allocate image; fail out on error
1179 stbi_set_flip_vertically_on_load(true);
1180#ifdef EMBEDDED_RESOURCES
1181 unsigned char* dataBuffer;
1182 if(internal)
1183 {
1184 ResourceHandle rh(filename);
1185 dataBuffer = stbi_load_from_memory(rh.data(), rh.size(), &width, &height, &channels, reqChannels);
1186 }
1187 else
1188 dataBuffer = stbi_load(filename.c_str(), &width, &height, &channels, reqChannels);
1189#else
1190 unsigned char* dataBuffer = stbi_load(filename.c_str(), &width, &height, &channels, reqChannels);
1191#endif
1192 if(dataBuffer == NULL)
1193 {
1194 cError("Failed to load texture from: %s", filename.c_str());
1195 return 0;
1196 }
1197
1198 if(channels != reqChannels)
1199 {
1200 cWarning("Texture has %d channels while expected %d channels!", channels, reqChannels);
1201 }
1202
1203 cInfo("Loaded texture from: %s", filename.c_str());
1204
1205 glGenTextures(1, &texture);
1206 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D, texture);
1207 if(srgb)
1208 glTexImage2D(GL_TEXTURE_2D, 0, alpha ? GL_SRGB8_ALPHA8 : GL_SRGB8, width, height, 0, alpha ? GL_RGBA : GL_RGB, GL_UNSIGNED_BYTE, dataBuffer);
1209 else
1210 glTexImage2D(GL_TEXTURE_2D, 0, alpha ? GL_RGBA8 : GL_RGB8, width, height, 0, alpha? GL_RGBA : GL_RGB, GL_UNSIGNED_BYTE, dataBuffer);
1211 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
1212 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR_MIPMAP_LINEAR);
1213 if(anisotropy > 0.f)
1214 glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MAX_ANISOTROPY_EXT, anisotropy);
1215 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
1216 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
1217 glGenerateMipmap(GL_TEXTURE_2D);
1218 OpenGLState::UnbindTexture(TEX_BASE);
1219
1220 stbi_image_free(dataBuffer);
1221
1222 return texture;
1223}

1230GLuint OpenGLContent::GenerateTexture(GLenum target, glm::uvec3 dimensions, GLenum internalFormat, GLenum format, GLenum type, const void* data,
1231 FilteringMode fm, bool repeat, bool anisotropy)
1232{
1233 GLuint texture = 0;
1234 glGenTextures(1, &texture);
1235 OpenGLState::BindTexture(TEX_BASE, target, texture);
1236
1237 switch(target)
1238 {
1239 case GL_TEXTURE_1D:
1240 if(dimensions.x == 0)
1241 {
1242 OpenGLState::UnbindTexture(TEX_BASE);
1243 glDeleteTextures(1, &texture);
1244 cError("Texture dimensions cannot be equal to 0!");
1245 return 0;
1246 }
1247 glTexImage1D(target, 0, internalFormat, dimensions.x, 0, format, type, data);
1248 break;
1249
1250 case GL_TEXTURE_2D:
1251 if(dimensions.x == 0 || dimensions.y == 0)
1252 {
1253 OpenGLState::UnbindTexture(TEX_BASE);
1254 glDeleteTextures(1, &texture);
1255 cError("Texture dimensions cannot be equal to 0!");
1256 return 0;
1257 }
1258 glTexImage2D(target, 0, internalFormat, dimensions.x, dimensions.y, 0, format, type, data);
1259 break;
1260
1261 case GL_TEXTURE_2D_ARRAY:
1262 case GL_TEXTURE_3D:
1263 if(dimensions.x == 0 || dimensions.y == 0 || dimensions.z == 0)
1264 {
1265 OpenGLState::UnbindTexture(TEX_BASE);
1266 glDeleteTextures(1, &texture);
1267 cError("Texture dimensions cannot be equal to 0!");
1268 return 0;
1269 }
1270 glTexImage3D(target, 0, internalFormat, dimensions.x, dimensions.y, dimensions.z, 0, format, type, data);
1271 break;
1272
1273 default:
1274 OpenGLState::UnbindTexture(TEX_BASE);
1275 glDeleteTextures(1, &texture);
1276 cError("Unsupported texture format requested!");
1277 return 0;
1278 }
1279
1280 GLenum repeatMode = repeat ? GL_REPEAT : GL_CLAMP_TO_EDGE;
1281 glTexParameteri(target, GL_TEXTURE_WRAP_S, repeatMode);
1282 glTexParameteri(target, GL_TEXTURE_WRAP_T, repeatMode);
1283 if(target == GL_TEXTURE_3D)
1284 glTexParameteri(target, GL_TEXTURE_WRAP_R, repeatMode);
1285
1286 if(anisotropy)
1287 glTexParameterf(target, GL_TEXTURE_MAX_ANISOTROPY, OpenGLState::GetMaxAnisotropy());
1288
1289 switch(fm)
1290 {
1291 case FilteringMode::NEAREST:
1292 glTexParameteri(target, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
1293 glTexParameteri(target, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
1294 break;
1295
1296 case FilteringMode::BILINEAR:
1297 glTexParameteri(target, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
1298 glTexParameteri(target, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
1299 break;
1300
1301 case FilteringMode::BILINEAR_MIPMAP:
1302 glTexParameteri(target, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_NEAREST);
1303 glTexParameteri(target, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
1304 glGenerateMipmap(target);
1305 break;
1306
1307 case FilteringMode::TRILINEAR:
1308 glTexParameteri(target, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
1309 glTexParameteri(target, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
1310 glGenerateMipmap(target);
1311 break;
1312 }
1313
1314 OpenGLState::UnbindTexture(TEX_BASE);
1315 return texture;
1316}

1318GLuint OpenGLContent::GenerateFramebuffer(const std::vector<FBOTexture>& textures)
1319{
1320 GLuint fbo;
1321 glGenFramebuffers(1, &fbo);
1322 OpenGLState::BindFramebuffer(fbo);
1323
1324 for(size_t i = 0; i < textures.size(); ++i)
1325 {
1326 if(textures[i].attach == GL_DEPTH_ATTACHMENT
1327 || textures[i].attach == GL_STENCIL_ATTACHMENT
1328 || textures[i].attach == GL_DEPTH_STENCIL_ATTACHMENT)
1329 {
1330 glFramebufferTexture(GL_FRAMEBUFFER, textures[i].attach, textures[i].tex, textures[i].lvl);
1331 continue;
1332 }
1333
1334 switch(textures[i].target)
1335 {
1336 case GL_TEXTURE_1D:
1337 glFramebufferTexture1D(GL_FRAMEBUFFER, textures[i].attach, GL_TEXTURE_1D, textures[i].tex, textures[i].lvl);
1338 break;
1339
1340 case GL_TEXTURE_2D:
1341 case GL_TEXTURE_RECTANGLE:
1342 case GL_TEXTURE_CUBE_MAP_POSITIVE_X:
1343 case GL_TEXTURE_CUBE_MAP_POSITIVE_Y:
1344 case GL_TEXTURE_CUBE_MAP_POSITIVE_Z:
1345 case GL_TEXTURE_CUBE_MAP_NEGATIVE_X:
1346 case GL_TEXTURE_CUBE_MAP_NEGATIVE_Y:
1347 case GL_TEXTURE_CUBE_MAP_NEGATIVE_Z:
1348 case GL_TEXTURE_2D_MULTISAMPLE:
1349 glFramebufferTexture2D(GL_FRAMEBUFFER, textures[i].attach, textures[i].target, textures[i].tex, textures[i].lvl);
1350 break;
1351
1352 case GL_TEXTURE_3D:
1353 glFramebufferTexture3D(GL_FRAMEBUFFER, textures[i].attach, GL_TEXTURE_3D, textures[i].tex, textures[i].lvl, textures[i].z);

1357 break;
1358 }
1359 }
1360
1361 GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
1362 if(status != GL_FRAMEBUFFER_COMPLETE)
1363 {
1364 cError("Framebuffer initialization failed!");
1365 OpenGLState::BindFramebuffer(0);
1366 glDeleteFramebuffers(1, &fbo);
1367 return 0;
1368 }
1369 else
1370 {
1371 OpenGLState::BindFramebuffer(0);
1372 return fbo;
1373 }
