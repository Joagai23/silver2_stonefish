76 struct MaterialShader
77 {
78 std::string shadingAlgorithm;
79 std::array<GLSLShader*, 8> shaders;
80
81 MaterialShader()
82 {
83 shadingAlgorithm = "";
84 }
85
86 MaterialShader(const MaterialShader &obj)
87 {
88 shadingAlgorithm = obj.shadingAlgorithm;
89 for(size_t i=0; i<shaders.size(); ++i)
90 shaders[i] = obj.shaders[i];
91 }
92 };
