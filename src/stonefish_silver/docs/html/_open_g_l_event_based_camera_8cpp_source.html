41OpenGLEventBasedCamera::OpenGLEventBasedCamera(glm::vec3 eyePosition, glm::vec3 direction, glm::vec3 cameraUp,
42 GLint x, GLint y, GLint width, GLint height, GLfloat horizontalFovDeg,
43 glm::vec2 range, glm::vec2 C, uint32_t Tr, bool continuousUpdate)
44 : OpenGLCamera(x, y, width, height, range), randDist(0.f, 1.f)
45{
46 _needsUpdate = false;
47 newData = false;
48 initialized = false;
49 lastSimTime = -1.f;
50 continuous = continuousUpdate;
51 camera = nullptr;
52 C_ = C;
53 Tr_ = Tr;
54 sigmaC_ = glm::vec2(0.f);
55
56 // Allocate necessary textures
57 // --> Logarithm of luminance
58 renderLogLumTex = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
59 GL_R32F, GL_RED, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
60 // --> Events (R -> pos x | pos y in px, G -> timestamp in ns, with polarization)
61 maxNumEvents = viewportWidth * viewportHeight * 10;
62 glm::uvec3 maxTextureSize = OpenGLState::GetMaxTextureSize();
63 glm::uvec3 eventsTextureSize;
64 if(maxNumEvents > maxTextureSize.x)
65 {
66 eventsTextureSize.x = maxTextureSize.x;
67 eventsTextureSize.y = maxNumEvents/maxTextureSize.x + 1;
68 eventsTextureSize.z = 0;
69 maxNumEvents = eventsTextureSize.x * eventsTextureSize.y;
70 }
71 else
72 {
73 eventsTextureSize.x = maxNumEvents;
74 eventsTextureSize.y = 1;
75 eventsTextureSize.z = 0;
76 }
77 renderEventTex[0] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, eventsTextureSize,
78 GL_RG32I, GL_RG_INTEGER, GL_INT, NULL, FilteringMode::NEAREST, false);
79 // --> Last event timestamp (in ns, with polarization)
80 renderEventTex[1] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
81 GL_R32I, GL_RED_INTEGER, GL_INT, NULL, FilteringMode::NEAREST, false);
82 // --> Crossings
83 renderEventTex[2] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
84 GL_R32F, GL_RED, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
85 // Allocate buffer to store the event counter
86 GLuint zero = 0;
87 glGenBuffers(1, &renderEventCounter);
88 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, renderEventCounter);
89 glBufferData(GL_ATOMIC_COUNTER_BUFFER, sizeof(GLuint), &zero, GL_STREAM_READ);
90 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, 0);
91
92 // Event display
93 displayTex = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
94 GL_RGB8, GL_RGB, GL_UNSIGNED_BYTE, NULL, FilteringMode::BILINEAR, false);
95 std::vector<FBOTexture> fboTextures;
96 fboTextures.push_back(FBOTexture(GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, displayTex));
97 displayFBO = OpenGLContent::GenerateFramebuffer(fboTextures);
98
99 // Setup view
100 SetupCamera(eyePosition, direction, cameraUp);
101 // Setup projection
102 fovx = horizontalFovDeg/180.f * M_PI;
103 GLfloat fovy = 2.f * atanf( (GLfloat)viewportHeight/(GLfloat)viewportWidth * tanf(fovx/2.f) );
104 projection = glm::perspectiveFov(fovy, (GLfloat)viewportWidth, (GLfloat)viewportHeight, near, far);
105
106 UpdateTransform();
107}

179void OpenGLEventBasedCamera::UpdateTransform()
180{
181 eye = tempEye;
182 dir = tempDir;
183 up = tempUp;
184 SetupCamera();
185
186 viewUBOData.VP = GetProjectionMatrix() * GetViewMatrix();
187 viewUBOData.eye = GetEyePosition();
188 ExtractFrustumFromVP(viewUBOData.frustum, viewUBOData.VP);
189
190 //Inform camera to run callback
191 if(newData)
192 {
193 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, renderEventCounter);
194 GLuint* pEventCounter = (GLuint*)glMapBuffer(GL_ATOMIC_COUNTER_BUFFER, GL_READ_ONLY);
195 GLuint lastEventCount = *pEventCounter;
196 glUnmapBuffer(GL_ATOMIC_COUNTER_BUFFER);
197 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, 0);
198
199 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
200 GLint* src = (GLint*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
201 if(src)
202 {
203 camera->NewDataReady(src, lastEventCount);
204 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
205 }
206 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
207 newData = false;
208 }
209}

221void OpenGLEventBasedCamera::ComputeOutput(double simTime)
222{
223 if(initialized)
224 {
225 // Reset event counter
226 GLuint zero = 0;
227 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, renderEventCounter);
228 glBufferSubData(GL_ATOMIC_COUNTER_BUFFER, 0, sizeof(GLuint), &zero);
229 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, 0);
230
231 // Compute events
232 glBindBufferBase(GL_ATOMIC_COUNTER_BUFFER, 0, renderEventCounter);
233 glBindImageTexture(TEX_POSTPROCESS1, renderColorTex[lastActiveRenderColorBuffer], 0, GL_FALSE, 0, GL_READ_ONLY, GL_RGBA32F);
234 glBindImageTexture(TEX_POSTPROCESS2, renderLogLumTex, 0, GL_FALSE, 0, GL_READ_WRITE, GL_R32F);
235 glBindImageTexture(TEX_POSTPROCESS3, renderEventTex[0], 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_RG32I);
236 glBindImageTexture(TEX_POSTPROCESS4, renderEventTex[1], 0, GL_FALSE, 0, GL_READ_WRITE, GL_R32I);
237 glBindImageTexture(TEX_POSTPROCESS5, renderEventTex[2], 0, GL_FALSE, 0, GL_READ_WRITE, GL_R32F);
238 eventOutputShaders[1]->Use();
239 eventOutputShaders[1]->SetUniform("currColor", TEX_POSTPROCESS1);
240 eventOutputShaders[1]->SetUniform("logLum", TEX_POSTPROCESS2);
241 eventOutputShaders[1]->SetUniform("events", TEX_POSTPROCESS3);
242 eventOutputShaders[1]->SetUniform("eventTimes", TEX_POSTPROCESS4);
243 eventOutputShaders[1]->SetUniform("crossings", TEX_POSTPROCESS5);
244 Scalar dt = simTime - lastSimTime;
245 eventOutputShaders[1]->SetUniform("dT", (int32_t)round(dt*1e9));
246 eventOutputShaders[1]->SetUniform("Tr", Tr_);
247 eventOutputShaders[1]->SetUniform("C", C_);
248 eventOutputShaders[1]->SetUniform("sigmaC", sigmaC_);
249 eventOutputShaders[1]->SetUniform("seed", glm::vec3(randDist(randGen), randDist(randGen), randDist(randGen)));
250 glMemoryBarrier(GL_FRAMEBUFFER_BARRIER_BIT);
251 glDispatchCompute((GLuint)ceilf(viewportWidth/16.f), (GLuint)ceilf(viewportHeight/16.f), 1);
252 glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT | GL_ATOMIC_COUNTER_BARRIER_BIT);
253 }
254 else
255 {
256 // Compute logarithm of luminance of the first frame (initialization)
257 glBindImageTexture(TEX_POSTPROCESS1, renderColorTex[lastActiveRenderColorBuffer], 0, GL_FALSE, 0, GL_READ_ONLY, GL_RGBA32F);
258 glBindImageTexture(TEX_POSTPROCESS2, renderLogLumTex, 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_R32F);
259 glBindImageTexture(TEX_POSTPROCESS4, renderEventTex[1], 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_R32I);
260 glBindImageTexture(TEX_POSTPROCESS5, renderEventTex[2], 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_R32F);
261
262 eventOutputShaders[0]->Use();
263 eventOutputShaders[0]->SetUniform("currColor", TEX_POSTPROCESS1);
264 eventOutputShaders[0]->SetUniform("logLum", TEX_POSTPROCESS2);
265 eventOutputShaders[0]->SetUniform("eventTimes", TEX_POSTPROCESS4);
266 eventOutputShaders[0]->SetUniform("crossings", TEX_POSTPROCESS5);
267 glMemoryBarrier(GL_FRAMEBUFFER_BARRIER_BIT);
268 glDispatchCompute((GLuint)ceilf(viewportWidth/16.f), (GLuint)ceilf(viewportHeight/16.f), 1);
269 glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
270 initialized = true;
271 }
272 lastSimTime = simTime;
273}

275void OpenGLEventBasedCamera::DrawLDR(GLuint destinationFBO, bool updated)
276{
277 //Check if there is a need to display image on screen
278 bool display = true;
279 unsigned int dispX, dispY;
280 GLfloat dispScale;
281 if(camera != nullptr)
282 display = camera->getDisplayOnScreen(dispX, dispY, dispScale);
283
284 //Draw on screen
285 if(display)
286 {
287 // Draw color image
288 OpenGLCamera::DrawLDR(displayFBO, updated);
289
290 // Overlay event information
291 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
292 int windowHeight = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowHeight();
293 int windowWidth = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowWidth();
294
295 glm::vec4 rect((GLfloat)dispX, (GLfloat)dispY, viewportWidth*dispScale, viewportHeight*dispScale);
296 rect.y = windowHeight-rect.y-rect.w;
297 rect.x /= windowWidth;
298 rect.y /= windowHeight;
299 rect.z /= windowWidth;
300 rect.w /= windowHeight;
301
302 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, displayTex);
303 OpenGLState::BindTexture(TEX_POSTPROCESS2, GL_TEXTURE_2D, renderEventTex[1]);
304
305 OpenGLState::BindFramebuffer(destinationFBO);
306 content->SetViewportSize(windowWidth, windowHeight);
307 OpenGLState::Viewport(0, 0, windowWidth, windowHeight);
308 OpenGLState::DisableCullFace();
309 eventVisualizeShader->Use();
310 eventVisualizeShader->SetUniform("rect", rect);
311 eventVisualizeShader->SetUniform("texCamera", TEX_POSTPROCESS1);
312 eventVisualizeShader->SetUniform("texEventTimes", TEX_POSTPROCESS2);
313 content->BindBaseVertexArray();
314 glDrawArrays(GL_TRIANGLES, 0, 6);
315 OpenGLState::BindVertexArray(0);
316 OpenGLState::UseProgram(0);
317 OpenGLState::EnableCullFace();
318 OpenGLState::BindFramebuffer(0);
319
320 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
321 OpenGLState::UnbindTexture(TEX_POSTPROCESS2);
322 }
323
324 //Copy data to camera buffer
325 if(camera != nullptr && updated)
326 {
327 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, renderEventTex[0]);
328 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
329 glGetTexImage(GL_TEXTURE_2D, 0, GL_RG_INTEGER, GL_INT, NULL);
330 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
331 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
332 newData = true;
333 }
334}

336void OpenGLEventBasedCamera::Init()
337{
338 glm::uvec3 maxTextureSize = OpenGLState::GetMaxTextureSize();
339
340 eventOutputShaders = new GLSLShader*[2];
341
342 std::vector<GLSLSource> sources;
343 sources.push_back(GLSLSource(GL_COMPUTE_SHADER, "eventInit.comp"));
344 eventOutputShaders[0] = new GLSLShader(sources);
345 eventOutputShaders[0]->AddUniform("currColor", ParameterType::INT);
346 eventOutputShaders[0]->AddUniform("logLum", ParameterType::INT);
347 eventOutputShaders[0]->AddUniform("eventTimes", ParameterType::INT);
348 eventOutputShaders[0]->AddUniform("crossings", ParameterType::INT);
349
350 sources.clear();
351 std::string header = "#version 430\n#define MAX_TEXTURE_WIDTH " + std::to_string(maxTextureSize.x) + "\n";
352 sources.push_back(GLSLSource(GL_COMPUTE_SHADER, "event.comp", header));
353 eventOutputShaders[1] = new GLSLShader(sources);
354 eventOutputShaders[1]->AddUniform("currColor", ParameterType::INT);
355 eventOutputShaders[1]->AddUniform("logLum", ParameterType::INT);
356 eventOutputShaders[1]->AddUniform("events", ParameterType::INT);
357 eventOutputShaders[1]->AddUniform("eventTimes", ParameterType::INT);
358 eventOutputShaders[1]->AddUniform("crossings", ParameterType::INT);
359 eventOutputShaders[1]->AddUniform("dT", ParameterType::INT);
360 eventOutputShaders[1]->AddUniform("Tr", ParameterType::INT);
361 eventOutputShaders[1]->AddUniform("C", ParameterType::VEC2);
362 eventOutputShaders[1]->AddUniform("sigmaC", ParameterType::VEC2);
363 eventOutputShaders[1]->AddUniform("seed", ParameterType::VEC3);
364
365
366 sources.clear();
367 sources.push_back(GLSLSource(GL_VERTEX_SHADER, "texQuad.vert"));
368 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "eventVisualize.frag"));
369 eventVisualizeShader = new GLSLShader(sources);
370 eventVisualizeShader->AddUniform("rect", ParameterType::VEC4);
371 eventVisualizeShader->AddUniform("texCamera", ParameterType::INT);
372 eventVisualizeShader->AddUniform("texEventTimes", ParameterType::INT);
373}

385 delete eventVisualizeShader;
