44OpenGLFLS::OpenGLFLS(glm::vec3 eyePosition, glm::vec3 direction, glm::vec3 sonarUp,
45 GLfloat horizontalFOVDeg, GLfloat verticalFOVDeg, GLuint numOfBeams, GLuint numOfBins, glm::vec2 range_)
46: OpenGLSonar(eyePosition, direction, sonarUp, glm::uvec2(2*numOfBins, numOfBins), range_)
47{
48 //FLS specs
49 sonar = nullptr;
50 nBeams = numOfBeams;
51 nBins = numOfBins;
52 nBeamSamples = glm::min((GLuint)ceilf(verticalFOVDeg * (GLfloat)numOfBins * FLS_VRES_FACTOR), (GLuint)2048);
53 noise = glm::vec2(0.f);
54
55 fov.x = glm::radians(horizontalFOVDeg);
56 fov.y = glm::radians(verticalFOVDeg);
57 GLfloat hFactor = sinf(fov.x/2.f);
58 viewportWidth = (GLint)ceilf(2.f*hFactor*numOfBins);
59 UpdateTransform();
60
61 //Calculate necessary number of camera views
62 GLuint nViews = (GLuint)ceilf(horizontalFOVDeg/FLS_MAX_SINGLE_FOV);
63 GLuint beams1 = (GLuint)roundf((GLfloat)nBeams/(GLfloat)nViews);
64 GLuint beams2 = nBeams - beams1*(nViews-1);
65 nViewBeams = glm::max(beams1, beams2);
66
67 //Input shader: range + echo intensity
68 //Set number of beams
69 for(GLuint i=0; i<nViews-1; ++i)
70 {
71 SonarView sv;
72 sv.nBeams = beams1;
73 views.push_back(sv);
74 }
75 SonarView sv;
76 sv.nBeams = beams2;
77 views.push_back(sv);
78
79 //Allocate resources
80 inputRangeIntensityTex = OpenGLContent::GenerateTexture(GL_TEXTURE_2D_ARRAY, glm::uvec3(nViewBeams, (GLuint)nBeamSamples, nViews),
81 GL_RG32F, GL_RG, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
82 glGenRenderbuffers(1, &inputDepthRBO);
83 glBindRenderbuffer(GL_RENDERBUFFER, inputDepthRBO);
84 glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH_COMPONENT32F, nViewBeams, (GLuint)nBeamSamples);
85 glBindRenderbuffer(GL_RENDERBUFFER, 0);
86
87 glGenFramebuffers(1, &renderFBO);
88 OpenGLState::BindFramebuffer(renderFBO);
89 glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT, GL_RENDERBUFFER, inputDepthRBO);
90 glFramebufferTextureLayer(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, inputRangeIntensityTex, 0, 0);
91 GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
92 if(status != GL_FRAMEBUFFER_COMPLETE)
93 cError("Sonar input FBO initialization failed!");
94
95 //Setup matrices
96 GLfloat viewFovCorr = (GLfloat)nViewBeams/(GLfloat)nBeams * fov.x;
97 GLfloat near = range.x * glm::cos(glm::max(viewFovCorr/2.f, fov.y/2.f));
98 GLfloat far = range.y;
99 projection = glm::perspective(fov.y, tanf(viewFovCorr/2.f)/tanf(fov.y/2.f), near, far);
100
101 GLfloat viewFovAcc = 0.f;
102 for(size_t i=0; i<views.size(); ++i)
103 {
104 views[i].view = glm::rotate(-fov.x/2.f + viewFovAcc + viewFovCorr/2.f, glm::vec3(0.f,1.f,0.f));
105 GLfloat viewFov = (GLfloat)views[i].nBeams/(GLfloat)nBeams * fov.x;
106 viewFovAcc += viewFov;
107 }
108
109 //Output shader: sonar image data
110 outputTex[0] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(nBeams, nBins, 1),
111 GL_R32F, GL_RED, GL_FLOAT, NULL, FilteringMode::BILINEAR, false);
112 outputTex[1] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(nBeams, nBins, 1),
113 GL_R8, GL_RED, GL_UNSIGNED_BYTE, NULL, FilteringMode::TRILINEAR, false);
114
115 //Sonar display fan
116 glGenTextures(1, &displayTex);
117 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D, displayTex);
118 glPixelStorei(GL_PACK_ALIGNMENT, 1);
119 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB8, viewportWidth, viewportHeight, 0, GL_RGB, GL_UNSIGNED_BYTE, NULL); //RGB image
120 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
121 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
122 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
123 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
124 OpenGLState::UnbindTexture(TEX_BASE);
125 std::vector<FBOTexture> textures;
126 textures.push_back(FBOTexture(GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, displayTex));
127 displayFBO = OpenGLContent::GenerateFramebuffer(textures);
128
129 glGenVertexArrays(1, &displayVAO);
130 OpenGLState::BindVertexArray(displayVAO);
131 glEnableVertexAttribArray(0);
132
133 fanDiv = btMin((GLuint)ceil(horizontalFOVDeg), nBeams);
134 GLfloat fanData[(fanDiv+1)*2][4];
135 GLfloat Rmin = range.x/range.y;
136
137 //Flipped vertically to account for OpenGL window coordinates
138 for(GLuint i=0; i<fanDiv+1; ++i)
139 {
140 GLfloat alpha = fov.x/2.f - i/(GLfloat)fanDiv * fov.x;
141 //Min range edge
142 fanData[i*2][0] = -Rmin*sinf(alpha)*1.f/hFactor;
143 fanData[i*2][1] = (1.f-Rmin*cosf(alpha))*2.f-1.f;
144 fanData[i*2][2] = i/(GLfloat)fanDiv;
145 fanData[i*2][3] = 1.f;
146 //Max range edge
147 fanData[i*2+1][0] = -sinf(alpha)*1.f/hFactor;
148 fanData[i*2+1][1] = (1.f-cosf(alpha))*2.f-1.f;
149 fanData[i*2+1][2] = i/(GLfloat)fanDiv;
150 fanData[i*2+1][3] = 0.f;
151 }
152
153 glGenBuffers(1, &displayVBO);
154 glBindBuffer(GL_ARRAY_BUFFER, displayVBO);
155 glBufferData(GL_ARRAY_BUFFER, sizeof(fanData), fanData, GL_STATIC_DRAW);
156 glVertexAttribPointer(0, 4, GL_FLOAT, GL_FALSE, 4*sizeof(GLfloat), 0);
157 glBindBuffer(GL_ARRAY_BUFFER, 0);
158 OpenGLState::BindVertexArray(0);
159
160 //Load output shader
161 std::string header = "#version 430\n#define N_BINS " + std::to_string(nBins)
162 + "\n#define N_BEAM_SAMPLES " + std::to_string(nBeamSamples) + "\n";
163 std::vector<GLSLSource> sources;
164 sources.push_back(GLSLSource(GL_COMPUTE_SHADER, "sonarOutput.comp", header));
165 sonarOutputShader = new GLSLShader(sources);
166 sonarOutputShader->AddUniform("sonarInput", ParameterType::INT);
167 sonarOutputShader->AddUniform("sonarOutput", ParameterType::INT);
168 sonarOutputShader->AddUniform("beams", ParameterType::UVEC2);
169 sonarOutputShader->AddUniform("range", ParameterType::VEC3);
170 sonarOutputShader->AddUniform("gain", ParameterType::FLOAT);
171 sonarOutputShader->AddUniform("noiseSeed", ParameterType::VEC3);
172 sonarOutputShader->AddUniform("noiseStddev", ParameterType::VEC2);
173
174 sonarOutputShader->Use();
175 sonarOutputShader->SetUniform("sonarInput", TEX_POSTPROCESS1);
176 sonarOutputShader->SetUniform("sonarOutput", TEX_POSTPROCESS2);
177 sonarOutputShader->SetUniform("beams", glm::uvec2(views.front().nBeams, views.back().nBeams));
178 sonarOutputShader->SetUniform("range", glm::vec3(range.x, range.y, (range.y-range.x)/(GLfloat)nBins));
179 sonarOutputShader->SetUniform("gain", gain);
180 OpenGLState::UseProgram(0);
181
182 //Load postprocess shader
183 sources.clear();
184 sources.push_back(GLSLSource(GL_COMPUTE_SHADER, "sonarPostprocess.comp"));
185 sonarPostprocessShader = new GLSLShader(sources);
186 sonarPostprocessShader->AddUniform("sonarOutput", ParameterType::INT);
187 sonarPostprocessShader->AddUniform("sonarPost", ParameterType::INT);
188 sonarPostprocessShader->Use();
189 sonarPostprocessShader->SetUniform("sonarOutput", TEX_POSTPROCESS1);
190 sonarPostprocessShader->SetUniform("sonarPost", TEX_POSTPROCESS2);
191 OpenGLState::UseProgram(0);
192}

195{
196 delete sonarOutputShader;
197 delete sonarPostprocessShader;
198 glDeleteTextures(2, outputTex);
199}

201void OpenGLFLS::UpdateTransform()
202{
203 OpenGLSonar::UpdateTransform();
204
205 if(sonar == nullptr)
206 return;
207
208 //Update settings if necessary
209 bool updateProjection = false;
210 glm::vec3 rangeGain((GLfloat)sonar->getRangeMin(), (GLfloat)sonar->getRangeMax(), (GLfloat)sonar->getGain());
211 if(rangeGain.x != range.x)
212 {
213 range.x = rangeGain.x;
214 updateProjection = true;
215 settingsUpdated = true;
216 }
217 if(rangeGain.y != range.y)
218 {
219 range.y = rangeGain.y;
220 updateProjection = true;
221 settingsUpdated = true;
222 }
223 if(rangeGain.z != gain)
224 {
225 gain = rangeGain.z;
226 settingsUpdated = true;
227 }
228 if(updateProjection)
229 {
230 GLfloat viewFovCorr = (GLfloat)nViewBeams/(GLfloat)nBeams * fov.x;
231 GLfloat near = range.x / 2.f;
232 GLfloat far = range.y;
233 projection[0] = glm::vec4(near/(near*tanf(viewFovCorr/2.f)), 0.f, 0.f, 0.f);
234 projection[1] = glm::vec4(0.f, near/(near*tanf(fov.y/2.f)), 0.f, 0.f);
235 projection[2] = glm::vec4(0.f, 0.f, -(far + near)/(far-near), -1.f);
236 projection[3] = glm::vec4(0.f, 0.f, -2.f*far*near/(far-near), 0.f);
237
238 GLfloat fanData[(fanDiv+1)*2][4];
239 GLfloat Rmin = range.x/range.y;
240 GLfloat hFactor = sinf(fov.x/2.f);
241 //Flipped vertically to account for OpenGL window coordinates
242 for(GLuint i=0; i<fanDiv+1; ++i)
243 {
244 GLfloat alpha = fov.x/2.f - i/(GLfloat)fanDiv * fov.x;
245 //Min range edge
246 fanData[i*2][0] = -Rmin*sinf(alpha)*1.f/hFactor;
247 fanData[i*2][1] = (1.f-Rmin*cosf(alpha))*2.f-1.f;
248 fanData[i*2][2] = i/(GLfloat)fanDiv;
249 fanData[i*2][3] = 1.f;
250 //Max range edge
251 fanData[i*2+1][0] = -sinf(alpha)*1.f/hFactor;
252 fanData[i*2+1][1] = (1.f-cosf(alpha))*2.f-1.f;
253 fanData[i*2+1][2] = i/(GLfloat)fanDiv;
254 fanData[i*2+1][3] = 0.f;
255 }
256
257 glBindBuffer(GL_ARRAY_BUFFER, displayVBO);
258 glBufferSubData(GL_ARRAY_BUFFER, 0, sizeof(fanData), fanData);
259 glBindBuffer(GL_ARRAY_BUFFER, 0);
260 }
261
262 //Inform sonar to run callback
263 if(newData)
264 {
265 glBindBuffer(GL_PIXEL_PACK_BUFFER, displayPBO);
266 GLubyte* src = (GLubyte*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
267 if(src)
268 {
269 sonar->NewDataReady(src, 0);
270 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
271 }
272
273 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
274 src = (GLubyte*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
275 if(src)
276 {
277 sonar->NewDataReady(src, 1);
278 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
279 }
280 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
281 newData = false;
282 }

305void OpenGLFLS::ComputeOutput(std::vector<Renderable>& objects)
306{
307 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
308 content->SetDrawingMode(DrawingMode::RAW);
309
310 //Generate sonar input
311 OpenGLState::BindFramebuffer(renderFBO);
312 OpenGLState::Viewport(0, 0, nViewBeams, nBeamSamples);
313 glDisable(GL_DEPTH_CLAMP);
314 sonarInputShader[1]->Use();
315 sonarInputShader[1]->SetUniform("eyePos", GetEyePosition());
316 sonarInputShader[0]->Use();
317 sonarInputShader[0]->SetUniform("eyePos", GetEyePosition());
318 GLSLShader* shader;
319 for(size_t i=0; i<views.size(); ++i) //For each of the sonar views
320 {
321 //Clear color and depth for particular framebuffer layer
322 glFramebufferTextureLayer(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, inputRangeIntensityTex, 0, i);
323 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
324 //Calculate view transform
325 glm::mat4 VP = GetProjectionMatrix() * views[i].view * GetViewMatrix();
326 //Draw objects
327 for(size_t h=0; h<objects.size(); ++h)
328 {
329 if(objects[h].type != RenderableType::SOLID)
330 continue;
331 const Object& obj = content->getObject(objects[h].objectId);
332 const Look& look = content->getLook(objects[h].lookId);
333 glm::mat4 M = objects[h].model;
334 Material mat = SimulationApp::getApp()->getSimulationManager()->getMaterialManager()->getMaterial(objects[h].materialName);
335 bool normalMapping = obj.texturable && (look.normalMap > 0);
336 shader = normalMapping ? sonarInputShader[1] : sonarInputShader[0];
337 shader->Use();
338 shader->SetUniform("MVP", VP * M);
339 shader->SetUniform("M", M);
340 shader->SetUniform("N", glm::mat3(glm::transpose(glm::inverse(M))));
341 shader->SetUniform("restitution", (GLfloat)mat.restitution);
342 if(normalMapping)
343 OpenGLState::BindTexture(TEX_MAT_NORMAL, GL_TEXTURE_2D, look.normalMap);
344 content->DrawObject(objects[h].objectId, objects[h].lookId, objects[h].model);
345 }
346 }
347 glEnable(GL_DEPTH_CLAMP);
348 OpenGLState::UnbindTexture(TEX_MAT_NORMAL);
349 OpenGLState::BindFramebuffer(0);
350
351 //Compute sonar output
352 glBindImageTexture(TEX_POSTPROCESS1, inputRangeIntensityTex, 0, GL_TRUE, 0, GL_READ_ONLY, GL_RG32F);
353 glBindImageTexture(TEX_POSTPROCESS2, outputTex[0], 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_R32F);
354 sonarOutputShader->Use();
355 sonarOutputShader->SetUniform("noiseSeed", glm::vec3(randDist(randGen), randDist(randGen), randDist(randGen)));
356 sonarOutputShader->SetUniform("noiseStddev", noise); //Multiplicative, additive (0.025f, 0.035f)
357 if(settingsUpdated)
358 {
359 sonarOutputShader->SetUniform("range", glm::vec3(range.x, range.y, (range.y-range.x)/(GLfloat)nBins));
360 sonarOutputShader->SetUniform("gain", gain);
361 settingsUpdated = false;
362 }
363 glMemoryBarrier(GL_FRAMEBUFFER_BARRIER_BIT);
364 glDispatchCompute((GLuint)ceilf(nViewBeams/64.f), (GLuint)views.size(), 1);
365
366 //Postprocess sonar output
367 glBindImageTexture(TEX_POSTPROCESS1, outputTex[0], 0, GL_FALSE, 0, GL_READ_ONLY, GL_R32F);
368 glBindImageTexture(TEX_POSTPROCESS2, outputTex[1], 0, GL_FALSE, 0, GL_WRITE_ONLY, GL_R8);
369 sonarPostprocessShader->Use();
370 glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
371 glDispatchCompute((GLuint)ceilf(nBeams/16.f), (GLuint)ceilf(nBins/16.f), 1);
372
373 OpenGLState::BindFramebuffer(displayFBO);
374 OpenGLState::Viewport(0, 0, viewportWidth, viewportHeight);
375 glClear(GL_COLOR_BUFFER_BIT);
376 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, outputTex[1]);
377 glGenerateMipmap(GL_TEXTURE_2D);
378 sonarVisualizeShader->Use();
379 sonarVisualizeShader->SetUniform("texSonarData", TEX_POSTPROCESS1);
380 sonarVisualizeShader->SetUniform("colorMap", static_cast<GLint>(cMap));
381 glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
382 OpenGLState::BindVertexArray(displayVAO);
383 glDrawArrays(GL_TRIANGLE_STRIP, 0, (fanDiv+1)*2);
384 OpenGLState::BindVertexArray(0);
385 OpenGLState::BindFramebuffer(0);
386 OpenGLState::UseProgram(0);
387 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
388}

390void OpenGLFLS::DrawLDR(GLuint destinationFBO, bool updated)
391{
392 //Check if there is a need to display image on screen
393 bool display = true;
394 unsigned int dispX, dispY;
395 GLfloat dispScale;
396 if(sonar != nullptr)
397 display = sonar->getDisplayOnScreen(dispX, dispY, dispScale);
398
399 //Draw on screen
400 if(display)
401 {
402 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
403 int windowHeight = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowHeight();
404 int windowWidth = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowWidth();
405 OpenGLState::BindFramebuffer(destinationFBO);
406 content->SetViewportSize(windowWidth, windowHeight);
407 OpenGLState::Viewport(0, 0, windowWidth, windowHeight);
408 OpenGLState::DisableCullFace();
409 content->DrawTexturedQuad(dispX, dispY+viewportHeight*dispScale, viewportWidth*dispScale, -viewportHeight*dispScale, displayTex);
410 OpenGLState::EnableCullFace();
411 OpenGLState::BindFramebuffer(0);
412 }
413
414 //Copy texture to sonar buffer
415 if(sonar != nullptr && updated)
416 {
417 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, outputTex[1]);
418 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
419 glGetTexImage(GL_TEXTURE_2D, 0, GL_RED, GL_UNSIGNED_BYTE, NULL);
420 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
421 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, displayTex);
422 glBindBuffer(GL_PIXEL_PACK_BUFFER, displayPBO);
423 glGetTexImage(GL_TEXTURE_2D, 0, GL_RGB, GL_UNSIGNED_BYTE, NULL);
424 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
425 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
426 newData = true;
427 }
428}
