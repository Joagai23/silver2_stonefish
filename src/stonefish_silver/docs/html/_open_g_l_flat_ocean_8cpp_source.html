38OpenGLFlatOcean::OpenGLFlatOcean(GLfloat size) : OpenGLOcean(200000.f)
39{

46 std::vector<GLuint> precompiled;

190void OpenGLFlatOcean::DrawSurface(OpenGLView* view)
191{
192 GLint* viewport = view->GetViewport();
193 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D_ARRAY, oceanTextures[3]);
194 OpenGLState::BindTexture(TEX_POSTPROCESS2, GL_TEXTURE_3D, oceanTextures[2]);
195
196 oceanShaders["surface"]->Use();
197 oceanShaders["surface"]->SetUniform("MVP", view->GetProjectionMatrix() * view->GetViewMatrix());
198 oceanShaders["surface"]->SetUniform("MV", glm::mat3(glm::transpose(glm::inverse(view->GetViewMatrix()))));
199 oceanShaders["surface"]->SetUniform("FC", view->GetLogDepthConstant());
200 oceanShaders["surface"]->SetUniform("size", oceanSize);
201 oceanShaders["surface"]->SetUniform("viewport", glm::vec2((GLfloat)viewport[2], (GLfloat)viewport[3]));
202 oceanShaders["surface"]->SetUniform("eyePos", view->GetEyePosition());
203 oceanShaders["surface"]->SetUniform("viewDir", view->GetLookingDirection());
204 oceanShaders["surface"]->SetUniform("gridSizes", params.gridSizes);
205 oceanShaders["surface"]->SetUniform("texWaveFFT", TEX_POSTPROCESS1);
206 oceanShaders["surface"]->SetUniform("texSlopeVariance", TEX_POSTPROCESS2);
207 OpenGLState::BindVertexArray(vao);
208 glDrawArrays(GL_TRIANGLES, 0, 12);
209 OpenGLState::BindVertexArray(0);
210
211 OpenGLState::UseProgram(0);
212 OpenGLState::UnbindTexture(TEX_POSTPROCESS2);
213 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
214 delete [] viewport;
215}

275void OpenGLFlatOcean::DrawUnderwaterMask(OpenGLView* view)
276{
277 glColorMask(GL_FALSE, GL_FALSE, GL_FALSE, GL_FALSE);
278 oceanShaders["mask"]->Use();
279 oceanShaders["mask"]->SetUniform("MVP", view->GetProjectionMatrix() * view->GetViewMatrix());
280 oceanShaders["mask"]->SetUniform("FC", view->GetLogDepthConstant());
281 oceanShaders["mask"]->SetUniform("size", oceanSize);
282
283 //1. Draw surface to depth buffer
284 OpenGLState::BindVertexArray(vao);
285 glDrawArrays(GL_TRIANGLES, 0, 12);
286 OpenGLState::BindVertexArray(0);
287
288 //2. Draw backsurface to depth and stencil buffer
289 OpenGLState::EnableStencilTest();
290 glStencilFunc(GL_ALWAYS, 1, 0xFF);
291 glStencilOp(GL_KEEP, GL_KEEP, GL_REPLACE);
292 glStencilMask(0xFF);
293 glClear(GL_STENCIL_BUFFER_BIT);
294 glCullFace(GL_FRONT);
295
296 OpenGLState::BindVertexArray(vao);
297 glDrawArrays(GL_TRIANGLES, 0, 12);
298 OpenGLState::BindVertexArray(0);
299
300 //3. Draw box around whole ocean
301 OpenGLOcean::DrawUnderwaterMask(view);
302
303 glCullFace(GL_BACK);
304 OpenGLState::DisableStencilTest();
305 glColorMask(GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE);
306 glClear(GL_DEPTH_BUFFER_BIT);
307}
