60void OpenGLOceanParticles::Create(glm::vec3 eyePos)
61{
62 lastEyePos = eyePos;
63 //Create particles randomly (uniformly) distributed inside a sphere
64 glBindBuffer(GL_SHADER_STORAGE_BUFFER, particlePosSSBO);
65 glm::vec4* positions = (glm::vec4*)glMapBufferRange(GL_SHADER_STORAGE_BUFFER, 0, sizeof(glm::vec4) * nParticles, GL_MAP_WRITE_BIT | GL_MAP_INVALIDATE_BUFFER_BIT);
66 for(GLuint i=0; i<nParticles; ++i)
67 {
68 GLfloat r = cbrtf(uniformd(generator)) * range; //cbrtf for uniform distribution in sphere volume
69 *(positions++) = glm::vec4(r * glm::normalize(glm::vec3(normald(generator), normald(generator), normald(generator))) + eyePos, uniformd(generator)*0.005f + 0.002f);
70 }
71 glUnmapBuffer(GL_SHADER_STORAGE_BUFFER);
72
73 glBindBuffer(GL_SHADER_STORAGE_BUFFER, particleVelSSBO);
74 glm::vec4* velocities = (glm::vec4*)glMapBufferRange(GL_SHADER_STORAGE_BUFFER, 0, sizeof(glm::vec4) * nParticles, GL_MAP_WRITE_BIT | GL_MAP_INVALIDATE_BUFFER_BIT);
75 memset(velocities, 0, sizeof(glm::vec4) * nParticles);
76 glUnmapBuffer(GL_SHADER_STORAGE_BUFFER);
77 glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
78 initialised = true;
79}

81void OpenGLOceanParticles::Update(OpenGLView* view, GLfloat dt)
82{
83 glm::vec3 eyePos = view->GetEyePosition();
84 lastEyePos = eyePos;
85
86 //Check if ever updated
87 if(!initialised)
88 {
89 Create(eyePos);
90 return;
91 }
92
93 //Move particles
94 updateShader->Use();
95 updateShader->SetUniform("dt", dt);
96 updateShader->SetUniform("numParticles", nParticles);
97 updateShader->SetUniform("eyePos", eyePos);
98 updateShader->SetUniform("R", range);
99 OpenGLState::BindTexture(TEX_MAT_ALBEDO, GL_TEXTURE_3D, noiseTexture);
100 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_PARTICLE_POS, particlePosSSBO);
101 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_PARTICLE_VEL, particleVelSSBO);
102 glDispatchCompute((GLuint)ceil(nParticles/256.0), 1, 1);
103 OpenGLState::UnbindTexture(TEX_MAT_ALBEDO);
104 OpenGLState::UseProgram(0);
105 glMemoryBarrier(GL_SHADER_STORAGE_BARRIER_BIT);
106}

108void OpenGLOceanParticles::Draw(OpenGLView* view, OpenGLOcean* glOcn)
109{
110 glm::mat4 MV = view->GetViewMatrix();
111 renderShader->Use();
112 renderShader->SetUniform("MV", MV);
113 renderShader->SetUniform("iMV", glm::inverse(MV));
114 renderShader->SetUniform("P", view->GetProjectionMatrix());
115 renderShader->SetUniform("FC", view->GetLogDepthConstant());
116 renderShader->SetUniform("eyePos", view->GetEyePosition());
117 renderShader->SetUniform("viewDir", view->GetLookingDirection());
118 renderShader->SetUniform("cWater", glOcn->getLightAttenuation());
119 renderShader->SetUniform("bWater", glOcn->getLightScattering());
120 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_PARTICLE_POS, particlePosSSBO);
121 OpenGLState::BindTexture(TEX_MAT_ALBEDO, GL_TEXTURE_2D, flakeTexture);
122 OpenGLState::EnableBlend();
123 glBlendFunc(GL_ONE, GL_ONE);
124 OpenGLState::BindVertexArray(particleVAO);
125 glDrawElements(GL_TRIANGLES, nParticles * 6, GL_UNSIGNED_INT, 0);
126 OpenGLState::BindVertexArray(0);
127 OpenGLState::DisableBlend();
128 OpenGLState::UnbindTexture(TEX_MAT_ALBEDO);
129 OpenGLState::UseProgram(0);
130}

151void OpenGLOceanParticles::Init()
152{
153 //Load shaders
154 std::vector<GLuint> precompiled;
155 precompiled.push_back(OpenGLAtmosphere::getAtmosphereAPI());
156
157 std::vector<GLSLSource> sources;
158 sources.push_back(GLSLSource(GL_COMPUTE_SHADER, "oceanParticle.comp"));
159 updateShader = new GLSLShader(sources);
160 updateShader->AddUniform("dt", ParameterType::FLOAT);
161 updateShader->AddUniform("numParticles", ParameterType::UINT);
162 updateShader->AddUniform("eyePos", ParameterType::VEC3);
163 updateShader->AddUniform("R", ParameterType::FLOAT);
164 updateShader->AddUniform("texNoise", ParameterType::INT);
165 updateShader->AddUniform("invNoiseSize", ParameterType::FLOAT);
166 updateShader->BindUniformBlock("OceanCurrents", UBO_OCEAN_CURRENTS);
167 updateShader->BindShaderStorageBlock("Positions", SSBO_PARTICLE_POS);
168 updateShader->BindShaderStorageBlock("Velocities", SSBO_PARTICLE_VEL);
169
170 GLuint noiseSize = 16;
171 updateShader->Use();
172 updateShader->SetUniform("texNoise", TEX_MAT_ALBEDO);
173 updateShader->SetUniform("invNoiseSize", 1.f/(GLfloat)noiseSize);
174 OpenGLState::UseProgram(0);
175
176 sources.clear();
177 sources.push_back(GLSLSource(GL_VERTEX_SHADER, "particle.vert"));
178 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "lightingNoShadow.frag"));
179 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "oceanOptics.frag"));
180 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "oceanSurfaceFlat.glsl"));
181 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "oceanParticle.frag"));
182 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "materialUUv.frag"));
183
184 renderShader = new GLSLShader(sources, precompiled);
185 renderShader->AddUniform("MV", ParameterType::MAT4);
186 renderShader->AddUniform("iMV", ParameterType::MAT4);
187 renderShader->AddUniform("P", ParameterType::MAT4);
188 renderShader->AddUniform("FC", ParameterType::FLOAT);
189 renderShader->AddUniform("eyePos", ParameterType::VEC3);
190 renderShader->AddUniform("viewDir", ParameterType::VEC3);
191 renderShader->AddUniform("color", ParameterType::VEC4);
192 renderShader->AddUniform("texAlbedo", ParameterType::INT);
193 renderShader->AddUniform("enableAlbedoTex", ParameterType::BOOLEAN);
194 renderShader->AddUniform("enableNormalTex", ParameterType::BOOLEAN);
195 renderShader->AddUniform("reflectivity", ParameterType::FLOAT);
196 renderShader->AddUniform("cWater", ParameterType::VEC3);
197 renderShader->AddUniform("bWater", ParameterType::VEC3);
198 renderShader->AddUniform("transmittance_texture", ParameterType::INT);
199 renderShader->AddUniform("scattering_texture", ParameterType::INT);
200 renderShader->AddUniform("irradiance_texture", ParameterType::INT);
201 renderShader->BindUniformBlock("SunSky", UBO_SUNSKY);
202 renderShader->BindUniformBlock("Lights", UBO_LIGHTS);
203 renderShader->BindShaderStorageBlock("Positions", SSBO_PARTICLE_POS);
204
205 renderShader->Use();
206 renderShader->SetUniform("texAlbedo", TEX_MAT_ALBEDO);
207 renderShader->SetUniform("enableAlbedoTex", true);
208 renderShader->SetUniform("enableNormalTex", false);
209 renderShader->SetUniform("transmittance_texture", TEX_ATM_TRANSMITTANCE);
210 renderShader->SetUniform("scattering_texture", TEX_ATM_SCATTERING);
211 renderShader->SetUniform("irradiance_texture", TEX_ATM_IRRADIANCE);
212 renderShader->SetUniform("reflectivity", 0.f);
213 renderShader->SetUniform("color", glm::vec4(0.f,0.f,0.f,0.3f));
214 OpenGLState::UseProgram(0);
215
216 sources.clear();
217 sources.push_back(GLSLSource(GL_VERTEX_SHADER, "particle.vert"));
218 sources.push_back(GLSLSource(GL_FRAGMENT_SHADER, "oceanParticleSegmentation.frag"));
219
220 renderIdShader = new GLSLShader(sources);
221 renderIdShader->AddUniform("MV", ParameterType::MAT4);
222 renderIdShader->AddUniform("iMV", ParameterType::MAT4);
223 renderIdShader->AddUniform("P", ParameterType::MAT4);
224 renderIdShader->AddUniform("FC", ParameterType::FLOAT);
225 renderIdShader->AddUniform("texAlbedo", ParameterType::INT);
226 renderIdShader->AddUniform("objectId", ParameterType::UINT);
227 renderIdShader->BindShaderStorageBlock("Positions", SSBO_PARTICLE_POS);
228
229 renderIdShader->Use();
230 renderIdShader->SetUniform("texAlbedo", TEX_MAT_ALBEDO);
231 OpenGLState::UseProgram(0);
232
233 //Load textures
234 flakeTexture = OpenGLContent::LoadInternalTexture("flake.png", true, true);
235
236 unsigned int seed = (unsigned int)GetTimeInMicroseconds();
237 std::mt19937 generator(seed);
238 std::uniform_int_distribution<int8_t> dist(-127,127);
239 glm::uvec3 noiseSize3(noiseSize, noiseSize, noiseSize);
240 int8_t* noiseData = new int8_t[noiseSize3.x * noiseSize3.y * noiseSize3.z * 4];
241 int8_t *ptr = noiseData;
242 for(unsigned int z=0; z<noiseSize3.z; ++z)
243 for(unsigned int y=0; y<noiseSize3.y; ++y)
244 for(unsigned int x=0; x<noiseSize3.x; ++x)
245 {
246 *ptr++ = dist(generator);
247 *ptr++ = dist(generator);
248 *ptr++ = dist(generator);
249 *ptr++ = dist(generator);
250 }
251 noiseTexture = OpenGLContent::GenerateTexture(GL_TEXTURE_3D, noiseSize3,
252 GL_RGBA8_SNORM, GL_RGBA, GL_BYTE, noiseData, FilteringMode::BILINEAR, true);
253 delete [] noiseData;
254}

256void OpenGLOceanParticles::Destroy()
257{
258 if(updateShader != nullptr) delete updateShader;
259 if(renderShader != nullptr) delete renderShader;
260 if(renderIdShader != nullptr) delete renderIdShader;
261 if(flakeTexture != 0) glDeleteTextures(1, &flakeTexture);
262 if(noiseTexture != 0) glDeleteTextures(1, &noiseTexture);
263}
