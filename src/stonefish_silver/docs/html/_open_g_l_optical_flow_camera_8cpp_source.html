43OpenGLOpticalFlowCamera::OpenGLOpticalFlowCamera(glm::vec3 eyePosition, glm::vec3 direction, glm::vec3 cameraUp,
44 GLint originX, GLint originY, GLint width, GLint height,
45 GLfloat horizontalFOVDeg, glm::vec2 range, bool continuousUpdate)
46 : OpenGLView(originX, originY, width, height), randDist(0.f, 1.f)
47{
48 _needsUpdate = false;
49 continuous = continuousUpdate;
50 newData = false;
51 camera = nullptr;
52 noiseVel = glm::vec2(0.f);
53 maxVel = width/2.f;
54 this->range = range;
55
56 SetupCamera(eyePosition, direction, cameraUp);
57 UpdateTransform();
58
59 fov.x = horizontalFOVDeg/180.f*M_PI;
60 fov.y = 2.f * atanf( (GLfloat)viewportHeight/(GLfloat)viewportWidth * tanf(fov.x/2.f) );
61 projection = glm::perspectiveFov(fov.y, (GLfloat)viewportWidth, (GLfloat)viewportHeight, range.x, range.y);
62 focalLength = ((GLfloat)viewportWidth/2.f)/tanf(fov.x/2.f);
63
64 //Direct flow velocity output
65 renderFlowTex[0] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
66 GL_RG32F, GL_RG, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
67 renderFlowTex[1] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
68 GL_RG32F, GL_RG, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
69 renderDepthTex = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0),
70 GL_DEPTH_COMPONENT32F, GL_DEPTH_COMPONENT, GL_FLOAT, NULL, FilteringMode::NEAREST, false);
71 std::vector<FBOTexture> fboTextures;
72 fboTextures.push_back(FBOTexture(GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, renderFlowTex[0]));
73 fboTextures.push_back(FBOTexture(GL_COLOR_ATTACHMENT1, GL_TEXTURE_2D, renderFlowTex[1]));
74 fboTextures.push_back(FBOTexture(GL_DEPTH_ATTACHMENT, GL_TEXTURE_2D, renderDepthTex));
75 renderFBO = OpenGLContent::GenerateFramebuffer(fboTextures);
76
77 //Flow visualization with color map
78 displayFlowTex = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3(viewportWidth, viewportHeight, 0), GL_RGB8, GL_RGB, GL_UNSIGNED_BYTE, NULL, FilteringMode::BILINEAR, false);
79
80 fboTextures.clear();
81 fboTextures.push_back(FBOTexture(GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, displayFlowTex));
82 displayFBO = OpenGLContent::GenerateFramebuffer(fboTextures);
83
84 //Display quad
85 GLfloat quadData[4][4];
86 quadData[0][0] = -1.f;
87 quadData[0][1] = 1.f;
88 quadData[0][2] = 0.f;
89 quadData[0][3] = 1.f;
90 quadData[1][0] = -1.f;
91 quadData[1][1] = -1.f;
92 quadData[1][2] = 0.f;
93 quadData[1][3] = 0.f;
94 quadData[2][0] = 1.f;
95 quadData[2][1] = 1.f;
96 quadData[2][2] = 1.f;
97 quadData[2][3] = 1.f;
98 quadData[3][0] = 1.f;
99 quadData[3][1] = -1.f;
100 quadData[3][2] = 1.f;
101 quadData[3][3] = 0.f;
102
103 glGenVertexArrays(1, &displayVAO);
104 OpenGLState::BindVertexArray(displayVAO);
105 glEnableVertexAttribArray(0);
106 glGenBuffers(1, &displayVBO);
107 glBindBuffer(GL_ARRAY_BUFFER, displayVBO);
108 glBufferData(GL_ARRAY_BUFFER, sizeof(quadData), quadData, GL_STATIC_DRAW);
109 glVertexAttribPointer(0, 4, GL_FLOAT, GL_FALSE, 4*sizeof(GLfloat), 0);
110 glBindBuffer(GL_ARRAY_BUFFER, 0);
111 OpenGLState::BindVertexArray(0);
112}

138void OpenGLOpticalFlowCamera::UpdateTransform()
139{
140 eye = tempEye;
141 dir = tempDir;
142 up = tempUp;
143 SetupCamera();
144
145 //Inform camera to run callback
146 if(newData)
147 {
148 glBindBuffer(GL_PIXEL_PACK_BUFFER, displayPBO);
149 GLubyte* src = (GLubyte*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
150 if(src)
151 {
152 camera->NewDataReady(src, 0);
153 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
154 }
155
156 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
157 GLfloat* src2 = (GLfloat*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
158 if(src2)
159 {
160 camera->NewDataReady(src2, 1);
161 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
162 }
163 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
164 newData = false;
165 }
166}

264void OpenGLOpticalFlowCamera::ComputeOutput(std::vector<Renderable>& objects)
265{
266 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
267 content->SetCurrentView(this);
268 content->SetDrawingMode(DrawingMode::RAW);
269
270 //Optical flow velocity output
271 OpenGLState::BindFramebuffer(renderFBO);
272 OpenGLState::Viewport(0, 0, viewportWidth, viewportHeight);
273 glDrawBuffer(GL_COLOR_ATTACHMENT0);
274 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
275 glm::mat4 VP = GetProjectionMatrix() * GetViewMatrix();
276
277 opticalFlowCameraOutputShader->Use();
278 opticalFlowCameraOutputShader->SetUniform("FC", GetLogDepthConstant());
279 opticalFlowCameraOutputShader->SetUniform("VR", glm::mat3(GetViewMatrix()));
280 opticalFlowCameraOutputShader->SetUniform("d", GetLookingDirection());
281 opticalFlowCameraOutputShader->SetUniform("c", glm::vec2(viewportWidth/2.f, viewportHeight/2.f));
282 opticalFlowCameraOutputShader->SetUniform("f", focalLength);
283 opticalFlowCameraOutputShader->SetUniform("P_c", GetEyePosition());
284
285 if(camera != nullptr)
286 {
287 Vector3 linear, angular;
288 camera->getSensorVelocity(linear, angular);
289 opticalFlowCameraOutputShader->SetUniform("v_c", glVectorFromVector(linear));
290 opticalFlowCameraOutputShader->SetUniform("w_c", glVectorFromVector(angular));
291 }
292 else
293 {
294 opticalFlowCameraOutputShader->SetUniform("v_c", glm::vec3(0.f));
295 opticalFlowCameraOutputShader->SetUniform("w_c", glm::vec3(0.f));
296 }
297
298 for(size_t i=0; i<objects.size(); ++i)
299 {
300 if(objects[i].type != RenderableType::SOLID)
301 continue;
302 opticalFlowCameraOutputShader->SetUniform("MVP", VP * objects[i].model);
303 opticalFlowCameraOutputShader->SetUniform("M", objects[i].model);
304 opticalFlowCameraOutputShader->SetUniform("P_b", objects[i].cor);
305 opticalFlowCameraOutputShader->SetUniform("v_b", objects[i].vel);
306 opticalFlowCameraOutputShader->SetUniform("w_b", objects[i].avel);
307 content->DrawObject(objects[i].objectId, -1, objects[i].model);
308 }
309
310 //Flip image
311 glDrawBuffer(GL_COLOR_ATTACHMENT1);
312 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
313 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, renderFlowTex[0]);
314 flipShader->Use();
315 flipShader->SetUniform("texSource", TEX_POSTPROCESS1);
316 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->DrawSAQ();
317
318 //Color mapped velocity display
319 OpenGLState::BindFramebuffer(displayFBO);
320 OpenGLState::Viewport(0, 0, viewportWidth, viewportHeight);
321 glClear(GL_COLOR_BUFFER_BIT);
322 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, renderFlowTex[1]);
323 opticalFlowVisualizeShader->Use();
324 opticalFlowVisualizeShader->SetUniform("texFlow", TEX_POSTPROCESS1);
325 opticalFlowVisualizeShader->SetUniform("maxVel", maxVel);
326 glMemoryBarrier(GL_SHADER_IMAGE_ACCESS_BARRIER_BIT);
327 OpenGLState::BindVertexArray(displayVAO);
328 glDrawArrays(GL_TRIANGLE_STRIP, 0, 4);
329 OpenGLState::BindVertexArray(0);
330 OpenGLState::BindFramebuffer(0);
331 OpenGLState::UseProgram(0);
332 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
333}

335void OpenGLOpticalFlowCamera::DrawLDR(GLuint destinationFBO, bool updated)
336{
337 //Check if there is a need to display image on screen
338 bool display = true;
339 unsigned int dispX, dispY;
340 GLfloat dispScale;
341 if(camera != nullptr)
342 display = camera->getDisplayOnScreen(dispX, dispY, dispScale);
343
344 //Draw on screen
345 if(display)
346 {
347 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
348 int windowHeight = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowHeight();
349 int windowWidth = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowWidth();
350 OpenGLState::BindFramebuffer(destinationFBO);
351 OpenGLState::Viewport(0, 0, windowWidth, windowHeight);
352 OpenGLState::DisableCullFace();
353 content->DrawTexturedQuad(dispX, dispY+viewportHeight*dispScale, viewportWidth*dispScale, -viewportHeight*dispScale, displayFlowTex);
354 OpenGLState::EnableCullFace();
355 OpenGLState::BindFramebuffer(0);
356 }
357
358 //Copy texture to camera buffer
359 if(camera != nullptr && updated)
360 {
361 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, renderFlowTex[1]);
362 glBindBuffer(GL_PIXEL_PACK_BUFFER, outputPBO);
363 glGetTexImage(GL_TEXTURE_2D, 0, GL_RG, GL_FLOAT, NULL);
364 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
365 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, displayFlowTex);
366 glBindBuffer(GL_PIXEL_PACK_BUFFER, displayPBO);
367 glGetTexImage(GL_TEXTURE_2D, 0, GL_RGB, GL_UNSIGNED_BYTE, NULL);
368 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
369 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
370 newData = true;
371 }
372}

375void OpenGLOpticalFlowCamera::Init()
376{
377 opticalFlowCameraOutputShader = new GLSLShader("opticalFlow.frag", "opticalFlow.vert");
378 opticalFlowCameraOutputShader->AddUniform("MVP", ParameterType::MAT4);
379 opticalFlowCameraOutputShader->AddUniform("M", ParameterType::MAT4);
380 opticalFlowCameraOutputShader->AddUniform("FC", ParameterType::FLOAT);
381 opticalFlowCameraOutputShader->AddUniform("VR", ParameterType::MAT3);
382 opticalFlowCameraOutputShader->AddUniform("d", ParameterType::VEC3);
383 opticalFlowCameraOutputShader->AddUniform("P_b", ParameterType::VEC3);
384 opticalFlowCameraOutputShader->AddUniform("v_b", ParameterType::VEC3);
385 opticalFlowCameraOutputShader->AddUniform("w_b", ParameterType::VEC3);
386 opticalFlowCameraOutputShader->AddUniform("P_c", ParameterType::VEC3);
387 opticalFlowCameraOutputShader->AddUniform("v_c", ParameterType::VEC3);
388 opticalFlowCameraOutputShader->AddUniform("w_c", ParameterType::VEC3);
389 opticalFlowCameraOutputShader->AddUniform("c", ParameterType::VEC2);
390 opticalFlowCameraOutputShader->AddUniform("f", ParameterType::FLOAT);
391
392 opticalFlowVisualizeShader = new GLSLShader("opticalFlowVisualize.frag");
393 opticalFlowVisualizeShader->AddUniform("texFlow", ParameterType::INT);
394 opticalFlowVisualizeShader->AddUniform("maxVel", ParameterType::FLOAT);
395
396 flipShader = new GLSLShader("verticalFlip.frag");
397 flipShader->AddUniform("texSource", ParameterType::INT);
398}
