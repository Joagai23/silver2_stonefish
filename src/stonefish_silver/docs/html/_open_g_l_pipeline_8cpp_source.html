54OpenGLPipeline::OpenGLPipeline(RenderSettings s, HelperSettings h) : rSettings(s), hSettings(h)
55{
56 drawingQueueMutex = SDL_CreateMutex();
57
58 //Set default OpenGL options
59 cInfo("Initialising OpenGL rendering pipeline...");
60
61 //Load shaders and create rendering buffers
62 cInfo("Loading shaders...");
63 OpenGLAtmosphere::Init();
64 OpenGLCamera::Init(rSettings);
65 OpenGLDepthCamera::Init();
66 OpenGLThermalCamera::Init();
67 OpenGLOpticalFlowCamera::Init();
68 OpenGLSegmentationCamera::Init();
69 OpenGLEventBasedCamera::Init();
70 OpenGLSonar::Init();
71 OpenGLOceanParticles::Init();
72 content = new OpenGLContent();
73
74 //Create display framebuffer
75 glGenFramebuffers(1, &screenFBO);
76 OpenGLState::BindFramebuffer(screenFBO);
77 glGenTextures(1, &screenTex);
78 OpenGLState::BindTexture(TEX_BASE, GL_TEXTURE_2D, screenTex);
79 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB8, rSettings.windowW, rSettings.windowH, 0, GL_RGB, GL_UNSIGNED_BYTE, NULL);
80 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR); //Cheaper/better gaussian blur for GUI background
81 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR); //Cheaper/better gaussian blur for GUI background
82 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
83 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
84 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, screenTex, 0);
85 OpenGLState::UnbindTexture(TEX_BASE);
86 GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
87 if(status != GL_FRAMEBUFFER_COMPLETE)
88 cError("Display FBO initialization failed!");
89 OpenGLState::BindFramebuffer(0);
90 lastSimTime = Scalar(-1);
91}

166void OpenGLPipeline::PerformDrawingQueueCopy(SimulationManager* sim)
167{
168 SDL_LockMutex(drawingQueueMutex);
169
170 //Update vision sensor transforms and copy generated data to ensure consistency
171 glMemoryBarrier(GL_PIXEL_BUFFER_BARRIER_BIT);
172 for(unsigned int i=0; i < content->getViewsCount(); ++i)
173 content->getView(i)->UpdateTransform();
174 //Update light transforms to ensure consistency
175 for(unsigned int i=0; i < content->getLightsCount(); ++i)
176 content->getLight(i)->UpdateTransform();
177 //Update ocean currents for particle systems
178 Ocean* ocean = sim->getOcean();
179 if(ocean != NULL) ocean->UpdateCurrentsData();
180
181 if(!drawingQueue.empty())
182 {
183 drawingQueueCopy.clear();
184 selectedDrawingQueueCopy.clear();
185 //Double buffering
186 drawingQueueCopy.insert(drawingQueueCopy.end(), drawingQueue.begin(), drawingQueue.end());
187 selectedDrawingQueueCopy.insert(selectedDrawingQueueCopy.end(), selectedDrawingQueue.begin(), selectedDrawingQueue.end());
188 //Enable update of drawing queue by clearing old queue
189 drawingQueue.clear();
190 selectedDrawingQueue.clear();
191 }
192
193 SDL_UnlockMutex(drawingQueueMutex);
194
195 //Sort objects by material to reduce uniform/texture switching
196 std::sort(drawingQueueCopy.begin(), drawingQueueCopy.end(), Renderable::SortByMaterial);
197}

221void OpenGLPipeline::DrawHelpers()
222{
223 //Coordinate systems
224 if(hSettings.showCoordSys)
225 {
226 content->DrawCoordSystem(glm::mat4(1.f), 1.f);
227
228 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
229 {
230 if(drawingQueueCopy[h].type == RenderableType::SOLID_CS)
231 content->DrawCoordSystem(drawingQueueCopy[h].model, 0.25f);
232 }
233 }
234
235 //Discrete and multibody joints
236 if(hSettings.showJoints)
237 {
238 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
239 {
240 if(drawingQueueCopy[h].type == RenderableType::MULTIBODY_AXIS)
241 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(1.f,0.5f,1.f,1.f), drawingQueueCopy[h].model);
242 else if(drawingQueueCopy[h].type == RenderableType::JOINT_LINES)
243 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(1.f,0.5f,1.f,1.f), drawingQueueCopy[h].model);
244 else if(drawingQueueCopy[h].type == RenderableType::PATH_POINTS)
245 content->DrawPrimitives(PrimitiveType::POINTS, drawingQueueCopy[h].points, glm::vec4(1.f,0.5f,1.f,1.f), drawingQueueCopy[h].model);
246 else if(drawingQueueCopy[h].type == RenderableType::PATH_LINE_STRIP)
247 content->DrawPrimitives(PrimitiveType::LINE_STRIP, drawingQueueCopy[h].points, glm::vec4(1.f,0.5f,1.f,1.f), drawingQueueCopy[h].model);
248 }
249 }
250
251 //Sensors
252 if(hSettings.showSensors)
253 {
254 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
255 {
256 if(drawingQueueCopy[h].type == RenderableType::SENSOR_CS)
257 content->DrawCoordSystem(drawingQueueCopy[h].model, 0.25f);
258 else if(drawingQueueCopy[h].type == RenderableType::SENSOR_POINTS)
259 content->DrawPrimitives(PrimitiveType::POINTS, drawingQueueCopy[h].points, glm::vec4(1.f,1.f,0,1.f), drawingQueueCopy[h].model);
260 else if(drawingQueueCopy[h].type == RenderableType::SENSOR_LINES)
261 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(1.f,1.f,0,1.f), drawingQueueCopy[h].model);
262 else if(drawingQueueCopy[h].type == RenderableType::SENSOR_LINE_STRIP)
263 content->DrawPrimitives(PrimitiveType::LINE_STRIP, drawingQueueCopy[h].points, glm::vec4(1.f,1.f,0,1.f), drawingQueueCopy[h].model);
264 }
265 }
266
267 //Actuators
268 if(hSettings.showActuators)
269 {
270 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
271 {
272 if(drawingQueueCopy[h].type == RenderableType::ACTUATOR_LINES)
273 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(1.f,0.5f,0,1.f), drawingQueueCopy[h].model);
274 }
275 }
276
277 //Fluid dynamics
278 if(hSettings.showFluidDynamics)
279 {
280 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
281 {
282 switch(drawingQueueCopy[h].type)
283 {
284 case RenderableType::HYDRO_CS:
285 content->DrawEllipsoid(drawingQueueCopy[h].model, glm::vec3(0.005f), glm::vec4(0.3f, 0.7f, 1.f, 1.f));
286 break;
287
288 case RenderableType::HYDRO_CYLINDER:
289 content->DrawCylinder(drawingQueueCopy[h].model, drawingQueueCopy[h].points[0], glm::vec4(0.2f, 0.5f, 1.f, 1.f));
290 break;
291
292 case RenderableType::HYDRO_ELLIPSOID:
293 content->DrawEllipsoid(drawingQueueCopy[h].model, drawingQueueCopy[h].points[0], glm::vec4(0.2f, 0.5f, 1.f, 1.f));
294 break;
295
296 case RenderableType::HYDRO_POINTS:
297 content->DrawPrimitives(PrimitiveType::POINTS, drawingQueueCopy[h].points, glm::vec4(0.3f, 0.7f, 1.f, 1.f), drawingQueueCopy[h].model);
298 break;
299
300 case RenderableType::HYDRO_LINES:
301 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(0.2f, 0.5f, 1.f, 1.f), drawingQueueCopy[h].model);
302 break;
303
304 case RenderableType::HYDRO_LINE_STRIP:
305 content->DrawPrimitives(PrimitiveType::LINE_STRIP, drawingQueueCopy[h].points, glm::vec4(0.2f, 0.5f, 1.f, 1.f), drawingQueueCopy[h].model);
306 break;
307
308 case RenderableType::HYDRO_TRIANGLES:
309 content->DrawPrimitives(PrimitiveType::TRIANGLES, drawingQueueCopy[h].points, glm::vec4(0.2f, 0.5f, 1.f, 1.f), drawingQueueCopy[h].model);
310 break;
311
312 default:
313 break;
314 }
315 }
316 }
317
318 //Forces
319 if(hSettings.showForces)
320 {
321 for(size_t h=0; h<drawingQueueCopy.size(); ++h)
322 {
323 switch(drawingQueueCopy[h].type)
324 {
325 case RenderableType::FORCE_BUOYANCY:
326 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(0.f,0.f,1.f,1.f), drawingQueueCopy[h].model);
327 break;
328
329 case RenderableType::FORCE_LINEAR_DRAG:
330 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(0.f,1.f,1.f,1.f), drawingQueueCopy[h].model);
331 break;
332
333 case RenderableType::FORCE_QUADRATIC_DRAG:
334 content->DrawPrimitives(PrimitiveType::LINES, drawingQueueCopy[h].points, glm::vec4(1.f,0.f,1.f,1.f), drawingQueueCopy[h].model);
335 break;
336
337 default:
338 break;
339 }
340 }
341 }

344void OpenGLPipeline::Render(SimulationManager* sim)
345{
346 //Update time step for animation purposes
347 Scalar now = sim->getSimulationTime();
348 Scalar dt = now-lastSimTime;
349 lastSimTime = now;
350
351 //Double-buffering of drawing queue
352 PerformDrawingQueueCopy(sim);
353
354 //Choose rendering mode
355 unsigned int renderMode = 0; //Defaults to rendering without ocean
356 Ocean* ocean = sim->getOcean();
357 if(ocean != nullptr)
358 {
359 ocean->getOpenGLOcean()->Simulate(dt);
360 renderMode = rSettings.ocean > RenderQuality::DISABLED && ocean->isRenderable() ? 1 : 0;
361 }
362 Atmosphere* atm = sim->getAtmosphere();
363 OpenGLState::EnableDepthTest();
364 OpenGLState::EnableCullFace();
365
366 //Bake shadow maps for lights (independent of view)
367 content->SetupLights();
368 if(rSettings.shadows > RenderQuality::DISABLED)
369 {
370 glCullFace(GL_FRONT);
371 glDisable(GL_DEPTH_CLAMP);
372 content->SetDrawingMode(DrawingMode::SHADOW);
373 for(size_t i=0; i<content->getLightsCount(); ++i)
374 {
375 if(content->getLight(i)->isActive())
376 content->getLight(i)->BakeShadowmap(this);
377 }
378 glEnable(GL_DEPTH_CLAMP);
379 glCullFace(GL_BACK);
380 }
381
382 //Clear display framebuffer
383 OpenGLState::BindFramebuffer(screenFBO);
384 glClear(GL_COLOR_BUFFER_BIT);
385
386 //Update the queue of views needing update
387 unsigned int updateCount = 0;
388 std::vector<unsigned int> viewsNoUpdate; //View that are not needing update but have to be displayed
389 for(int i=content->getViewsCount()-1; i >= 0; --i) //Go through views in reverse order
390 {
391 OpenGLView* view = content->getView(i);
392
393 if(!view->isEnabled()) //Skip disabled views
394 continue;
395
396 if(view->needsUpdate())
397 {
398 if(view->isContinuous()) //Has to always get updated independent from the number of views in the queue
399 {
400 if(std::find(viewsQueue.begin(), viewsQueue.end(), i) == viewsQueue.end()) //Not already in the queue
401 {
402 viewsQueue.push_front(i);
403 ++updateCount;
404 }
405 }
406 else // Will be updated now or in the following frames
407 {
408 if(std::find(viewsQueue.begin(), viewsQueue.end(), i) == viewsQueue.end()) //Not already in the queue
409 {
410 viewsQueue.push_back(i);
411 viewsNoUpdate.push_back(i);
412 }
413 }
414 }
415 else // Does not need update
416 {
417 if(std::find(viewsNoUpdate.begin(), viewsNoUpdate.end(), i) == viewsNoUpdate.end()) //Not already in the queue
418 viewsNoUpdate.push_back(i);
419 }
420 }
421
422 if(viewsQueue.size() > content->getViewsCount()) // Safety condition to avoid runaway
423 {
424 updateCount = viewsQueue.size();
425 viewsNoUpdate.clear();
426 }
427 else if(updateCount < (unsigned int)viewsQueue.size())
428 {
429 ++updateCount;
430 auto it = std::find(viewsNoUpdate.begin(), viewsNoUpdate.end(), viewsQueue[updateCount-1]);
431 if(it != viewsNoUpdate.end())
432 viewsNoUpdate.erase(std::find(viewsNoUpdate.begin(), viewsNoUpdate.end(), viewsQueue[updateCount-1]));
433 }
434
435 //Loop through all views -> trackballs, cameras, depth cameras...
436 for(unsigned int i=0; i<updateCount; ++i)
437 {
438 OpenGLState::EnableDepthTest();
439 OpenGLState::EnableCullFace();
440 OpenGLState::DisableBlend();
441 OpenGLView* view = content->getView(viewsQueue[i]);
442
443 switch(view->getType())
444 {
445 case ViewType::DEPTH_CAMERA:
446 {
447 OpenGLDepthCamera* camera = static_cast<OpenGLDepthCamera*>(view);
448 //Draw objects and compute depth data
449 camera->ComputeOutput(drawingQueueCopy);
450 //Draw camera output
451 camera->DrawLDR(screenFBO, true);
452 }
453 break;
454
455 case ViewType::THERMAL_CAMERA:
456 {
457 OpenGLThermalCamera* camera = static_cast<OpenGLThermalCamera*>(view);
458 glm::vec3 eye = camera->GetEyePosition();
459
460 if(renderMode == 1 && ocean->GetDepth(eye) > 0.f) //Camera underwater (not working)
461 {
462 //Clear camera to water temperature
463 OpenGLState::BindFramebuffer(camera->getRenderFBO());
464 glDrawBuffer(GL_COLOR_ATTACHMENT0);
465 glClearColor(ocean->getOpenGLOcean()->getWaterTemperature(), 0.f, 0.f, 0.f);
466 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
467 }
468 else //Camera above water or no ocean enabled
469 {
470 //Apply view properties
471 GLint* viewport = camera->GetViewport();
472 content->SetViewportSize(viewport[2],viewport[3]);
473
474 //Bake parallel-split shadowmaps for sun
475 if(rSettings.shadows > RenderQuality::DISABLED)
476 {
477 content->SetDrawingMode(DrawingMode::SHADOW);
478 atm->getOpenGLAtmosphere()->BakeShadowmaps(this, camera);
479 }
480 atm->getOpenGLAtmosphere()->SetupMaterialShaders();
481
482 //Clear main framebuffer and setup camera
483 OpenGLState::BindFramebuffer(camera->getRenderFBO());
484 glDrawBuffer(GL_COLOR_ATTACHMENT0);
485 glClearColor(0.f, 0.f, 0.f, 0.f);
486 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
487
488 camera->SetViewport();
489 content->SetCurrentView(camera);
490
491 //Draw scene
492 if(renderMode == 0) //NO OCEAN
493 {
494 //Render all objects
495 content->SetDrawingMode(DrawingMode::TEMPERATURE);
496 DrawObjects();
497
498 //Render sky (at the end to take profit of early bailing)
499 atm->getOpenGLAtmosphere()->DrawSkyAndSunTemperature(camera);
500 }
501 else if(renderMode == 1) //OCEAN
502 {
503 OpenGLOcean* glOcean = ocean->getOpenGLOcean();
504
505 //Update ocean for this camera
506 if(ocean->hasWaves())
507 glOcean->UpdateSurface(camera);
508
509 //Rendering only above water
510 //possible because camera near plane is (virtually) removed with logarithmic depth buffer.
511 glOcean->DrawSurfaceTemperature(camera);
512 //Draw all objects as above surface
513 //(depth testing will secure drawing only what is above water)
514 content->SetDrawingMode(DrawingMode::TEMPERATURE);
515 DrawObjects();
516
517 //Render sky (left for the end to only fill empty spaces)
518 atm->getOpenGLAtmosphere()->DrawSkyAndSunTemperature(camera);
519 }
520 }
521 //Flip thermal image and render display texture
522 camera->ComputeOutput();
523
524 //Draw camera output
525 camera->DrawLDR(screenFBO, true);
526 }
527 break;
528
529 case ViewType::OPTICAL_FLOW_CAMERA:
530 {
531 OpenGLOpticalFlowCamera* camera = static_cast<OpenGLOpticalFlowCamera*>(view);
532 //Draw objects and compute camera data
533 camera->ComputeOutput(drawingQueueCopy);
534 //Draw camera output
535 camera->DrawLDR(screenFBO, true);
536 }
537 break;
538
539 case ViewType::SEGMENTATION_CAMERA:
540 {
541 OpenGLSegmentationCamera* camera = static_cast<OpenGLSegmentationCamera*>(view);
542 //Draw objects and compute camera data
543 camera->ComputeOutput(drawingQueueCopy, ocean);
544 //Draw camera output
545 camera->DrawLDR(screenFBO, true);
546 }
547 break;
548
549 case ViewType::SONAR:
550 {
551 OpenGLSonar* sonar = static_cast<OpenGLSonar*>(view);
552 //Draw objects and compute sonar data
553 sonar->ComputeOutput(drawingQueueCopy);
554 //Draw sonar output
555 sonar->DrawLDR(screenFBO, true);
556 }
557 break;
558
559 case ViewType::CAMERA:
560 case ViewType::TRACKBALL:
561 case ViewType::EVENT_BASED_CAMERA:
562 {
563 //Apply view properties
564 OpenGLCamera* camera = static_cast<OpenGLCamera*>(view);
565 OpenGLLight::SetCamera(camera);
566 GLint* viewport = camera->GetViewport();
567 content->SetViewportSize(viewport[2],viewport[3]);
568
569 //Bake parallel-split shadowmaps for sun
570 if(rSettings.shadows > RenderQuality::DISABLED)
571 {
572 content->SetDrawingMode(DrawingMode::SHADOW);
573 atm->getOpenGLAtmosphere()->BakeShadowmaps(this, camera);
574 }
575 atm->getOpenGLAtmosphere()->SetupMaterialShaders();
576
577 //Clear main framebuffer and setup camera
578 OpenGLState::BindFramebuffer(camera->getRenderFBO());
579 camera->SetRenderBuffers(0, true, false);
580 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT | GL_STENCIL_BUFFER_BIT);
581
582 camera->SetViewport();
583 content->SetCurrentView(camera);
584
585 //Draw scene
586 if(renderMode == 0) //NO OCEAN
587 {
588 //Render all objects
589 content->SetDrawingMode(DrawingMode::FULL);
590 DrawObjects();
591 DrawLights();
592
593 //Ambient occlusion
594 if(rSettings.ao > RenderQuality::DISABLED)
595 camera->DrawAO(1.0f);
596
597 //Render sky (at the end to take profit of early bailing)
598 atm->getOpenGLAtmosphere()->DrawSkyAndSun(camera);
599 }
600 else if(renderMode == 1) //OCEAN
601 {
602 OpenGLOcean* glOcean = ocean->getOpenGLOcean();
603
604 //Update ocean for this camera
605 if(ocean->hasWaves())
606 glOcean->UpdateSurface(camera);
607
608 //Two separate rendering paths: above water and under water,
609 //possible because camera near plane is (virtually) removed with logarithmic depth buffer.
610 glm::vec3 eye = camera->GetEyePosition();
611 if(ocean->GetDepth(eye) > 0.0) //Underwater
612 {
613 content->SetDrawingMode(DrawingMode::UNDERWATER);
614 DrawObjects();
615 glOcean->DrawBackground(camera);
616 glOcean->DrawBacksurface(camera);
617 //camera->GenerateBloom();
618 DrawLights();
619
620 if(rSettings.ssr > RenderQuality::DISABLED)
621 {
622 //Linear depth front faces
623 camera->GenerateLinearDepth(true);
624
625 //Linear depth back faces
626 OpenGLState::BindFramebuffer(camera->getPostprocessFBO());
627 glClear(GL_DEPTH_BUFFER_BIT);
628 glColorMask(GL_FALSE, GL_FALSE, GL_FALSE, GL_FALSE);
629 glCullFace(GL_FRONT);
630 content->SetDrawingMode(DrawingMode::FLAT);
631 DrawObjects();
632 glCullFace(GL_BACK);
633 glColorMask(GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE);
634 camera->GenerateLinearDepth(false);
635
636 //Draw screen-space reflections
637 camera->DrawSSR();
638 }
639
640 //Draw bloom effect simulating scattering
641 //camera->DrawBloom((GLfloat)ocean->getWaterType());
642
643 //Suspended particles only below surface
644 glDepthMask(GL_FALSE);
645 glOcean->DrawParticles(camera);
646 glDepthMask(GL_TRUE);
647
648 }
649 else //Above water
650 {
651 content->SetDrawingMode(DrawingMode::UNDERWATER);
652 DrawObjects();
653 DrawLights();
654 glOcean->DrawBackground(camera);
655
656 //Draw surface to back buffer
657 camera->SetRenderBuffers(1, false, true); //Clearing color buffer
658 camera->SetRenderBuffers(1, true, false); //Color + Normal
659 glOcean->DrawSurface(camera);
660 camera->SetRenderBuffers(0, false, false); //Color only
661
662 //Blend surface on top of scene
663 OpenGLState::DisableDepthTest();
664 OpenGLState::EnableBlend();

758 }

764 view->DrawLDR(screenFBO, false);
