38OpenGLRealCamera::OpenGLRealCamera(glm::vec3 eyePosition, glm::vec3 direction, glm::vec3 cameraUp,
39 GLint x, GLint y, GLint width, GLint height,
40 GLfloat horizontalFovDeg, glm::vec2 range, bool continuousUpdate)
41 : OpenGLCamera(x, y, width, height, range)
42{
43 _needsUpdate = false;
44 newData = false;
45 continuous = continuousUpdate;
46 camera = NULL;
47 cameraFBO = 0;
48
49 //Setup view
50 SetupCamera(eyePosition, direction, cameraUp);
51 //Setup projection
52 fovx = horizontalFovDeg/180.f * M_PI;
53 GLfloat fovy = 2.f * atanf( (GLfloat)viewportHeight/(GLfloat)viewportWidth * tanf(fovx/2.f) );
54 projection = glm::perspectiveFov(fovy, (GLfloat)viewportWidth, (GLfloat)viewportHeight, near, far);
55
56 UpdateTransform();
57}

90void OpenGLRealCamera::setCamera(ColorCamera* cam)
91{
92 //Connect with camera sensor
93 camera = cam;
94
95 //Generate buffers
96 cameraColorTex[0] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3((GLuint)viewportWidth, (GLuint)viewportHeight, 0),
97 GL_RGB8, GL_RGB, GL_UNSIGNED_BYTE, NULL, sf::FilteringMode::NEAREST, false);
98 cameraColorTex[1] = OpenGLContent::GenerateTexture(GL_TEXTURE_2D, glm::uvec3((GLuint)viewportWidth, (GLuint)viewportHeight, 0),
99 GL_RGB8, GL_RGB, GL_UNSIGNED_BYTE, NULL, sf::FilteringMode::NEAREST, false);
100 std::vector<FBOTexture> textures;
101 textures.push_back(FBOTexture(GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, cameraColorTex[0]));
102 textures.push_back(FBOTexture(GL_COLOR_ATTACHMENT1, GL_TEXTURE_2D, cameraColorTex[1]));
103 cameraFBO = OpenGLContent::GenerateFramebuffer(textures);
104
105 glGenBuffers(1, &cameraPBO);
106 glBindBuffer(GL_PIXEL_PACK_BUFFER, cameraPBO);
107 glBufferData(GL_PIXEL_PACK_BUFFER, viewportWidth * viewportHeight * 3, 0, GL_STREAM_READ);
108 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
109}

133void OpenGLRealCamera::UpdateTransform()
134{
135 eye = tempEye;
136 dir = tempDir;
137 up = tempUp;
138 SetupCamera();
139
140 viewUBOData.VP = GetProjectionMatrix() * GetViewMatrix();
141 viewUBOData.eye = GetEyePosition();
142 ExtractFrustumFromVP(viewUBOData.frustum, viewUBOData.VP);
143
144 //Inform camera to run callback
145 if(newData)
146 {
147 glBindBuffer(GL_PIXEL_PACK_BUFFER, cameraPBO);
148 GLubyte* src = (GLubyte*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
149 if(src)
150 {
151 camera->NewDataReady(src);
152 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
153 }
154 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
155 newData = false;
156 }
157}

169void OpenGLRealCamera::DrawLDR(GLuint destinationFBO, bool updated)
170{
171 if(camera != nullptr && updated)
172 {
173 OpenGLCamera::DrawLDR(cameraFBO, updated);
174
175 OpenGLState::BindFramebuffer(cameraFBO);
176 OpenGLState::Viewport(0, 0, viewportWidth, viewportHeight);
177 glDrawBuffer(GL_COLOR_ATTACHMENT1);
178 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, cameraColorTex[0]);
179 flipShader->Use();
180 flipShader->SetUniform("texSource", TEX_POSTPROCESS1);
181 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->DrawSAQ();
182 OpenGLState::UseProgram(0);
183 OpenGLState::BindFramebuffer(0);
184
185 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, cameraColorTex[1]);
186 glBindBuffer(GL_PIXEL_PACK_BUFFER, cameraPBO);
187 glGetTexImage(GL_TEXTURE_2D, 0, GL_RGB, GL_UNSIGNED_BYTE, NULL);
188 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
189 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
190 newData = true;
191 }
192
193 //Check if there is a need to display image on screen
194 bool display = true;
195 unsigned int dispX, dispY;
196 GLfloat dispScale;
197 if(camera != nullptr)
198 display = camera->getDisplayOnScreen(dispX, dispY, dispScale);
199
200 //Draw on screen
201 if(display)
202 {
203 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
204 int windowHeight = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowHeight();
205 int windowWidth = ((GraphicalSimulationApp*)SimulationApp::getApp())->getWindowWidth();
206 OpenGLState::BindFramebuffer(destinationFBO);
207 OpenGLState::DisableCullFace();
208 OpenGLState::Viewport(0, 0, windowWidth, windowHeight);
209 content->SetViewportSize(windowWidth, windowHeight);
210 content->DrawTexturedQuad(dispX, dispY, viewportWidth*dispScale, viewportHeight*dispScale, cameraColorTex[0]);
211 OpenGLState::EnableCullFace();
212 OpenGLState::BindFramebuffer(0);
213 }
214}
