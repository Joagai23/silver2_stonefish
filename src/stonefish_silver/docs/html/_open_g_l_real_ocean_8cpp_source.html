190 //FFT data transfer
191 size_t fftDataSize = params.fftSize * params.fftSize * 4 * layers;
192 fftData = new GLfloat[fftDataSize];

194
195 glGenBuffers(1, &fftPBO);
196 glBindBuffer(GL_PIXEL_PACK_BUFFER, fftPBO);
197 glBufferData(GL_PIXEL_PACK_BUFFER, params.fftSize * params.fftSize * 4 * layers * sizeof(GLfloat), fftData, GL_STREAM_READ);
198 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
199
200 //Quad tree buffers
201 glGenBuffers(2, oceanBuffers);
202 //Grid vertex data (ARRAY) x2
203 size_t vertices_byte_size = sizeof(glm::vec2) * sqr(qtGridTessFactor);
204 size_t indexes_byte_size = sizeof(uint16_t) * sqr(qtGridTessFactor - 1) * 4;
205 glm::vec2 *vertices = (glm::vec2*)malloc(vertices_byte_size);
206 uint16_t *indexes = (uint16_t*)malloc(indexes_byte_size);
207 int i, j;
208
209 for(i = 0; i < qtGridTessFactor; ++i)
210 for(j = 0; j < qtGridTessFactor; ++j)
211 {
212 glm::vec2 *vertex = vertices + i * qtGridTessFactor + j;
213
214 (*vertex)[0] = (float)i / (qtGridTessFactor - 1) - 0.5f;
215 (*vertex)[1] = (float)j / (qtGridTessFactor - 1) - 0.5f;
216 }
217
218 for(i = 0; i < qtGridTessFactor - 1; ++i)
219 for(j = 0; j < qtGridTessFactor - 1; ++j)
220 {
221 uint16_t *index = indexes + 4 * i * (qtGridTessFactor - 1) + 4 * j;
222
223 index[0] = i + qtGridTessFactor * j;
224 index[1] = i + 1 + qtGridTessFactor * j;
225 index[2] = i + 1 + qtGridTessFactor * (j + 1);
226 index[3] = i + qtGridTessFactor * (j + 1);
227 }
228
229 glBindBuffer(GL_ARRAY_BUFFER, oceanBuffers[0]); //GRID_VERTICES
230 glBufferData(GL_ARRAY_BUFFER, vertices_byte_size, vertices, GL_STATIC_DRAW);
231 glBindBuffer(GL_ARRAY_BUFFER, 0);
232
233 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, oceanBuffers[1]); //GRID_INDEXES
234 glBufferData(GL_ELEMENT_ARRAY_BUFFER, indexes_byte_size, indexes, GL_STATIC_DRAW);
235 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
236
237 qtPatchIndexCount = indexes_byte_size / sizeof(uint16_t);
238 free(vertices);
239 free(indexes);
240
241 //Vertex arrays:
242 // - Terrain
243 glGenVertexArrays(1, &vao);
244 OpenGLState::BindVertexArray(vao);
245 glEnableVertexAttribArray(0);
246 glBindBuffer(GL_ARRAY_BUFFER, oceanBuffers[0]);
247 glVertexAttribPointer(0, 2, GL_FLOAT, 0, 0, 0);
248 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, oceanBuffers[1]);
249 OpenGLState::BindVertexArray(0);
250
251 InitializeSimulation();
252}

280GLfloat OpenGLRealOcean::ComputeInterpolatedWaveData(GLfloat x, GLfloat y, GLuint channel)
281{
282 //BILINEAR INTERPOLATION ACCORDING TO OPENGL SPECIFICATION (4.5)
283 //Calculate pixel cooridnates
284 //x and y are already divided by the phyiscal dimensions of the texture (represented area in [m])
285 //so they are directly texture coordinates
286 float tmp;
287
288 //First coordinate pair
289 float i0f = modff(x - 0.5f/(float)params.fftSize, &tmp);
290 float j0f = modff(y - 0.5f/(float)params.fftSize, &tmp);
291 if(i0f < 0.f) i0f = 1.f - fabsf(i0f);
292 if(j0f < 0.f) j0f = 1.f - fabsf(j0f);
293 int i0 = (int)truncf(i0f * (float)params.fftSize);
294 int j0 = (int)truncf(j0f * (float)params.fftSize);
295
296 //Second coordinate pair
297 float i1f = modff(x + 0.5f/(float)params.fftSize, &tmp);
298 float j1f = modff(y + 0.5f/(float)params.fftSize, &tmp);
299 if(i1f < 0.f) i1f = 1.f - fabsf(i1f);
300 if(j1f < 0.f) j1f = 1.f - fabsf(j1f);
301 int i1 = (int)truncf(i1f * (float)params.fftSize);
302 int j1 = (int)truncf(j1f * (float)params.fftSize);
303
304 //Calculate weigths
305 float alpha = modff(i0f * (float)params.fftSize, &tmp);
306 float beta = modff(j0f * (float)params.fftSize, &tmp);
307
308 //Get texel values
309 float t[4];
310 t[0] = fftData[(j0 * params.fftSize + i0) * 4 + channel];
311 t[1] = fftData[(j0 * params.fftSize + i1) * 4 + channel];
312 t[2] = fftData[(j1 * params.fftSize + i0) * 4 + channel];
313 t[3] = fftData[(j1 * params.fftSize + i1) * 4 + channel];
314
315 //Interpolate
316 float h = (1.f - alpha)*(1.f - beta)*t[0] + alpha*(1.f - beta)*t[1] + (1.f - alpha)*beta*t[2] + alpha*beta*t[3];
317
318 return h;
319}

321GLfloat OpenGLRealOcean::ComputeWaveHeight(GLfloat x, GLfloat y)
322{
323 //Z,X are reversed because the coordinate system used to draw ocean has Z axis pointing up!
324 GLfloat z = 0.f;
325 z -= ComputeInterpolatedWaveData(x/params.gridSizes.x, y/params.gridSizes.x, 0);
326 z -= ComputeInterpolatedWaveData(x/params.gridSizes.y, y/params.gridSizes.y, 1);
327 //The components below have low importance and were excluded to lower the computational cost
328 //z -= ComputeInterpolatedWaveData(x/params.gridSizes.z, y/params.gridSizes.z, 2);
329 //z -= ComputeInterpolatedWaveData(x/params.gridSizes.w, y/params.gridSizes.w, 3);
330 return z;
331}

333void OpenGLRealOcean::Simulate(GLfloat dt)
334{
335 if(SDL_TryLockMutex(hydroMutex) == 0)
336 {
337 glBindBuffer(GL_PIXEL_PACK_BUFFER, fftPBO);
338 GLfloat* src = (GLfloat*)glMapBuffer(GL_PIXEL_PACK_BUFFER, GL_READ_ONLY);
339 if(src)
340 {
341 memcpy(fftData, src, params.fftSize * params.fftSize * 4 * 4 * sizeof(GLfloat));
342 glUnmapBuffer(GL_PIXEL_PACK_BUFFER); //Release pointer to the mapped buffer
343 }
344 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
345 SDL_UnlockMutex(hydroMutex);
346 }
347
348 OpenGLOcean::Simulate(dt);
349
350 //Copy wave data to RAM for hydrodynamic computations
351 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D_ARRAY, oceanTextures[3]);
352 glBindBuffer(GL_PIXEL_PACK_BUFFER, fftPBO);
353 glGetTexImage(GL_TEXTURE_2D_ARRAY, 0, GL_RGBA, GL_FLOAT, NULL);
354 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
355 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
356}

358void OpenGLRealOcean::ResetSurface(OpenGLView* view)
359{
360 auto it = oceanTrees.find(view); //Check if a quad tree exists for this camera
361 if(it != oceanTrees.end())
362 {
363 //Delete buffers in GPU memory
364 glDeleteBuffers(4, it->second.patchSSBO);
365 glDeleteBuffers(1, &it->second.patchDEI);

368 //Delete map entry
369 oceanTrees.erase(it);
370 }
371}

373void OpenGLRealOcean::UpdateSurface(OpenGLView* view)
374{
375 //Check if a quad tree exists for this camera
376 OceanQT* tree;
377 try
378 {
379 tree = &oceanTrees.at(view);
380 }
381 catch(const std::out_of_range& e)
382 {
383 OceanQT newTree;
384 //Generate buffers
385 glGenBuffers(4, newTree.patchSSBO);
386 glBindBuffer(GL_SHADER_STORAGE_BUFFER, newTree.patchSSBO[0]);
387 glBufferData(GL_SHADER_STORAGE_BUFFER, sizeof(GLuint) << 16, NULL, GL_STATIC_DRAW);
388 //Init tree structure
389 const GLuint dummy[] = {0,0,0,0};
390 GLuint* first = (GLuint*)glMapBufferRange(GL_SHADER_STORAGE_BUFFER, 0, sizeof(GLuint) * 4, GL_MAP_WRITE_BIT | GL_MAP_INVALIDATE_BUFFER_BIT);
391 memcpy(first, dummy, sizeof(GLuint) * 4);
392 glUnmapBuffer(GL_SHADER_STORAGE_BUFFER);
393 glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
394
395 glBindBuffer(GL_SHADER_STORAGE_BUFFER, newTree.patchSSBO[1]);
396 glBufferData(GL_SHADER_STORAGE_BUFFER, sizeof(GLuint) << 16, NULL, GL_STATIC_DRAW);
397 glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
398
399 glBindBuffer(GL_SHADER_STORAGE_BUFFER, newTree.patchSSBO[2]);
400 glBufferData(GL_SHADER_STORAGE_BUFFER, sizeof(GLfloat) << 16, NULL, GL_STATIC_DRAW);
401 glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
402
403 GLuint one = 1;
404 glBindBuffer(GL_SHADER_STORAGE_BUFFER, newTree.patchSSBO[3]);
405 glBufferData(GL_SHADER_STORAGE_BUFFER, sizeof(GLuint), &one, GL_STATIC_DRAW);
406 glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
407
408 GLuint zero = 0;
409 glGenBuffers(1, &newTree.patchAC);
410 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, newTree.patchAC);
411 glBufferData(GL_ATOMIC_COUNTER_BUFFER, sizeof(GLuint), &zero, GL_DYNAMIC_DRAW);
412 glBindBuffer(GL_ATOMIC_COUNTER_BUFFER, 0);
413
414 DrawElementsIndirectCommand cmd;
415 cmd.count = qtPatchIndexCount;
416 cmd.instanceCount = 0;
417 cmd.firstIndex = 0;
418 cmd.baseVertex = 0;
419 cmd.baseInstance = 0;
420 glGenBuffers(1, &newTree.patchDEI);
421 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, newTree.patchDEI);
422 glBufferData(GL_DRAW_INDIRECT_BUFFER, sizeof(DrawElementsIndirectCommand), &cmd, GL_DYNAMIC_DRAW);
423 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, 0);
424
425 DispatchIndirectCommand cmd2;
426 cmd2.numGroupsX = 1;
427 cmd2.numGroupsY = 1;
428 cmd2.numGroupsZ = 1;
429 glGenBuffers(1, &newTree.patchDI);
430 glBindBuffer(GL_DISPATCH_INDIRECT_BUFFER, newTree.patchDI);
431 glBufferData(GL_DISPATCH_INDIRECT_BUFFER, sizeof(DispatchIndirectCommand), &cmd2, GL_DYNAMIC_DRAW);
432 glBindBuffer(GL_DISPATCH_INDIRECT_BUFFER, 0);
433
434 //Initialize
435 newTree.pingpong = 1;
436 oceanTrees[view] = newTree;
437 tree = &oceanTrees[view];
438 }
439
440 GLuint zero = 0;
441 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_IN, tree->patchSSBO[1 - tree->pingpong]);
442 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_OUT, tree->patchSSBO[tree->pingpong]);
443 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_CULL, tree->patchSSBO[2]);
444 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_SIZE, tree->patchSSBO[3]);
445 glBindBufferBase(GL_ATOMIC_COUNTER_BUFFER, AC_QTREE_LOD, tree->patchAC);
446 glBindBufferRange(GL_ATOMIC_COUNTER_BUFFER, AC_QTREE_CULL, tree->patchDEI, sizeof(GLuint), sizeof(GLuint));
447 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, tree->patchDEI);
448 glBufferSubData(GL_DRAW_INDIRECT_BUFFER, sizeof(GLuint), sizeof(GLuint), &zero);
449 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, 0);
450 tree->pingpong = 1 - tree->pingpong;
451
452 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D_ARRAY, oceanTextures[3]);
453 oceanShaders["lod"]->Use();
454 oceanShaders["lod"]->SetUniform("sceneSize", oceanSize);
455 oceanShaders["lod"]->SetUniform("gridSizes", params.gridSizes);
456 oceanShaders["lod"]->SetUniform("texWaveFFT", TEX_POSTPROCESS1);
457 glBindBuffer(GL_DISPATCH_INDIRECT_BUFFER, tree->patchDI);
458 glDispatchComputeIndirect(0);
459 glBindBuffer(GL_DISPATCH_INDIRECT_BUFFER, 0);
460 OpenGLState::UseProgram(0);
461 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
462 glMemoryBarrier(GL_SHADER_STORAGE_BARRIER_BIT | GL_ATOMIC_COUNTER_BARRIER_BIT);
463
464 glBindBufferRange(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_INDIRECT, tree->patchDI, 0, sizeof(GLuint));
465 oceanShaders["indirect"]->Use();
466 glDispatchCompute(1, 1, 1);
467 OpenGLState::UseProgram(0);
468 glMemoryBarrier(GL_SHADER_STORAGE_BARRIER_BIT);

587void OpenGLRealOcean::DrawUnderwaterMask(OpenGLView* view)
588{
589 OceanQT& tree = oceanTrees[view];
590 glColorMask(GL_FALSE, GL_FALSE, GL_FALSE, GL_FALSE);
591 glBindBufferBase(GL_SHADER_STORAGE_BUFFER, SSBO_QTREE_CULL, tree.patchSSBO[2]);
592 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D_ARRAY, oceanTextures[3]);
593 OpenGLState::BindVertexArray(vao);
594
595 oceanShaders["mask"]->Use();
596 oceanShaders["mask"]->SetUniform("MVP", view->GetProjectionMatrix() * view->GetViewMatrix());
597 oceanShaders["mask"]->SetUniform("FC", view->GetLogDepthConstant());
598 oceanShaders["mask"]->SetUniform("gridSizes", params.gridSizes);
599 oceanShaders["mask"]->SetUniform("texWaveFFT", TEX_POSTPROCESS1);
600 oceanShaders["mask"]->SetUniform("eyePos", view->GetEyePosition());
601 oceanShaders["mask"]->SetUniform("u_scene_size", oceanSize);
602 oceanShaders["mask"]->SetUniform("u_gpu_tess_factor", (GLfloat)qtGPUTessFactor);
603
604 //1. Draw surface to depth buffer
605 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, tree.patchDEI);
606 glDrawElementsIndirect(GL_PATCHES, GL_UNSIGNED_SHORT, NULL);
607
608 //2. Draw backsurface to depth and stencil buffer
609 OpenGLState::EnableStencilTest();
610 glStencilFunc(GL_ALWAYS, 1, 0xFF);
611 glStencilOp(GL_KEEP, GL_KEEP, GL_REPLACE);
612 glStencilMask(0xFF);
613 glClear(GL_STENCIL_BUFFER_BIT);
614 glCullFace(GL_FRONT);
615
616 glDrawElementsIndirect(GL_PATCHES, GL_UNSIGNED_SHORT, NULL);
617 glBindBuffer(GL_DRAW_INDIRECT_BUFFER, 0);
618
619 OpenGLState::BindVertexArray(0);
620 OpenGLState::UseProgram(0);
621 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
622
623 //3. Draw box around whole ocean
624 OpenGLOcean::DrawUnderwaterMask(view);
625
626 glCullFace(GL_BACK);
627 OpenGLState::DisableStencilTest();
628 glColorMask(GL_TRUE, GL_TRUE, GL_TRUE, GL_TRUE);
629 glClear(GL_DEPTH_BUFFER_BIT);
630}
