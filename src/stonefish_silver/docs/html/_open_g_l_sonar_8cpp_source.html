164void OpenGLSonar::Init()
165{
166 sonarInputShader[0] = new GLSLShader("sonarInput.frag", "sonarInput.vert");
167 sonarInputShader[0]->AddUniform("MVP", ParameterType::MAT4);
168 sonarInputShader[0]->AddUniform("M", ParameterType::MAT4);
169 sonarInputShader[0]->AddUniform("N", ParameterType::MAT3);
170 sonarInputShader[0]->AddUniform("eyePos", ParameterType::VEC3);
171 sonarInputShader[0]->AddUniform("restitution", ParameterType::FLOAT);
172
173 sonarInputShader[1] = new GLSLShader("sonarInputUv.frag", "sonarInputUv.vert");
174 sonarInputShader[1]->AddUniform("MVP", ParameterType::MAT4);
175 sonarInputShader[1]->AddUniform("M", ParameterType::MAT4);
176 sonarInputShader[1]->AddUniform("N", ParameterType::MAT3);
177 sonarInputShader[1]->AddUniform("eyePos", ParameterType::VEC3);
178 sonarInputShader[1]->AddUniform("restitution", ParameterType::FLOAT);
179 sonarInputShader[1]->AddUniform("texNormal", ParameterType::INT);
180 sonarInputShader[1]->Use();
181 sonarInputShader[1]->SetUniform("texNormal", TEX_MAT_NORMAL);
182 OpenGLState::UseProgram(0);
183
184 sonarVisualizeShader = new GLSLShader("sonarVisualize.frag", "printer.vert");
185 sonarVisualizeShader->AddUniform("texSonarData", ParameterType::INT);
186 sonarVisualizeShader->AddUniform("colorMap", ParameterType::INT);
187}

189void OpenGLSonar::Destroy()

192 if(sonarInputShader[1] != nullptr) delete sonarInputShader[1];
193 if(sonarVisualizeShader != nullptr) delete sonarVisualizeShader;
194}
