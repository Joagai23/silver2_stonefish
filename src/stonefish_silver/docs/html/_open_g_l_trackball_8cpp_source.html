39OpenGLTrackball::OpenGLTrackball(glm::vec3 centerPosition, GLfloat orbitRadius, glm::vec3 up,
40 GLint x, GLint y, GLint width, GLint height, GLfloat horizontalFovDeg,
41 glm::vec2 range) : OpenGLCamera(x, y, width, height, range)
42{
43 this->up = glm::normalize(up);
44 rotation = glm::rotation(this->up, glm::vec3(0,0,1.f));
45 center = centerPosition;
46 radius = orbitRadius;
47 fovx = horizontalFovDeg/180.f*M_PI;
48 GLfloat fovy = 2.f * atanf( (GLfloat)viewportHeight/(GLfloat)viewportWidth * tanf(fovx/2.f) );
49 projection = glm::perspectiveFov(fovy, (GLfloat)viewportWidth, (GLfloat)viewportHeight, near, far);
50 dragging = false;
51 transMode = false;
52 continuous = true;
53 holdingEntity = nullptr;
54
55 outlineShader[0] = new GLSLShader("outline.frag");
56 outlineShader[0]->AddUniform("color", ParameterType::VEC4);
57 outlineShader[0]->AddUniform("texStencil", ParameterType::INT);
58 outlineShader[0]->AddUniform("uvOffset", ParameterType::VEC2);
59
60 outlineShader[1] = new GLSLShader("smallBlur.frag");
61 outlineShader[1]->AddUniform("tex", ParameterType::INT);
62 outlineShader[1]->AddUniform("invTexSize", ParameterType::VEC2);
63
64 UpdateTransform();
65}

68{
69 if(outlineShader[0] != nullptr)
70 delete outlineShader[0];
71 if(outlineShader[1] != nullptr)
72 delete outlineShader[1];
73}

110void OpenGLTrackball::UpdateTransform()
111{
112 if(holdingEntity != nullptr) center = tempCenter;
113 trackballTransform = glm::lookAt(GetEyePosition(), center, GetUpDirection());
114
115 viewUBOData.VP = GetProjectionMatrix() * GetViewMatrix();
116 viewUBOData.eye = GetEyePosition();
117 ExtractFrustumFromVP(viewUBOData.frustum, viewUBOData.VP);
118}

120GLfloat OpenGLTrackball::calculateZ(GLfloat x, GLfloat y)
121{
122 if(x*x+y*y <= 0.5f)
123 return sqrtf(1.f-(x*x+y*y));
124 else
125 return 0.5f/sqrtf(x*x+y*y);
126}

128void OpenGLTrackball::MouseDown(GLfloat x, GLfloat y, bool translate)
129{
130 x_start = x;
131 y_start = y;
132
133 if(translate)
134 {
135 transMode = true;
136 translation_start = center;
137 }
138 else
139 {
140 transMode = false;
141 z_start = calculateZ(x_start, y_start);
142 rotation_start = rotation;
143 }
144
145 dragging = true;
146}

153void OpenGLTrackball::MouseMove(GLfloat x, GLfloat y)
154{
155 if(dragging)
156 {
157 if(transMode)
158 {
159 glm::vec3 right = glm::normalize(glm::cross(GetLookingDirection(), GetUpDirection()));
160 center = translation_start + (GetUpDirection() * (y-y_start) * -0.5f + right * (x-x_start) * -0.5f) * radius;
161 }
162 else //rotate
163 {
164 GLfloat z = calculateZ(x, y);
165 glm::quat rotation_new = glm::rotation(glm::normalize(glm::vec3(-x_start, z_start, y_start)), glm::normalize(glm::vec3(-x, z, y)));
166 rotation = rotation_new * rotation_start;
167 }
168 }
169}

202void OpenGLTrackball::DrawSelection(const std::vector<Renderable>& r, GLuint destinationFBO)
203{
204 if(r.size() == 0) //No selection
205 return;
206
207 OpenGLContent* content = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent();
208
209 //1. Draw flat shape to color and stencil buffer
210 OpenGLState::BindFramebuffer(getRenderFBO());
211 SetRenderBuffers(0, false, false);
212 OpenGLState::EnableStencilTest();
213 glStencilFunc(GL_ALWAYS, 1, 0xFF);
214 glStencilOp(GL_KEEP, GL_KEEP, GL_REPLACE);
215 glStencilMask(0xFF);
216 glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT | GL_STENCIL_BUFFER_BIT);
217 content->SetDrawingMode(DrawingMode::FLAT);
218 for(size_t i=0; i<r.size(); ++i)
219 if(r[i].type == RenderableType::SOLID)
220 content->DrawObject(r[i].objectId, -1, r[i].model);
221 OpenGLState::DisableStencilTest();
222
223 //2. Grow
224 outlineShader[0]->Use();
225 outlineShader[0]->SetUniform("color", glm::vec4(1.f,0.55f,0.1f,1.f)); //glm::vec4(0.4f,0.9f,1.f,1.f));
226 outlineShader[0]->SetUniform("texStencil", TEX_POSTPROCESS1);
227 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, getColorTexture(0));
228 SetRenderBuffers(1, false, false);
229 outlineShader[0]->SetUniform("uvOffset", glm::vec2(1.f/(GLfloat)viewportWidth, 1.f/(GLfloat)viewportHeight)*1.5f);
230 content->DrawSAQ();
231
232 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, getColorTexture(1));
233 SetRenderBuffers(0, false, false);
234 content->DrawSAQ();
235
236 //3. Cut outline
237 SetRenderBuffers(1, false, true);
238 OpenGLState::EnableStencilTest();
239 glStencilMask(0x00);
240 glStencilFunc(GL_EQUAL, 0, 0xFF);
241 content->DrawTexturedSAQ(getColorTexture(0));
242 OpenGLState::DisableStencilTest();
243
244 //4. Gaussian blur 3x3
245 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, getColorTexture(1));
246 SetRenderBuffers(0, false, false);
247 outlineShader[1]->Use();
248 outlineShader[1]->SetUniform("tex", TEX_POSTPROCESS1);
249 outlineShader[1]->SetUniform("invTexSize", glm::vec2(1.f/(GLfloat)viewportWidth, 1.f/(GLfloat)viewportHeight));
250 content->DrawSAQ();
251
252 OpenGLState::BindTexture(TEX_POSTPROCESS1, GL_TEXTURE_2D, getColorTexture(0));
253 SetRenderBuffers(1, false, false);
254 content->DrawSAQ();
255 OpenGLState::UnbindTexture(TEX_POSTPROCESS1);
256 OpenGLState::UseProgram(0);
257
258 //5. Overlay
259 OpenGLState::BindFramebuffer(destinationFBO); //No depth buffer, just one color buffer
260 OpenGLState::EnableBlend();
261 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
262 content->DrawTexturedSAQ(getColorTexture(1));
263 OpenGLState::DisableBlend();
264}
