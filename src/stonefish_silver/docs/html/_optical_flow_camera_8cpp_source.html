36OpticalFlowCamera::OpticalFlowCamera(std::string uniqueName, unsigned int resolutionX, unsigned int resolutionY, Scalar hFOVDeg, Scalar frequency,
37 Scalar minDistance, Scalar maxDistance) : Camera(uniqueName, resolutionX, resolutionY, hFOVDeg, frequency)
38{
39 depthRange = glm::vec2((GLfloat)minDistance, (GLfloat)maxDistance);
40 noiseStdDev = glm::vec2(0.f);
41 displayMaxVelocity = resolutionX/2.f;

43 flowData = nullptr;
44 displayData = nullptr;
45 glCamera = nullptr;
46}

53void OpticalFlowCamera::setNoise(float velocityXStdDev, float velocityYStdDev)
54{

88void OpticalFlowCamera::InitGraphics()
89{
90 glCamera = new OpenGLOpticalFlowCamera(glm::vec3(0,0,0), glm::vec3(0,0,1.f), glm::vec3(0,-1.f,0), 0, 0, resX, resY, (GLfloat)fovH, depthRange, freq < Scalar(0));
91 glCamera->setNoise(noiseStdDev);
92 glCamera->setMaxVelocity(displayMaxVelocity);
93 glCamera->setCamera(this);
94 UpdateTransform();
95 glCamera->UpdateTransform();
96 InternalUpdate(0);
97 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddView(glCamera);
98
99 unsigned int w, h;
100 getResolution(w, h);
101 displayData = new GLubyte[w*h*3];
102}

104void OpticalFlowCamera::SetupCamera(const Vector3& eye, const Vector3& dir, const Vector3& up)
105{
106 glm::vec3 eye_ = glm::vec3((GLfloat)eye.x(), (GLfloat)eye.y(), (GLfloat)eye.z());
107 glm::vec3 dir_ = glm::vec3((GLfloat)dir.x(), (GLfloat)dir.y(), (GLfloat)dir.z());
108 glm::vec3 up_ = glm::vec3((GLfloat)up.x(), (GLfloat)up.y(), (GLfloat)up.z());
109 glCamera->SetupCamera(eye_, dir_, up_);
110}

117void OpticalFlowCamera::NewDataReady(void* data, unsigned int index)
118{
119 if(newDataCallback != nullptr)
120 {
121 if(index == 0)
122 {
123 unsigned int w, h;
124 getResolution(w, h);
125 memcpy(displayData, data, w*h*3);
126 }
127 else
128 {
129 flowData = (GLfloat*)data;
130 newDataCallback(this);
131 flowData = nullptr;
132 }
133 }
134}
