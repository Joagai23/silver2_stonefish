49void PWLTrajectory::AddKeyPoint(Scalar keyTime, Transform keyTransform)
50{
51 //Check if time correct
52 if(keyTime < Scalar(0)) return;
53
54 //Create key point
55 KeyPoint k;
56 k.t = keyTime;
57 k.T = keyTransform;
58
59 //Add to the list
60 auto it = std::find(points.begin(), points.end(), k);
61 if(it != points.end())
62 *it = k;
63 else
64 points.push_back(k);
65
66 //Sort key points by time
67 std::sort(points.begin(), points.end());
68
69 //Reset
70 playTime = Scalar(0);
71 endTime = points.back().t;
72 forward = true;
73 BuildGraphicalPath();
74 Interpolate();
75}

77void PWLTrajectory::Interpolate()
78{
79 if(points.size() == 1)
80 {
81 interpTrans = points[0].T;
82 interpVel = V0();
83 interpAngVel = V0();
84 return;
85 }
86
87 //Find current path segment
88 auto it = std::find_if(points.begin(), points.end(),
89 [&](const auto& key){ return key.t >= playTime; });
90
91 if(it->t == playTime) //No interpolation needed
92 {
93 interpTrans = it->T;
94 if(it == points.begin()) //Time = 0
95 calculateVelocityShortestPath(it->T, (it+1)->T, (it+1)->t, interpVel, interpAngVel);
96 else
97 calculateVelocityShortestPath((it-1)->T, it->T, it->t-(it-1)->t, interpVel, interpAngVel);
98 }
99 else
100 {
101 Scalar alpha = (playTime - (it-1)->t)/(it->t - (it-1)->t);
102 interpTrans.setOrigin(lerp((it-1)->T.getOrigin(), it->T.getOrigin(), alpha));
103 interpTrans.setRotation(slerp((it-1)->T.getRotation(), it->T.getRotation(), alpha));
104 calculateVelocityShortestPath((it-1)->T, it->T, it->t-(it-1)->t, interpVel, interpAngVel);
105 }
106
107 if(!forward)
108 {
109 interpVel = -interpVel;
110 interpAngVel = -interpAngVel;
111 }
112}

114void PWLTrajectory::BuildGraphicalPath()
115{
116 vis[0].points.clear();
117 vis[1].points.clear();
118 for(size_t i=0; i<points.size(); ++i)
119 vis[0].points.push_back(glVectorFromVector(points[i].T.getOrigin()));
120 vis[1].points = vis[0].points;
121}
