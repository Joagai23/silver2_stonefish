33PrismaticJoint::PrismaticJoint(std::string uniqueName, SolidEntity* solidA, SolidEntity* solidB, const Vector3& axis, bool collideLinked) : Joint(uniqueName, collideLinked)
34{
35 btRigidBody* bodyA = solidA->rigidBody;
36 btRigidBody* bodyB = solidB->rigidBody;
37
38 Vector3 sliderAxis = axis.normalized();
39 Vector3 v2;
40 if(fabs(sliderAxis.z()) > Scalar(0.8)) v2 = Vector3(1,0,0); else v2 = Vector3(0,0,1);
41 Vector3 v3 = (sliderAxis.cross(v2)).normalized();
42 v2 = (v3.cross(sliderAxis)).normalized();
43 Matrix3 sliderBasis(sliderAxis.x(), v2.x(), v3.x(),
44 sliderAxis.y(), v2.y(), v3.y(),
45 sliderAxis.z(), v2.z(), v3.z());
46 Transform sliderFrame(sliderBasis, (bodyA->getCenterOfMassPosition() + bodyB->getCenterOfMassPosition())/Scalar(2.));
47
48 Transform frameInA = bodyA->getCenterOfMassTransform().inverse() * sliderFrame;
49 Transform frameInB = bodyB->getCenterOfMassTransform().inverse() * sliderFrame;
50 axisInA = frameInA.getBasis().getColumn(0).normalized();
51
52 btSliderConstraint* slider = new btSliderConstraint(*bodyA, *bodyB, frameInA, frameInB, true);
53 slider->setLowerLinLimit(Scalar(1));
54 slider->setUpperLinLimit(Scalar(-1));
55 slider->setLowerAngLimit(Scalar(0));
56 slider->setUpperAngLimit(Scalar(0));
57 setConstraint(slider);
58
59 sigDamping = Scalar(0);
60 velDamping = Scalar(0);
61 displacementIC = Scalar(0);
62}

95void PrismaticJoint::ApplyForce(Scalar F)
96{
97 btRigidBody& bodyA = getConstraint()->getRigidBodyA();
98 btRigidBody& bodyB = getConstraint()->getRigidBodyB();
99 Vector3 axis = (bodyA.getCenterOfMassTransform().getBasis() * axisInA).normalized();
100 Vector3 force = axis * F;
101 bodyA.applyCentralForce(force);
102 bodyB.applyCentralForce(-force);
103}

105void PrismaticJoint::ApplyDamping()
106{
107 if(sigDamping > Scalar(0) || velDamping > Scalar(0))
108 {
109 btRigidBody& bodyA = getConstraint()->getRigidBodyA();
110 btRigidBody& bodyB = getConstraint()->getRigidBodyB();
111 Vector3 axis = (bodyA.getCenterOfMassTransform().getBasis() * axisInA).normalized();
112 Vector3 relativeV = bodyA.getLinearVelocity() - bodyB.getLinearVelocity();
113 Scalar v = relativeV.dot(axis);
114
115 if(v != Scalar(0.))
116 {
117 Scalar F = sigDamping * v/fabs(v) + velDamping * v;
118 Vector3 force = axis * -F;
119
120 bodyA.applyCentralForce(force);
121 bodyB.applyCentralForce(-force);
122 }
123 }
124}

126std::vector<Renderable> PrismaticJoint::Render()
127{
128 std::vector<Renderable> items(0);
129 Renderable item;
130 item.model = glm::mat4(1.f);
131 item.type = RenderableType::JOINT_LINES;
132
133 btTypedConstraint* slider = getConstraint();
134 Vector3 A = slider->getRigidBodyA().getCenterOfMassPosition();
135 Vector3 B = slider->getRigidBodyB().getCenterOfMassPosition();
136 Vector3 pivot = (A+B)/Scalar(2.);
137 Vector3 axis = (slider->getRigidBodyA().getCenterOfMassTransform().getBasis() * axisInA).normalized();
138
139 //calculate axis ends
140 Scalar e1 = (A-pivot).dot(axis);
141 Scalar e2 = (B-pivot).dot(axis);
142 Vector3 C1 = pivot + e1 * axis;
143 Vector3 C2 = pivot + e2 * axis;
144
145 item.points.push_back(glm::vec3(A.getX(), A.getY(), A.getZ()));
146 item.points.push_back(glm::vec3(C1.getX(), C1.getY(), C1.getZ()));
147 item.points.push_back(glm::vec3(B.getX(), B.getY(), B.getZ()));
148 item.points.push_back(glm::vec3(C2.getX(), C2.getY(), C2.getZ()));
149
150 item.points.push_back(glm::vec3(C1.getX(), C1.getY(), C1.getZ()));
151 item.points.push_back(glm::vec3(C2.getX(), C2.getY(), C2.getZ()));
152
153 items.push_back(item);
154 return items;
155}
