37Propeller::Propeller(std::string uniqueName, std::shared_ptr<SolidEntity> propeller, Scalar diameter, Scalar thrustCoeff, Scalar torqueCoeff, Scalar maxRPM, bool rightHand, bool inverted) : LinkActuator(uniqueName)
38{
39 D = diameter;
40 kT0 = thrustCoeff;
41 kQ0 = torqueCoeff;
42 kp = Scalar(8.0);
43 ki = Scalar(3.0);
44 iLim = Scalar(2.0);
45 RH = rightHand;
46 inv = inverted;
47 omegaLim = maxRPM/Scalar(60) * Scalar(2) * M_PI; //In rad/s
48
49 theta = Scalar(0);
50 omega = Scalar(0);
51 thrust = Scalar(0);
52 torque = Scalar(0);
53 setpoint = Scalar(0);
54 iError = Scalar(0);
55
56 propeller_ = propeller;
57 propeller_->BuildGraphicalObject();
58}

97void Propeller::Update(Scalar dt)
98{
99 Actuator::Update(dt);
100
101 //Update thruster velocity
102 Scalar error = setpoint * omegaLim - omega;
103 Scalar motorTorque = kp * error + ki * iError;
104 omega += (motorTorque - torque)*dt;
105 theta += omega * dt; //Just for animation
106
107 //Integrate error
108 iError += error * dt;
109 iError = iError > iLim ? iLim : (iError < -iLim ? -iLim : iError);
110
111 if(attach != NULL)
112 {
113 //Get transforms
114 Transform solidTrans = attach->getCGTransform();
115 Transform propTrans = attach->getOTransform() * o2a;
116 Vector3 relPos = propTrans.getOrigin() - solidTrans.getOrigin();
117 Vector3 velocity = attach->getLinearVelocityInLocalPoint(relPos);
118
119 Atmosphere* atm = SimulationApp::getApp()->getSimulationManager()->getAtmosphere();
120
121 if(atm->IsInsideFluid(propTrans.getOrigin()))
122 {
123 //Calculate thrust
124 //Thrust coefficient depends on advance ratio J = u/(n*D), kT0 -> J=0
125 //The lower the p/D the more linear the dependence of Kt on J
126 Scalar n = omega/(Scalar(2) * M_PI);
127 Scalar u = -propTrans.getBasis().getColumn(0).dot(atm->GetFluidVelocity(propTrans.getOrigin()) - velocity); //Incoming air velocity
128 Scalar alpha(-0.095/0.8);
129 //kT(J) = kT0 + alpha * J --> approximated with linear function
130 thrust = (RH ? Scalar(1) : Scalar(-1)) * atm->getGas().density * D*D*D * btFabs(n) * (D*kT0*n + alpha*u);
131 torque = -kQ0 * atm->getGas().density * btFabs(n)*n * D*D*D*D*D; //Torque is the loading of propeller due to drag
132
133 //Apply forces and torques
134 Vector3 thrustV(thrust, 0, 0);
135 Vector3 torqueV(torque, 0, 0);
136 attach->ApplyCentralForce(propTrans.getBasis() * thrustV);
137 attach->ApplyTorque((propTrans.getOrigin() - solidTrans.getOrigin()).cross(propTrans.getBasis() * thrustV));
138 attach->ApplyTorque(propTrans.getBasis() * torqueV);
139 }
140 }
141}

143std::vector<Renderable> Propeller::Render()
144{
145 Transform propTrans = Transform::getIdentity();
146 if(attach != nullptr)
147 propTrans = attach->getOTransform() * o2a;
148 else
149 LinkActuator::Render();
150
151 //Rotate propeller
152 propTrans *= Transform(Quaternion(0, 0, theta), Vector3(0,0,0));
153
154 //Add renderable
155 std::vector<Renderable> items(0);
156 Renderable item;
157 item.type = RenderableType::SOLID;
158 item.materialName = propeller_->getMaterial().name;
159 item.objectId = propeller_->getGraphicalObject();
160 item.lookId = dm == DisplayMode::GRAPHICAL ? propeller_->getLook() : -1;
161 item.model = glMatrixFromTransform(propTrans);
162 items.push_back(item);
163
164 item.type = RenderableType::ACTUATOR_LINES;
165 item.points.push_back(glm::vec3(0,0,0));
166 item.points.push_back(glm::vec3(0.1f*thrust,0,0));
167 items.push_back(item);
168
169 return items;
170}
