59Scalar RotaryEncoder::GetRawAngle()
60{
61 if(j != nullptr && j->getType() == JointType::REVOLUTE)
62 {
63 return ((RevoluteJoint*)j)->getAngle();
64 }
65 else if(fe != nullptr)
66 {
67 Scalar mbAngle;
68 btMultibodyLink::eFeatherstoneJointType jt = btMultibodyLink::eInvalid;
69 fe->getJointPosition(jId, mbAngle, jt);
70
71 if(jt == btMultibodyLink::eRevolute)
72 return mbAngle;
73 else
74 return Scalar(0);
75 }
76 else if(motor != nullptr)
77 {
78 return motor->getAngle();
79 }
80 else if(thrust != nullptr)
81 {
82 return thrust->getAngle();
83 }
84 else
85 return Scalar(0);
86}

88Scalar RotaryEncoder::GetRawAngularVelocity()
89{
90 if(j != nullptr && j->getType() == JointType::REVOLUTE)
91 {
92 return ((RevoluteJoint*)j)->getAngularVelocity();
93 }
94 else if(fe != nullptr)
95 {
96 Scalar mbAV;
97 btMultibodyLink::eFeatherstoneJointType jt = btMultibodyLink::eInvalid;
98 fe->getJointVelocity(jId, mbAV, jt);
99
100 if(jt == btMultibodyLink::eRevolute)
101 return mbAV;
102 else
103 return Scalar(0);
104 }
105 else if(motor != nullptr)
106 {
107 return motor->getAngularVelocity();
108 }
109 else if(thrust != nullptr)
110 {
111 return thrust->getOmega();
112 }
113 else
114 return Scalar(0);
115}

117void RotaryEncoder::InternalUpdate(Scalar dt)
118{
119 //new angle
120 Scalar actualAngle = GetRawAngle();
121
122 //accumulate
123 Scalar angle0 = angle;
124
125 if(lastAngle * actualAngle < Scalar(0))
126 {
127 if(lastAngle > M_PI_4)
128 angle += ((actualAngle + 2 * M_PI) - lastAngle);
129 else if(lastAngle < -M_PI_4)
130 angle += (actualAngle - (lastAngle + 2 * M_PI));
131 else
132 angle += (actualAngle - lastAngle);
133 }
134 else
135 angle += (actualAngle - lastAngle);
136
137 lastAngle = actualAngle;
138
139 //angular velocity
140 Scalar angularVelocity = (angle - angle0)/dt; // Less noisy than reading raw velocity
141
142 //record sample
143 Sample s{std::vector<Scalar>({angle, angularVelocity})};
144 AddSampleToHistory(s);
145}
