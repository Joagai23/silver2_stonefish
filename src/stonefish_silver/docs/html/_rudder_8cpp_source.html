81void Rudder::Update(Scalar dt)
82{
83 //Update rudder angle
84 if(maxAngularRate > Scalar(0) && btFabs(setpoint-theta)/dt > maxAngularRate)
85 {
86 Scalar dTheta = setpoint-theta > Scalar(0) ? maxAngularRate * dt : -maxAngularRate * dt;
87 theta += dTheta;
88 }
89 else
90 theta = setpoint;
91
92 if(attach != NULL)
93 {
94 Quaternion rudderRot(theta, 0, 0);
95
96 //Get transforms
97 Transform solidTrans = attach->getCGTransform();
98 // o2a is the transform of the actuator
99 Transform rudderTrans = attach->getOTransform() * o2a * Transform(rudderRot);
100 Vector3 relPos = rudderTrans.getOrigin() - solidTrans.getOrigin();
101
102 Ocean* ocn = SimulationApp::getApp()->getSimulationManager()->getOcean();
103
104 Vector3 absVel = attach->getLinearVelocityInLocalPoint(relPos);
105 Vector3 fluidVel = ocn->GetFluidVelocity(rudderTrans.getOrigin());
106 Vector3 velocity = rudderTrans.getBasis().transpose()*(absVel - fluidVel);
107
108 Scalar angle = atan2(velocity.getY(), velocity.getX());
109
110 // If rudder facing velocity, reverse equations
111 if (fabs(angle) > .5*M_PI)
112 {
113 angle = atan2(-velocity.getY(), -velocity.getX());
114 }
115
116 if(ocn != nullptr && ocn->IsInsideFluid(rudderTrans.getOrigin()) && !velocity.isZero())
117 {
118 // Calculate quadratic approximations for lift and drag
119
120 // Equations from:
121 // Engelhardtsen, Oystein. 3D AUV Collision Avoidance.
122 // MS thesis. Institutt for teknisk kybernetikk, 2007.
123
124 // Fluid density
125 Scalar rho = ocn->getLiquid().density;
126
127 // Speed
128 Scalar u = velocity.length();
129
130 // d = angle
131 Scalar du2 = angle * u*u;
132
133 // Approximated rudder parameter
134 Scalar X = .5 * area * rho * liftCoeff;
135
136 // Equation 3.2.19
137 Scalar lift = du2 * X;
138
139 // Equation 3.2.21
140 Scalar drag = angle * du2 * X;
141
142 // Drag is opposite to velocity
143 dragV = -velocity;
144 dragV.normalize() *= drag;
145
146 // Lift is restricted to XY plane
147 // liftV = (-vy, vx, 0)
148 liftV = Vector3(0., 0., -1.).cross(velocity);
149 liftV.normalize() *= lift;
150
151 // if angle greater than stall -> no lift
152 if (fabs(angle) > stallAngle)
153 {
154 liftV = Vector3(0, 0, 0);
155 }
156
157 // Torque vectors
158 Vector3 dragT = relPos.cross(rudderTrans.getBasis() * dragV);
159 Vector3 liftT = relPos.cross(rudderTrans.getBasis() * liftV);
160
161 // Apply forces and torques
162 attach->ApplyTorque(dragT);
163 attach->ApplyTorque(liftT);
164 attach->ApplyCentralForce(rudderTrans.getBasis() * dragV);
165 attach->ApplyCentralForce(rudderTrans.getBasis() * liftV);
166 }
167 else
168 {
169 dragV = Vector3(0, 0, 0);
170 liftV = Vector3(0, 0, 0);
171 }
172 }
173}

175std::vector<Renderable> Rudder::Render()
176{
177 Transform rudderTrans = Transform::getIdentity();
178 if(attach != NULL)
179 rudderTrans = attach->getOTransform() * o2a;
180 else
181 LinkActuator::Render();
182
183 //Rotate rudder
184 rudderTrans *= Transform(Quaternion(theta, 0, 0)) * rudder->getO2GTransform();
185
186 //Add renderable
187 std::vector<Renderable> items(0);
188 Renderable item;
189 item.type = RenderableType::SOLID;
190 item.materialName = rudder->getMaterial().name;
191 item.objectId = rudder->getGraphicalObject();
192 item.lookId = dm == DisplayMode::GRAPHICAL ? rudder->getLook() : -1;
193 item.model = glMatrixFromTransform(rudderTrans);
194 items.push_back(item);
195
196 item.type = RenderableType::ACTUATOR_LINES;
197 item.points.push_back(glm::vec3(0,0,0));
198 Vector3 VG = .1*(rudder->getO2GTransform().inverse().getBasis()*(liftV + dragV));
199 item.points.push_back(glm::vec3(VG.getX(),VG.getY(),VG.getZ()));
200 items.push_back(item);
201
202 return items;
203}
