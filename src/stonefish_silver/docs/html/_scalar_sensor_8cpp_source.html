106void ScalarSensor::AddSampleToHistory(const Sample& s)
107{
108 if(historyLen < 0 && history.size() > 0) //No history
109 {
110 delete history[0];
111 history.pop_front();
112 }
113 else if(historyLen > 0 && (int)history.size() == historyLen) //Specified history length
114 {
115 delete history[0];
116 history.pop_front();
117 }
118 //else == 0 --> unlimited history
119
120 Sample* sample = new Sample(s, sampleCount);
121 ++sampleCount;
122
123 for(unsigned int i=0; i<sample->getNumOfDimensions(); ++i)
124 {
125 Scalar* data = sample->getDataPointer();
126
127 //Add noise
128 if(channels[i].stdDev > Scalar(0) && data[i] < channels[i].rangeMax && data[i] > channels[i].rangeMin)
129 data[i] += channels[i].noise(randomGenerator);
130
131 //Limit readings
132 if(data[i] > channels[i].rangeMax)
133 data[i] = channels[i].rangeMax;
134 else if(data[i] < channels[i].rangeMin)
135 data[i] = channels[i].rangeMin;
136 }
137
138 //Add to history
139 history.push_back(sample);
140}

150void ScalarSensor::SaveMeasurementsToTextFile(const std::string& path, bool includeTime, unsigned int fixedPrecision)
151{
152 if(history.size() == 0)
153 return;
154
155 cInfo("Saving %s measurements to: %s", getName().c_str(), path.c_str());
156
157 FILE* fp = fopen(path.c_str(), "wt");
158 if(fp == NULL)
159 {
160 cError("File could not be opened!");
161 return;
162 }
163
164 //Write header
165 fprintf(fp, "#Measurements from %s\n", getName().c_str());
166 fprintf(fp, "#Number of channels: %ld\n", channels.size());
167 fprintf(fp, "#Number of samples: %ld\n", history.size());
168 if(freq <= Scalar(0.))
169 fprintf(fp, "#Frequency: %1.3lf Hz\n", SimulationApp::getApp()->getSimulationManager()->getStepsPerSecond());
170 else
171 fprintf(fp, "#Frequency: %1.3lf Hz\n", freq);
172 fprintf(fp, "#Unit system: SI\n\n");
173
174 //Write data header
175 if(includeTime)
176 fprintf(fp, "#Time\t");
177 else
178 fprintf(fp, "#");
179
180 for(unsigned int i = 0; i < channels.size(); i++)
181 {
182 fprintf(fp, "%s", channels[i].name.c_str());
183
184 if(i < channels.size() - 1)
185 fprintf(fp, "\t");
186 else
187 fprintf(fp, "\n");
188 }
189
190 //Write data
191 std::string format = "%1." + std::to_string(fixedPrecision) + "lf";
192
193 for(unsigned int i = 0; i < history.size(); i++)
194 {
195 Sample* s = history[i];
196
197 if(includeTime)
198 {
199 fprintf(fp, format.c_str(), s->getTimestamp());
200 fprintf(fp, "\t");
201 }
202
203 for(unsigned int h = 0; h < channels.size(); h++)
204 {
205 Scalar v = s->getValue(h);
206
207 fprintf(fp, format.c_str(), v);
208
209 if(h < channels.size() - 1)
210 fprintf(fp, "\t");
211 else
212 fprintf(fp, "\n");
213 }
214 }
215
216 fclose(fp);
217}

219void ScalarSensor::SaveMeasurementsToOctaveFile(const std::string& path, bool includeTime, bool separateChannels)
220{
221 if(history.size() == 0)
222 return;
223
224 //build data structure
225 ScientificData data("");
226
227 if(separateChannels) //channels separated and saved in vecotors
228 {
229 if(includeTime) //write time vector
230 {
231 ScientificDataItem* it = new ScientificDataItem();
232 it->name = "Time";
233 it->type = DATA_VECTOR;
234
235 btVectorXu* vector = new btVectorXu((unsigned int)history.size());
236 it->value = vector;
237
238 for(unsigned int i = 0; i < history.size(); ++i)
239 {
240 Sample* s = history[i];
241 (*vector)[i] = s->getTimestamp();
242 }
243
244 data.addItem(it);
245 }
246
247 //write channel vectors
248 for(unsigned int i = 0; i < channels.size(); ++i)
249 {
250 ScientificDataItem* it = new ScientificDataItem();
251 it->name = channels[i].name;
252 it->type = DATA_VECTOR;
253
254 btVectorXu* vector = new btVectorXu((unsigned int)history.size());
255 it->value = vector;
256
257 for(unsigned int h = 0; h < history.size(); ++h)
258 {
259 Sample* s = history[h];
260 Scalar v = s->getValue(i);
261 (*vector)[h] = v;
262 }
263
264 data.addItem(it);
265 }
266 }
267 else //channels combined in a matrix
268 {
269 ScientificDataItem* it = new ScientificDataItem();
270 it->name = getName();
271 it->type = DATA_MATRIX;
272
273 btMatrixXu* matrix = new btMatrixXu((unsigned int)history.size(), (unsigned int)channels.size() + (includeTime ? 1 : 0));
274 it->value = matrix;
275
276 for(unsigned int i = 0; i < history.size(); ++i)
277 {
278 Sample* s = history[i];
279
280 if(includeTime)
281 matrix->setElem(i, 0, s->getTimestamp());
282
283 for(unsigned int h = 0; h < channels.size(); ++h)
284 {
285 Scalar v = s->getValue(h);
286 matrix->setElem(i, h + (includeTime ? 1 : 0), v);
287 }
288 }
289
290 data.addItem(it);
291 }
292
293 //save data structure to file
294 SaveOctaveData(path, data);
295}
