31Scalar ScrewConstraint::getAngularPosition()
32{
33 this->calculateTransforms(m_rbA.getCenterOfMassTransform(), m_rbB.getCenterOfMassTransform());
34 const Vector3 axisA0 = m_calculatedTransformA.getBasis().getColumn(1);
35 const Vector3 axisA1 = m_calculatedTransformA.getBasis().getColumn(2);
36 const Vector3 axisB0 = m_calculatedTransformB.getBasis().getColumn(1);
37 return btAtan2(axisB0.dot(axisA1), axisB0.dot(axisA0));
38}

60void ScrewConstraint::_getInfo2NonVirtual(btConstraintInfo2* info, const Transform& transA, const Transform& transB,
61 const Vector3& linVelA, const Vector3& linVelB, Scalar rbAinvMass, Scalar rbBinvMass)
62{
67
69 const Transform& trA = getCalculatedTransformA();
70 const Transform& trB = getCalculatedTransformB();
71
72 btAssert(!m_useSolveConstraintObsolete);
73 int i, s = info->rowskip;
74
75 Scalar signFact = m_useLinearReferenceFrameA ?
76 Scalar(1.0f) : Scalar(-1.0f);
77
78 // difference between frames in WCS
79 Vector3 ofs = trB.getOrigin() - trA.getOrigin();
80 // now get weight factors depending on masses
81 Scalar miA = rbAinvMass;
82 Scalar miB = rbBinvMass;
83 bool hasStaticBody = (miA < SIMD_EPSILON) || (miB < SIMD_EPSILON);
84 Scalar miS = miA + miB;
85 Scalar factA, factB;
86 if (miS > Scalar(0.f))
87 {
88 factA = miB / miS;
89 }
90 else
91 {
92 factA = Scalar(0.5f);
93 }
94 factB = Scalar(1.0f) - factA;
95 Vector3 ax1, p, q;
96 Vector3 ax1A = trA.getBasis().getColumn(0);
97 Vector3 ax1B = trB.getBasis().getColumn(0);
98 if (m_useOffsetForConstraintFrame)
99 {
100 // get the desired direction of slider axis
101 // as weighted sum of X-orthos of frameA and frameB in WCS
102 ax1 = ax1A * factA + ax1B * factB;
103 ax1.normalize();
104 // construct two orthos to slider axis
105 btPlaneSpace1(ax1, p, q);
106 }
107 else
108 { // old way - use frameA
109 ax1 = trA.getBasis().getColumn(0);
110 // get 2 orthos to slider axis (Y, Z)
111 p = trA.getBasis().getColumn(1);
112 q = trA.getBasis().getColumn(2);
113 }
114 // make rotations around these orthos equal
115 // the slider axis should be the only unconstrained
116 // rotational axis, the angular velocity of the two bodies perpendicular to
117 // the slider axis should be equal. thus the constraint equations are
118 // p*w1 - p*w2 = 0
119 // q*w1 - q*w2 = 0
120 // where p and q are unit vectors normal to the slider axis, and w1 and w2
121 // are the angular velocity vectors of the two bodies.
122 info->m_J1angularAxis[0] = p[0];
123 info->m_J1angularAxis[1] = p[1];
124 info->m_J1angularAxis[2] = p[2];
125 info->m_J1angularAxis[s+0] = q[0];
126 info->m_J1angularAxis[s+1] = q[1];
127 info->m_J1angularAxis[s+2] = q[2];
128
129 info->m_J2angularAxis[0] = -p[0];
130 info->m_J2angularAxis[1] = -p[1];
131 info->m_J2angularAxis[2] = -p[2];
132 info->m_J2angularAxis[s+0] = -q[0];
133 info->m_J2angularAxis[s+1] = -q[1];
134 info->m_J2angularAxis[s+2] = -q[2];
135 // compute the right hand side of the constraint equation. set relative
136 // body velocities along p and q to bring the slider back into alignment.
137 // if ax1A,ax1B are the unit length slider axes as computed from bodyA and
138 // bodyB, we need to rotate both bodies along the axis u = (ax1 x ax2).
139 // if "theta" is the angle between ax1 and ax2, we need an angular velocity
140 // along u to cover angle erp*theta in one step :
141 // |angular_velocity| = angle/time = erp*theta / stepsize
142 // = (erp*fps) * theta
143 // angular_velocity = |angular_velocity| * (ax1 x ax2) / |ax1 x ax2|
144 // = (erp*fps) * theta * (ax1 x ax2) / sin(theta)
145 // ...as ax1 and ax2 are unit length. if theta is smallish,
146 // theta ~= sin(theta), so
147 // angular_velocity = (erp*fps) * (ax1 x ax2)
148 // ax1 x ax2 is in the plane space of ax1, so we project the angular
149 // velocity to p and q to find the right hand side.
150 // Scalar k = info->fps * info->erp * getSoftnessOrthoAng();
151 Scalar currERP = (m_flags & BT_SLIDER_FLAGS_ERP_ORTANG) ?
152 m_softnessOrthoAng : m_softnessOrthoAng * info->erp;
153 Scalar k = info->fps * currERP;
154
155 Vector3 u = ax1A.cross(ax1B);
156 info->m_constraintError[0] = k * u.dot(p);
157 info->m_constraintError[s] = k * u.dot(q);
158 if (m_flags & BT_SLIDER_FLAGS_CFM_ORTANG)
159 {
160 info->cfm[0] = m_cfmOrthoAng;
161 info->cfm[s] = m_cfmOrthoAng;
162 }
163
164 // last filled row
165 int nrow = 1;
166 int srow;
167 Scalar limit_err;
168 int limit;
169 int powered;
170
171 // next two rows.
172 // we want: velA + wA x relA == velB + wB x relB ... but this would
173 // result in three equations, so we project along two orthos to the
174 // slider axis
175
176 Transform bodyA_trans = transA;
177 Transform bodyB_trans = transB;
178 nrow++;
179 int s2 = nrow * s;
180 nrow++;
181 int s3 = nrow * s;
182 Vector3 tmpA(0, 0, 0), tmpB(0, 0, 0), relA(0, 0, 0),
183 relB(0, 0, 0), c(0, 0, 0);
184 if (m_useOffsetForConstraintFrame)
185 {
186 // get vector from bodyB to frameB in WCS
187 relB = trB.getOrigin() - bodyB_trans.getOrigin();
188 // get its projection to slider axis
189 Vector3 projB = ax1 * relB.dot(ax1);
190 // get vector directed from bodyB to slider axis (and orthogonal to it)
191 Vector3 orthoB = relB - projB;
192 // same for bodyA
193 relA = trA.getOrigin() - bodyA_trans.getOrigin();
194 Vector3 projA = ax1 * relA.dot(ax1);
195 Vector3 orthoA = relA - projA;
196 // get desired offset between frames A and B along slider axis
197 Scalar sliderOffs = m_linPos - m_depth[0];
198 // desired vector from projection of center of bodyA to projection of
199 // center of bodyB to slider axis
200 Vector3 totalDist = projA + ax1 * sliderOffs - projB;
201 // get offset vectors relA and relB
202 relA = orthoA + totalDist * factA;
203 relB = orthoB - totalDist * factB;
204 // now choose average ortho to slider axis
205 p = orthoB * factA + orthoA * factB;
206 Scalar len2 = p.length2();
207 if (len2 > SIMD_EPSILON)
208 {
209 p /= btSqrt(len2);
210 }
211 else
212 {
213 p = trA.getBasis().getColumn(1);
214 }
215 // make one more ortho
216 q = ax1.cross(p);
217 // fill two rows
218 tmpA = relA.cross(p);
219 tmpB = relB.cross(p);
220 for (i = 0; i < 3; ++i)
221 {
222 info->m_J1angularAxis[s2+i] = tmpA[i];
223 info->m_J2angularAxis[s2+i] = -tmpB[i];
224 }
225 tmpA = relA.cross(q);
226 tmpB = relB.cross(q);
227 if (hasStaticBody && getSolveAngLimit())
228 { // to make constraint between static and dynamic objects more rigid
229 // remove wA (or wB) from equation if angular limit is hit
230 tmpB *= factB;
231 tmpA *= factA;
232 }
233 for (i = 0; i < 3; ++i)
234 {
235 info->m_J1angularAxis[s3+i] = tmpA[i];
236 info->m_J2angularAxis[s3+i] = -tmpB[i];
237 info->m_J1linearAxis[s2+i] = p[i];
238 info->m_J1linearAxis[s3+i] = q[i];
239 info->m_J2linearAxis[s2+i] = -p[i];
240 info->m_J2linearAxis[s3+i] = -q[i];
241 }
242 }
243 else
244 {
245 // old way - maybe incorrect if bodies are not on the slider axis
246 // see discussion "Bug in slider constraint"
247 // http://bulletphysics.org/Bullet/phpBB3/viewtopic.php?f=9&t=4024&start=0
248 c = bodyB_trans.getOrigin() - bodyA_trans.getOrigin();
249 Vector3 tmp = c.cross(p);
250 for (i = 0; i < 3; ++i)
251 {
252 info->m_J1angularAxis[s2+i] = factA*tmp[i];
253 info->m_J2angularAxis[s2+i] = factB*tmp[i];
254 }
255 tmp = c.cross(q);
256 for (i = 0; i < 3; ++i)
257 {
258 info->m_J1angularAxis[s3+i] = factA*tmp[i];
259 info->m_J2angularAxis[s3+i] = factB*tmp[i];
260 info->m_J1linearAxis[s2+i] = p[i];
261 info->m_J1linearAxis[s3+i] = q[i];
262 info->m_J2linearAxis[s2+i] = -p[i];
263 info->m_J2linearAxis[s3+i] = -q[i];
264 }
265 }
266 // compute two elements of right hand side
267
268 // k = info->fps * info->erp * getSoftnessOrthoLin();
269 currERP = (m_flags & BT_SLIDER_FLAGS_ERP_ORTLIN) ?
270 m_softnessOrthoLin : m_softnessOrthoLin * info->erp;
271 k = info->fps * currERP;
272
273 Scalar rhs = k * p.dot(ofs);
274 info->m_constraintError[s2] = rhs;
275 rhs = k * q.dot(ofs);
276 info->m_constraintError[s3] = rhs;
277 if (m_flags & BT_SLIDER_FLAGS_CFM_ORTLIN)
278 {
279 info->cfm[s2] = m_cfmOrthoLin;
280 info->cfm[s3] = m_cfmOrthoLin;
281 }
282
283 // Screw Constraint (coupled linear and angular motion)
284 {
285 nrow++;
286 srow = nrow * info->rowskip;
287 info->m_J1linearAxis[srow+0] = -ax1[0] * this->threadPitch;
288 info->m_J1linearAxis[srow+1] = -ax1[1] * this->threadPitch;
289 info->m_J1linearAxis[srow+2] = -ax1[2] * this->threadPitch;
290
291 info->m_J1angularAxis[srow+0] = ax1[0];
292 info->m_J1angularAxis[srow+1] = ax1[1];
293 info->m_J1angularAxis[srow+2] = ax1[2];
294
295 info->m_J2linearAxis[srow+0] = ax1[0] * this->threadPitch;
296 info->m_J2linearAxis[srow+1] = ax1[1] * this->threadPitch;
297 info->m_J2linearAxis[srow+2] = ax1[2] * this->threadPitch;
298
299 info->m_J2angularAxis[srow+0] = -ax1[0];
300 info->m_J2angularAxis[srow+1] = -ax1[1];
301 info->m_J2angularAxis[srow+2] = -ax1[2];
302
303 // correction
304 // rhs = k * ax1.dot(ofs); // from hinge constraint
305 Scalar lin_disp = ax1.dot(ofs);
306 Scalar ang_pos = this->getAngularPosition();
307 info->m_constraintError[srow] =
308 -k * (lin_disp * this->threadPitch - ang_pos);
309 info->cfm[srow] = -m_cfmOrthoLin;
310
311 // debug, set cfm to 0
312 // info->cfm[srow] = 0;
313 // debug, set error correction to 0
314 // info->m_constraintError[srow] = 0.0;
315 }
316
317 // notes: below enforces
318 // joint limit
319 // powered joint
320 // rotation along slider axis
321
322 // check linear limits
323 limit_err = Scalar(0.0);
324 limit = 0;
325 if (getSolveLinLimit())
326 {
327 limit_err = getLinDepth() * signFact;
328 limit = (limit_err > Scalar(0.0)) ? 2 : 1;
329 }
330 powered = 0;
331 if (getPoweredLinMotor())
332 {
333 powered = 1;
334 }
335 // if the slider has joint limits or motor, add in the extra row
336 if (limit || powered)
337 {
338 nrow++;
339 srow = nrow * info->rowskip;
340 info->m_J1linearAxis[srow+0] = ax1[0];
341 info->m_J1linearAxis[srow+1] = ax1[1];
342 info->m_J1linearAxis[srow+2] = ax1[2];
343 info->m_J2linearAxis[srow+0] = -ax1[0];
344 info->m_J2linearAxis[srow+1] = -ax1[1];
345 info->m_J2linearAxis[srow+2] = -ax1[2];
346 // linear torque decoupling step:
347 //
348 // we have to be careful that the linear constraint forces (+/- ax1)
349 // applied to the two bodies
350 // do not create a torque couple. in other words, the points that the
351 // constraint force is applied at must lie along the same ax1 axis.
352 // a torque couple will result in limited slider-jointed free
353 // bodies from gaining angular momentum.
354 if (m_useOffsetForConstraintFrame)
355 {
356 // this is needed only when bodyA and bodyB are both dynamic.
357 if (!hasStaticBody)
358 {
359 tmpA = relA.cross(ax1);
360 tmpB = relB.cross(ax1);
361 info->m_J1angularAxis[srow+0] = tmpA[0];
362 info->m_J1angularAxis[srow+1] = tmpA[1];
363 info->m_J1angularAxis[srow+2] = tmpA[2];
364 info->m_J2angularAxis[srow+0] = -tmpB[0];
365 info->m_J2angularAxis[srow+1] = -tmpB[1];
366 info->m_J2angularAxis[srow+2] = -tmpB[2];
367 }
368 }
369 else
370 { // The old way. May be incorrect if bodies are not on the slider axis
371 // Linear Torque Decoupling vector (a torque)
372 Vector3 ltd;
373 ltd = c.cross(ax1);
374 info->m_J1angularAxis[srow+0] = factA*ltd[0];
375 info->m_J1angularAxis[srow+1] = factA*ltd[1];
376 info->m_J1angularAxis[srow+2] = factA*ltd[2];
377 info->m_J2angularAxis[srow+0] = factB*ltd[0];
378 info->m_J2angularAxis[srow+1] = factB*ltd[1];
379 info->m_J2angularAxis[srow+2] = factB*ltd[2];
380 }
381 // right-hand part
382 Scalar lostop = getLowerLinLimit();
383 Scalar histop = getUpperLinLimit();
384
385 // issue #1104:
386 // if (limit && (lostop == histop)) raises warnings, using
387 // a warning-less implementation.
388 if (limit && (std::fabs(lostop - histop) <= static_cast<Scalar>(std::numeric_limits<double>::epsilon())))
389 {
390 // the joint motor is ineffective
391 powered = 0;
392 }
393 info->m_constraintError[srow] = 0.;
394 info->m_lowerLimit[srow] = 0.;
395 info->m_upperLimit[srow] = 0.;
396 currERP = (m_flags & BT_SLIDER_FLAGS_ERP_LIMLIN) ?
397 m_softnessLimLin : info->erp;
398 if (powered)
399 {
400 if (m_flags & BT_SLIDER_FLAGS_CFM_DIRLIN)
401 {
402 info->cfm[srow] = m_cfmDirLin;
403 }
404 Scalar tag_vel = getTargetLinMotorVelocity();
405 Scalar mot_fact = getMotorFactor(m_linPos, m_lowerLinLimit,
406 m_upperLinLimit, tag_vel, info->fps * currERP);
407 info->m_constraintError[srow] -=
408 signFact * mot_fact * getTargetLinMotorVelocity();
409 info->m_lowerLimit[srow] += -getMaxLinMotorForce() * info->fps;
410 info->m_upperLimit[srow] += getMaxLinMotorForce() * info->fps;
411 }
412 if (limit)
413 {
414 k = info->fps * currERP;
415 info->m_constraintError[srow] += k * limit_err;
416 if (m_flags & BT_SLIDER_FLAGS_CFM_LIMLIN)
417 {
418 info->cfm[srow] = m_cfmLimLin;
419 }
420 // issue #1104:
421 // if (lostop == histop) raises warnings, using
422 // a warning-less implementation.
423 if (std::fabs(lostop - histop) <= static_cast<Scalar>(std::numeric_limits<double>::epsilon()))
424 {
425 // limited low and high simultaneously
426 info->m_lowerLimit[srow] = -SIMD_INFINITY;
427 info->m_upperLimit[srow] = SIMD_INFINITY;
428 }
429 else if (limit == 1)
430 {
431 // low limit
432 info->m_lowerLimit[srow] = -SIMD_INFINITY;
433 info->m_upperLimit[srow] = 0;
434 }
435 else
436 { // high limit
437 info->m_lowerLimit[srow] = 0;
438 info->m_upperLimit[srow] = SIMD_INFINITY;
439 }
440 // bounce (we'll use slider parameter abs(1.0 - m_dampingLimLin)
441 // for that)
442 Scalar bounce = btFabs(Scalar(1.0) - getDampingLimLin());
443 if (bounce > Scalar(0.0))
444 {
445 Scalar vel = linVelA.dot(ax1);
446 vel -= linVelB.dot(ax1);
447 vel *= signFact;
448 // only apply bounce if the velocity is incoming, and if the
449 // resulting c[] exceeds what we already have.
450 if (limit == 1)
451 { // low limit
452 if (vel < 0)
453 {
454 Scalar newc = -bounce * vel;
455 if (newc > info->m_constraintError[srow])
456 {
457 info->m_constraintError[srow] = newc;
458 }
459 }
460 }
461 else
462 { // high limit - all those computations are reversed
463 if (vel > 0)
464 {
465 Scalar newc = -bounce * vel;
466 if (newc < info->m_constraintError[srow])
467 {
468 info->m_constraintError[srow] = newc;
469 }
470 }
471 }
472 }
473 info->m_constraintError[srow] *= getSoftnessLimLin();
474 }
475 // line above is the end of if (limit)
476 }
477 // line above is the end of if linear limit
478
479 // printf("tp: %f\n", this->threadPitch);
480
481 // check angular limits
482 limit_err = Scalar(0.0);
483 limit = 0;
484 if (getSolveAngLimit())
485 {
486 limit_err = getAngDepth();
487 limit = (limit_err > Scalar(0.0)) ? 1 : 2;
488 }
489 // if the slider has joint limits, add in the extra row
490 powered = 0;
491 if (getPoweredAngMotor())
492 {
493 powered = 1;
494 }
495 if (limit || powered)
496 {
497 nrow++;
498 srow = nrow * info->rowskip;
499 info->m_J1angularAxis[srow+0] = ax1[0];
500 info->m_J1angularAxis[srow+1] = ax1[1];
501 info->m_J1angularAxis[srow+2] = ax1[2];
502
503 info->m_J2angularAxis[srow+0] = -ax1[0];
504 info->m_J2angularAxis[srow+1] = -ax1[1];
505 info->m_J2angularAxis[srow+2] = -ax1[2];
506
507 Scalar lostop = getLowerAngLimit();
508 Scalar histop = getUpperAngLimit();
509 // issue #1104:
510 // if (limit && (lostop == histop)) raises warnings, using
511 // a warning-less implementation.
512 if (limit && (std::fabs(lostop - histop) <= static_cast<Scalar>(std::numeric_limits<double>::epsilon())))
513 { // the joint motor is ineffective
514 powered = 0;
515 }
516 currERP = (m_flags & BT_SLIDER_FLAGS_ERP_LIMANG) ?
517 m_softnessLimAng : info->erp;
518 if (powered)
519 {
520 if (m_flags & BT_SLIDER_FLAGS_CFM_DIRANG)
521 {
522 info->cfm[srow] = m_cfmDirAng;
523 }
524 Scalar mot_fact = getMotorFactor(m_angPos, m_lowerAngLimit,
525 m_upperAngLimit, getTargetAngMotorVelocity(), info->fps * currERP);
526 info->m_constraintError[srow] = mot_fact * getTargetAngMotorVelocity();
527 info->m_lowerLimit[srow] = -getMaxAngMotorForce() * info->fps;
528 info->m_upperLimit[srow] = getMaxAngMotorForce() * info->fps;
529 }
530 if (limit)
531 {
532 k = info->fps * currERP;
533 info->m_constraintError[srow] += k * limit_err;
534 if (m_flags & BT_SLIDER_FLAGS_CFM_LIMANG)
535 {
536 info->cfm[srow] = m_cfmLimAng;
537 }
538 // issue #1104:
539 // if (lostop == histop) raises warnings, using
540 // a warning-less implementation.
541 if (std::fabs(lostop - histop) <= static_cast<Scalar>(std::numeric_limits<double>::epsilon()))
542 {
543 // limited low and high simultaneously
544 info->m_lowerLimit[srow] = -SIMD_INFINITY;
545 info->m_upperLimit[srow] = SIMD_INFINITY;
546 }
547 else if (limit == 1)
548 { // low limit
549 info->m_lowerLimit[srow] = 0;
550 info->m_upperLimit[srow] = SIMD_INFINITY;
551 }
552 else
553 { // high limit
554 info->m_lowerLimit[srow] = -SIMD_INFINITY;
555 info->m_upperLimit[srow] = 0;
556 }
557 // bounce (we'll use slider parameter abs(1.0 - m_dampingLimAng)
558 // for that)
559 Scalar bounce = btFabs(Scalar(1.0) - getDampingLimAng());
560 if (bounce > Scalar(0.0))
561 {
562 Scalar vel = m_rbA.getAngularVelocity().dot(ax1);
563 vel -= m_rbB.getAngularVelocity().dot(ax1);
564 // only apply bounce if the velocity is incoming, and if the
565 // resulting c[] exceeds what we already have.
566 if (limit == 1)
567 { // low limit
568 if (vel < 0)
569 {
570 Scalar newc = -bounce * vel;
571 if (newc > info->m_constraintError[srow])
572 {
573 info->m_constraintError[srow] = newc;
574 }
575 }
576 }
577 else
578 { // high limit - all those computations are reversed
579 if (vel > 0)
580 {
581 Scalar newc = -bounce * vel;
582 if (newc < info->m_constraintError[srow])
583 {
584 info->m_constraintError[srow] = newc;
585 }
586 }
587 }
588 }
589 info->m_constraintError[srow] *= getSoftnessLimAng();
590 }
591 // line above is the end of if (limit)
592 }
593 // line above is the end of if angular limit or powered
594}

596void ScrewConstraint::_getInfo1NonVirtual(btConstraintInfo1* info)
597{
604
605 // info->m_numConstraintRows = 6;
606 // Fixed 2 linear + 2 angular + 1 limit (even if not used)
607 // info->nub = 0;
608
609 if (m_useSolveConstraintObsolete)
610 {
611 info->m_numConstraintRows = 0;
612 info->nub = 0;
613 }
614 else
615 {
616 // Fixed 2 linear + 2 angular
617 info->m_numConstraintRows = 4;
618 info->nub = 2;
619
620 // Add constraint for screw motion, coupling linear and angular motion.
621 info->m_numConstraintRows++;
622 info->nub++;
623
624 // prepare constraint
625 calculateTransforms(
626 m_rbA.getCenterOfMassTransform(), m_rbB.getCenterOfMassTransform());
627 testLinLimits();
628 if (getSolveLinLimit() || getPoweredLinMotor())
629 {
630 // limit 3rd linear as well
631 info->m_numConstraintRows++;
