105void Sensor::setVisual(const std::string& meshFilename, Scalar scale, const std::string& look)
106{
107 if(!SimulationApp::getApp()->hasGraphics())
108 return;
109
110 Mesh* mesh = OpenGLContent::LoadMesh(meshFilename, scale, false);
111 if(mesh == nullptr)
112 return;
113
114 graObjectId = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->BuildObject(mesh);
115 lookId = ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->getLookId(look);
116 delete mesh;
117}

125void Sensor::Update(Scalar dt)
126{
127 if(!enabled)
128 return;
129
130 SDL_LockMutex(updateMutex);
131
132 if(freq <= Scalar(0)) // Every simulation tick
133 {
134 InternalUpdate(dt);
135 newDataAvailable = true;
136 }
137 else //Fixed rate
138 {
139 eleapsedTime += dt;
140 Scalar invFreq = Scalar(1)/freq;
141
142 if(eleapsedTime >= invFreq)
143 {
144 InternalUpdate(invFreq);
145 eleapsedTime -= invFreq;
146 newDataAvailable = true;
147 }
148 }
149
150 SDL_UnlockMutex(updateMutex);
151}

153std::vector<Renderable> Sensor::Render()
154{
155 std::vector<Renderable> items(0);
156 if(renderable && graObjectId > 0)
157 {
158 Renderable item;
159 item.type = RenderableType::SOLID;
160 item.materialName = "";
161 item.objectId = graObjectId;
162 item.lookId = lookId;
163 item.model = glMatrixFromTransform(getSensorFrame());
164 items.push_back(item);
165 }
166 return items;
167}
