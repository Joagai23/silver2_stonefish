96void SimulationApp::Run(bool autostart, bool autostep, Scalar timeStep)
97{
98 autostep_ = autostep;
99 timeStep_ = timeStep < Scalar(0) ? Scalar(0) : timeStep;
100
101 Init();
102 if(autostart) StartSimulation();
103 Loop();
104 CleanUp();
105}

107void SimulationApp::Loop()
108{
109 startTime_ = GetTimeInMicroseconds();
110 while(state_ != SimulationState::FINISHED)
111 LoopInternal();
112}

133void SimulationApp::StepSimulation()
134{
135 if (timeStep_ == Scalar(0)) // Real time simulation
136 {
137 simManager_->AdvanceSimulation();
138 }
139 else // Fixed step simulation
140 {
141 simManager_->StepSimulation(timeStep_);
142 simManager_->SimulationStepCompleted(timeStep_);
143 }
144}
