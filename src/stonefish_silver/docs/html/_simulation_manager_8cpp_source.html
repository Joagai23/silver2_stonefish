310int SimulationManager::CheckCollision(const Entity *entA, const Entity *entB)
311{
312 for(size_t i = 0; i < collisions.size(); ++i)
313 {
314 if((collisions[i].A == entA && collisions[i].B == entB)
315 || (collisions[i].B == entA && collisions[i].A == entB))
316 return (int)i;
317 }
318
319 return -1;
320}
321
322void SimulationManager::EnableCollision(const Entity* entA, const Entity* entB)
323{
324 int colId = CheckCollision(entA, entB);
325
326 if(collisionFilter == CollisionFilteringType::COLLISION_INCLUSIVE && colId == -1)
327 {
328 Collision c;
329 c.A = const_cast<Entity*>(entA);
330 c.B = const_cast<Entity*>(entB);
331 collisions.push_back(c);
332 }
333 else if(collisionFilter == CollisionFilteringType::COLLISION_EXCLUSIVE && colId > -1)
334 {
335 collisions.erase(collisions.begin() + colId);
336 }
337}
338
339void SimulationManager::DisableCollision(const Entity* entA, const Entity* entB)
340{
341 int colId = CheckCollision(entA, entB);
342 if(collisionFilter == CollisionFilteringType::COLLISION_EXCLUSIVE && colId == -1)
343 {
344 Collision c;
345 c.A = const_cast<Entity*>(entA);
346 c.B = const_cast<Entity*>(entB);
347 collisions.push_back(c);
348 cInfo("Disabling collisions between '%s' and '%s'.", entA->getName().c_str(), entB->getName().c_str());
349 }
350 else if(collisionFilter == CollisionFilteringType::COLLISION_INCLUSIVE && colId > -1)
351 {
352 collisions.erase(collisions.begin() + colId);
353 cInfo("Disabling collisions between '%s' and '%s'.", entA->getName().c_str(), entB->getName().c_str());
354 }
355}

635void SimulationManager::getWorldAABB(Vector3& min, Vector3& max)
636{
637 min.setValue(BT_LARGE_FLOAT, BT_LARGE_FLOAT, BT_LARGE_FLOAT);
638 max.setValue(-BT_LARGE_FLOAT, -BT_LARGE_FLOAT, -BT_LARGE_FLOAT);
639
640 for(unsigned int i = 0; i < entities.size(); i++)
641 {
642 Vector3 entAabbMin, entAabbMax;
643 entities[i]->getAABB(entAabbMin, entAabbMax);
644 if(entAabbMin.x() < min.x()) min.setX(entAabbMin.x());
645 if(entAabbMin.y() < min.y()) min.setY(entAabbMin.y());
646 if(entAabbMin.z() < min.z()) min.setZ(entAabbMin.z());
647 if(entAabbMax.x() > max.x()) max.setX(entAabbMax.x());
648 if(entAabbMax.y() > max.y()) max.setY(entAabbMax.y());
649 if(entAabbMax.z() > max.z()) max.setZ(entAabbMax.z());
650 }

737void SimulationManager::InitializeSolver()
738{
739 dwBroadphase = new btDbvtBroadphase(); //btAxisSweep3(Vector3(-50000.0, -50000.0, -10000.0), Vector3(50000.0, 50000.0, 10000.0));
740 dwCollisionConfig = new btSoftBodyRigidBodyCollisionConfiguration();
741
742 //Choose collision dispatcher
743 switch(collisionFilter)
744 {
745 case CollisionFilteringType::COLLISION_INCLUSIVE:
746 dwDispatcher = new FilteredCollisionDispatcher(dwCollisionConfig, true);
747 break;
748
749 case CollisionFilteringType::COLLISION_EXCLUSIVE:
750 dwDispatcher = new FilteredCollisionDispatcher(dwCollisionConfig, false);
751 break;
752 }
753 //dwDispatcher = new btCollisionDispatcher(dwCollisionConfig);
754
755 //Choose constraint solver
756 if(solver == SolverType::SOLVER_SI)
757 {
758 mbSolver = new btMultiBodyConstraintSolver();
759 }
760 else
761 {
762 btMLCPSolverInterface* mlcp;
763
764 switch(solver)
765 {
766 default:
767 case SolverType::SOLVER_DANTZIG:
768 mlcp = new btDantzigSolver();
769 break;
770
771 case SolverType::SOLVER_PGS:
772 mlcp = new btSolveProjectedGaussSeidel();
773 break;
774
775 case SolverType::SOLVER_LEMKE:
776 mlcp = new btLemkeSolver();
777 //((btLemkeSolver*)mlcp)->m_maxLoops = 10000;
778 break;
779 }
780
781 mbSolver = new btMultiBodyMLCPConstraintSolver(mlcp); //ResearchConstraintSolver(mlcp);
782 }
783
784 sbSolver = new btDefaultSoftBodySolver();
785
786 //Create dynamics world
787 dynamicsWorld = new btSoftMultiBodyDynamicsWorld(dwDispatcher, dwBroadphase, mbSolver, dwCollisionConfig, sbSolver);
788
789 //Basic configuration
790 dynamicsWorld->getSolverInfo().m_solverMode = SOLVER_USE_WARMSTARTING | SOLVER_SIMD | SOLVER_USE_2_FRICTION_DIRECTIONS; //SOLVER_RANDMIZE_ORDER | SOLVER_ENABLE_FRICTION_DIRECTION_CACHING;
791 dynamicsWorld->getSolverInfo().m_warmstartingFactor = Scalar(1.);
792 dynamicsWorld->getSolverInfo().m_minimumSolverBatchSize = 256;
793 dynamicsWorld->getSolverInfo().m_timeStep = Scalar(1)/getStepsPerSecond();
794
795 //Quality/stability
796 dynamicsWorld->getSolverInfo().m_tau = Scalar(1.); //mass factor

1008bool SimulationManager::SolveICProblem()
1009{
1010 //Solve for joint positions
1011 icProblemSolved = false;
1012
1013 //Should use gravity?
1014 if(icUseGravity)
1015 dynamicsWorld->setGravity(Vector3(0,0,g));
1016 else
1017 dynamicsWorld->setGravity(Vector3(0,0,0));
1018
1019 //Set IC callback
1020 dynamicsWorld->setInternalTickCallback(SolveICTickCallback, this, true); //Pre-tick
1021 dynamicsWorld->setInternalTickCallback(nullptr, this, false); //Post-tick
1022
1023 uint64_t icTime = GetTimeInMicroseconds();
1024 unsigned int iterations = 0;
1025
1026 do
1027 {
1028 if(iterations > icMaxIter) //Check iterations limit
1029 {
1030 cError("IC problem not solved! Reached maximum interation count.");
1031 return false;
1032 }
1033 else if((GetTimeInMicroseconds() - icTime)/(double)1e6 > icMaxTime) //Check time limit
1034 {
1035 cError("IC problem not solved! Reached maximum time.");
1036 return false;
1037 }
1038
1039 //Simulate world
1040 dynamicsWorld->stepSimulation(icTimeStep, 1, icTimeStep);
1041 iterations++;
1042 }
1043 while(!icProblemSolved);
1044
1045 double solveTime = (GetTimeInMicroseconds() - icTime)/(double)1e6;
1046
1047 //Synchronize body transforms
1048 dynamicsWorld->synchronizeMotionStates();
1049 simulationTime = Scalar(0.);
1050
1051 //Solving time
1052 cInfo("IC problem solved with %d iterations in %1.6lf s.", iterations, solveTime);
1053
1054 //Set gravity
1055 dynamicsWorld->setGravity(Vector3(0,0,g));
1056
1057 //Set simulation tick
1058 dynamicsWorld->setInternalTickCallback(SimulationTickCallback, this, true); //Pre-tick
1059 dynamicsWorld->setInternalTickCallback(SimulationPostTickCallback, this, false); //Post-tick
1060 return true;
1061}

1063void SimulationManager::AdvanceSimulation()
1064{
1065 //Check if initial conditions solved
1066 if(!icProblemSolved)
1067 return;
1068
1069 //Calculate eleapsed time
1070 uint64_t deltaTime;
1071
1072 if(currentTime == 0) //Start of simulation
1073 {
1074 deltaTime = 0.0;
1075 simulationTime = 0.0;
1076 currentTime = getSimulationClock();
1077 timeOffset = currentTime;
1078 return;
1079 }
1080
1081 uint64_t timeInMicroseconds = getSimulationClock(); //Realtime factor included in clock
1082 deltaTime = timeInMicroseconds - currentTime;
1083 currentTime = timeInMicroseconds;
1084
1085 if(deltaTime < ssus) //Sleep if clock did not tick one simulation step
1086 {
1087 SimulationClockSleep(ssus - deltaTime);
1088 timeInMicroseconds = getSimulationClock();
1089 deltaTime += timeInMicroseconds - currentTime;
1090 currentTime = timeInMicroseconds;
1091 }
1092
1093 StepSimulation((Scalar)deltaTime/Scalar(1000000.0));
1094
1095 SDL_LockMutex(simInfoMutex);
1096 Scalar cpuUsageNow = (Scalar)perfMon.getPhysicsTime()/(Scalar)deltaTime * Scalar(100);
1097 Scalar filter(0.001);
1098 cpuUsage = filter * cpuUsageNow + (Scalar(1)-filter) * cpuUsage;
1099 SDL_UnlockMutex(simInfoMutex);
1100}

1235bool SimulationManager::CustomMaterialCombinerCallback(btManifoldPoint& cp, const btCollisionObjectWrapper* colObj0Wrap,int partId0,int index0,const btCollisionObjectWrapper* colObj1Wrap,int partId1,int index1)
1236{
1237 //Retrieve entities associated with colliding objects
1238 Entity* ent0 = (Entity*)colObj0Wrap->getCollisionObject()->getUserPointer();
1239 Entity* ent1 = (Entity*)colObj1Wrap->getCollisionObject()->getUserPointer();
1240
1241 //Check if entities are real
1242 if(ent0 == nullptr || ent1 == nullptr)
1243 {
1244 cp.m_combinedFriction = Scalar(0.);
1245 cp.m_combinedRollingFriction = Scalar(0.);
1246 cp.m_combinedRestitution = Scalar(0.);
1247 return true;
1248 }
1249
1250 //Get material and contact velocity information
1251 MaterialManager* mm = SimulationApp::getApp()->getSimulationManager()->getMaterialManager();
1252
1253 Material mat0;
1254 Vector3 contactVelocity0;
1255 Scalar contactAngularVelocity0;
1256
1257 if(ent0->getType() == EntityType::STATIC)
1258 {
1259 StaticEntity* sent0 = (StaticEntity*)ent0;
1260 mat0 = sent0->getMaterial();
1261 contactVelocity0.setZero();
1262 contactAngularVelocity0 = Scalar(0);
1263 }
1264 else if(ent0->getType() == EntityType::SOLID)
1265 {
1266 SolidEntity* sent0 = (SolidEntity*)ent0;
1267 if(sent0->getSolidType() == SolidType::COMPOUND)
1268 mat0 = ((Compound*)sent0)->getMaterial(((Compound*)sent0)->getPartId(index0));
1269 else
1270 mat0 = sent0->getMaterial();
1271 //Vector3 localPoint0 = sent0->getTransform().getBasis() * cp.m_localPointA;
1272 Vector3 localPoint0 = sent0->getCGTransform().inverse() * cp.getPositionWorldOnA();
1273 contactVelocity0 = sent0->getLinearVelocityInLocalPoint(localPoint0);
1274 contactAngularVelocity0 = sent0->getAngularVelocity().dot(-cp.m_normalWorldOnB);
1275 }
1276 else
1277 {
1278 cp.m_combinedFriction = Scalar(0);
1279 cp.m_combinedRollingFriction = Scalar(0);
1280 cp.m_combinedRestitution = Scalar(0);
1281 return true;
1282 }
1283
1284 Material mat1;
1285 Vector3 contactVelocity1;
1286 Scalar contactAngularVelocity1;
1287
1288 if(ent1->getType() == EntityType::STATIC)
1289 {
1290 StaticEntity* sent1 = (StaticEntity*)ent1;
1291 mat1 = sent1->getMaterial();
1292 contactVelocity1.setZero();
1293 contactAngularVelocity1 = Scalar(0);
1294 }
1295 else if(ent1->getType() == EntityType::SOLID)
1296 {
1297 SolidEntity* sent1 = (SolidEntity*)ent1;
1298 if(sent1->getSolidType() == SolidType::COMPOUND)
1299 mat1 = ((Compound*)sent1)->getMaterial(((Compound*)sent1)->getPartId(index1));
1300 else
1301 mat1 = sent1->getMaterial();
1302 //Vector3 localPoint1 = sent1->getTransform().getBasis() * cp.m_localPointB;
1303 Vector3 localPoint1 = sent1->getCGTransform().inverse() * cp.getPositionWorldOnB();
1304 contactVelocity1 = sent1->getLinearVelocityInLocalPoint(localPoint1);
1305 contactAngularVelocity1 = sent1->getAngularVelocity().dot(cp.m_normalWorldOnB);
1306 }
1307 else
1308 {
1309 cp.m_combinedFriction = Scalar(0);
1310 cp.m_combinedRollingFriction = Scalar(0);
1311 cp.m_combinedRestitution = Scalar(0);
1312 return true;
1313 }
1314
1315 //Calculate contact forces
1316 //A. Stribeck friction model
1317 Vector3 relLocalVel = contactVelocity1 - contactVelocity0;
1318 Vector3 normalVel = cp.m_normalWorldOnB * cp.m_normalWorldOnB.dot(relLocalVel);
1319 Vector3 slipVel = relLocalVel - normalVel;
1320 Scalar sigma = 1000;
1321 // f = (static - dynamic)/(sigma * v^2 + 1) + dynamic
1322 Friction f = mm->GetMaterialsInteraction(mat0.name, mat1.name);
1323 cp.m_combinedFriction = (f.fStatic - f.fDynamic)/(sigma * slipVel.length2() + Scalar(1)) + f.fDynamic;
1324
1325 //Rolling friction not possible to generalize - needs special treatment
1326 cp.m_combinedRollingFriction = Scalar(0);
1327 cp.m_combinedSpinningFriction = Scalar(0);
1328
1329 //Save user data
1330 ContactInfo* cInfo = new ContactInfo();
1331 cInfo->totalAppliedImpulse = Scalar(0);
1332 cInfo->slip = slipVel;
1333 cp.m_userPersistentData = cInfo;
1334
1335 //Damping angular velocity around contact normal (reduce spinning)
1336 //calculate relative angular velocity
1337 Scalar relAngularVelocity01 = contactAngularVelocity0 - contactAngularVelocity1;
1338 Scalar relAngularVelocity10 = contactAngularVelocity1 - contactAngularVelocity0;
1339
1340 //calculate contact normal force and friction torque
1341 Scalar normalForce = cp.m_appliedImpulse * SimulationApp::getApp()->getSimulationManager()->getStepsPerSecond();
1342 Scalar T = cp.m_combinedFriction * normalForce * 0.002;
1343
1344 //apply damping torque
1345 if(ent0->getType() == EntityType::SOLID && !btFuzzyZero(relAngularVelocity01))
1346 ((SolidEntity*)ent0)->ApplyTorque(cp.m_normalWorldOnB * relAngularVelocity01/btFabs(relAngularVelocity01) * T);
1347
1348 if(ent1->getType() == EntityType::SOLID && !btFuzzyZero(relAngularVelocity10))
1349 ((SolidEntity*)ent1)->ApplyTorque(cp.m_normalWorldOnB * relAngularVelocity10/btFabs(relAngularVelocity10) * T);
1350
1351 //Restitution
1352 cp.m_combinedRestitution = mat0.restitution * mat1.restitution;
1353
1354 //B. Magnetic attraction (only between magnet and ferromagnetic body, no magnet-magnet support)
1355 if((mat0.magnetic < Scalar(0) && mat1.magnetic > Scalar(0))
1356 || (mat0.magnetic > Scalar(0) && mat1.magnetic < Scalar(0)))
1357 {
1358 Scalar d = btClamped(cp.getDistance(), Scalar(0.0001), BT_LARGE_FLOAT);
1359 Scalar mag = (btFabs(mat0.magnetic) * btFabs(mat1.magnetic))/(d*d)/Scalar(1e4);
1360 btClamp(mag, Scalar(0), Scalar(10000)); //Arbitrary limit of 10kN
1361 Vector3 mForce = cp.m_normalWorldOnB * mag;
1362
1363 if(ent0->getType() == EntityType::SOLID)
1364 {
1365 SolidEntity* sent0 = (SolidEntity*)ent0;
1366 sent0->ApplyCentralForce(-mForce);
1367 sent0->ApplyTorque((cp.m_positionWorldOnA - sent0->getCGTransform().getOrigin()).cross(-mForce));
1368 }
1369 if(ent1->getType() == EntityType::SOLID)
1370 {
1371 SolidEntity* sent1 = (SolidEntity*)ent1;
1372 sent1->ApplyCentralForce(mForce);
1373 sent1->ApplyTorque((cp.m_positionWorldOnB - sent1->getCGTransform().getOrigin()).cross(mForce));
1374 }
1375
1376 cp.m_combinedRestitution = Scalar(0); //Allows sticking of bodies together
1377 }
1378
1379 return true;
1380}

1382void SimulationManager::SolveICTickCallback(btDynamicsWorld* world, Scalar timeStep)
1383{
1384 SimulationManager* simManager = (SimulationManager*)world->getWorldUserInfo();
1385 btMultiBodyDynamicsWorld* researchWorld = (btMultiBodyDynamicsWorld*)world;
1386
1387 //Clear all forces to ensure that no summing occurs
1388 researchWorld->clearForces(); //Includes clearing of multibody forces!
1389
1390 //Solve for objects settling
1391 bool objectsSettled = true;
1392
1393 if(simManager->icUseGravity)
1394 {
1395 //Apply gravity to bodies
1396 for(size_t i = 0; i < simManager->entities.size(); ++i)
1397 {
1398 if(simManager->entities[i]->getType() == EntityType::SOLID)
1399 {
1400 SolidEntity* solid = (SolidEntity*)simManager->entities[i];
1401 solid->ApplyGravity(world->getGravity());
1402 }
1403 else if(simManager->entities[i]->getType() == EntityType::FEATHERSTONE)
1404 {
1405 FeatherstoneEntity* feather = (FeatherstoneEntity*)simManager->entities[i];
1406 feather->ApplyGravity(world->getGravity());
1407 }
1408 }
1409
1410 if(simManager->simulationTime < Scalar(0.01)) //Wait for a few cycles to ensure bodies started moving
1411 objectsSettled = false;
1412 else
1413 {

1416 {

1451 objectsSettled = false;
1452 break;
1453
1454 case btMultibodyLink::ePrismatic:
1455 if(Vector3(jVelocity,0,0).length() > simManager->icLinTolerance * Scalar(100.))
1456 objectsSettled = false;
1457 break;

1459 default:
1460 break;
1461 }
1462
1463 if(!objectsSettled)
1464 break;
1465 }
1466 }
1467 }
1468 }

1473

1476 jointsICSolved = false;
1477
1478 //Check if everything solved
1479 if(objectsSettled && jointsICSolved)
1480 simManager->icProblemSolved = true;
1481
1482 //Update time
1483 simManager->simulationTime += timeStep;

1623 //Update motion data

1625 {
