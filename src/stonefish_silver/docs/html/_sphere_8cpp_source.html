33Sphere::Sphere(std::string uniqueName, BodyPhysicsSettings phy, Scalar radius, const Transform& origin, std::string material, std::string look, Scalar thickness)
34 : SolidEntity(uniqueName, phy, material, look, thickness)
35{
36 r = radius;
37 T_O2G = T_O2C = T_O2H = origin;
38 T_CG2O = origin.inverse();
39 T_CG2C = T_CG2G = I4();
40 P_CB = Vector3(0,0,0);
41
42 //Calculate physical properties
43 if(thick > Scalar(0) && thick/Scalar(2) < r)
44 {
45 Scalar r1 = r - thick/Scalar(2);
46 Scalar r2 = r + thick/Scalar(2);
47 volume = Scalar(4)/Scalar(3)*M_PI*(r2*r2*r2 - r1*r1*r1);
48 surface = Scalar(4)*M_PI*r2*r2;
49 mass = volume * mat.density;
50 Scalar I = Scalar(2)/Scalar(5)*mass*((r2*r2*r2*r2*r2 - r1*r1*r1*r1*r1)/(r2*r2*r2 - r1*r1*r1));
51 Ipri = Vector3(I,I,I);
52 }
53 else
54 {
55 volume = Scalar(4)/Scalar(3)*M_PI*r*r*r;
56 surface = Scalar(4)*M_PI*r*r;
57 mass = volume * mat.density;
58 Scalar I = Scalar(2)/Scalar(5)*mass*r*r;
59 Ipri = Vector3(I,I,I);
60 }
61
62 //Build geometry
63 phyMesh = OpenGLContent::BuildSphere((GLfloat)r);
64
65 //Compute hydrodynamic properties
66 ComputeFluidDynamicsApprox( GeometryApproxType::SPHERE);
67 //dragCoeff = Vector3(Scalar(0.47)*M_PI*radius*radius, Scalar(0.47)*M_PI*radius*radius, Scalar(0.47)*M_PI*radius*radius);
68}
