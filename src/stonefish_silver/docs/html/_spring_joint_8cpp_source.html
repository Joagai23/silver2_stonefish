35SpringJoint::SpringJoint(std::string uniqueName, SolidEntity* solid, const Transform& attachment,
36 const Vector3& linearStiffness, const Vector3& angularStiffness,
37 const Vector3& linearDamping, const Vector3& angularDamping) : Joint(uniqueName, false)
38{
39 btRigidBody* bodyA = solid->rigidBody;
40 Transform frameInA = solid->getCGTransform().inverse() * attachment;
41
42 btGeneric6DofSpring2Constraint* spring = new btGeneric6DofSpring2Constraint(*bodyA, frameInA, RO_ZYX);
43 spring->enableSpring(0, true);
44 spring->enableSpring(1, true);
45 spring->enableSpring(2, true);
46 spring->enableSpring(3, true);
47 spring->enableSpring(4, true);
48 spring->enableSpring(5, true);
49 spring->setStiffness(0, linearStiffness.getX());
50 spring->setStiffness(1, linearStiffness.getY());
51 spring->setStiffness(2, linearStiffness.getZ());
52 spring->setStiffness(3, angularStiffness.getX());
53 spring->setStiffness(4, angularStiffness.getY());
54 spring->setStiffness(5, angularStiffness.getZ());
55 spring->setDamping(0, linearDamping.getX());
56 spring->setDamping(1, linearDamping.getY());
57 spring->setDamping(2, linearDamping.getZ());
58 spring->setDamping(3, angularDamping.getX());
59 spring->setDamping(4, angularDamping.getY());
60 spring->setDamping(5, angularDamping.getZ());
61 setConstraint(spring);
62}

64SpringJoint::SpringJoint(std::string uniqueName, SolidEntity* solidA, SolidEntity* solidB, const Transform& attachment,
65 const Vector3& linearStiffness, const Vector3& angularStiffness,
66 const Vector3& linearDamping, const Vector3& angularDamping) : Joint(uniqueName, false)
67{
68 btRigidBody* bodyA = solidA->rigidBody;
69 btRigidBody* bodyB = solidB->rigidBody;
70 Transform frameInA = bodyA->getCenterOfMassTransform().inverse() * attachment;
71 Transform frameInB = bodyB->getCenterOfMassTransform().inverse() * attachment;
72
73 btGeneric6DofSpring2Constraint* spring = new btGeneric6DofSpring2Constraint(*bodyA, *bodyB, frameInA, frameInB, RO_ZYX);
74 spring->enableSpring(0, true);
75 spring->enableSpring(1, true);
76 spring->enableSpring(2, true);
77 spring->enableSpring(3, true);
78 spring->enableSpring(4, true);
79 spring->enableSpring(5, true);
80 spring->setStiffness(0, linearStiffness.getX());
81 spring->setStiffness(1, linearStiffness.getY());
82 spring->setStiffness(2, linearStiffness.getZ());
83 spring->setStiffness(3, angularStiffness.getX());
84 spring->setStiffness(4, angularStiffness.getY());
85 spring->setStiffness(5, angularStiffness.getZ());
86 spring->setDamping(0, linearDamping.getX());
87 spring->setDamping(1, linearDamping.getY());
88 spring->setDamping(2, linearDamping.getZ());
89 spring->setDamping(3, angularDamping.getX());
90 spring->setDamping(4, angularDamping.getY());
91 spring->setDamping(5, angularDamping.getZ());
92 setConstraint(spring);
93}

100std::vector<Renderable> SpringJoint::Render()
101{
102 std::vector<Renderable> items(0);
103 btGeneric6DofSpring2Constraint* c = (btGeneric6DofSpring2Constraint*)getConstraint();
104 if(c != nullptr)
105 {
106 Renderable item;
107 item.model = glm::mat4(1.f);
108 item.type = RenderableType::JOINT_LINES;
109 Vector3 A = (c->getRigidBodyA().getCenterOfMassTransform() * c->getFrameOffsetA()).getOrigin();
110 Vector3 B = (c->getRigidBodyB().getCenterOfMassTransform() * c->getFrameOffsetB()).getOrigin();
111 item.points.push_back(glm::vec3(A.getX(), A.getY(), A.getZ()));
112 item.points.push_back(glm::vec3(B.getX(), B.getY(), B.getZ()));
113 items.push_back(item);
114 }
115 return items;
116}
