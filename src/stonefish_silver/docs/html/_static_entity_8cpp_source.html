102std::vector<Renderable> StaticEntity::Render()
103{
104 std::vector<Renderable> items(0);
105
106 if(rigidBody != nullptr && phyObjectId >= 0 && isRenderable())
107 {
108 Transform trans;
109 rigidBody->getMotionState()->getWorldTransform(trans);
110
111 Renderable item;
112 item.type = RenderableType::SOLID;
113 item.materialName = mat.name;
114 item.objectId = phyObjectId;
115 item.lookId = dm == DisplayMode::GRAPHICAL ? lookId : -1;
116 item.model = glMatrixFromTransform(trans);
117 items.push_back(item);
118 }
119
120 return items;
121}

131void StaticEntity::BuildRigidBody(btCollisionShape* shape)
132{
133 btDefaultMotionState* motionState = new btDefaultMotionState();
134
135 btRigidBody::btRigidBodyConstructionInfo rigidBodyCI(Scalar(0), motionState, shape, Vector3(0,0,0));
136 rigidBodyCI.m_friction = rigidBodyCI.m_rollingFriction = rigidBodyCI.m_restitution = Scalar(0); //not used
137 rigidBodyCI.m_linearDamping = rigidBodyCI.m_angularDamping = Scalar(0); //not used
138 rigidBodyCI.m_linearSleepingThreshold = rigidBodyCI.m_angularSleepingThreshold = Scalar(0); //not used
139 rigidBodyCI.m_additionalDamping = false;
140
141 rigidBody = new btRigidBody(rigidBodyCI);
142 rigidBody->setUserPointer(this);
143 rigidBody->setCollisionFlags(rigidBody->getCollisionFlags() | btCollisionObject::CF_STATIC_OBJECT | btCollisionObject::CF_CUSTOM_MATERIAL_CALLBACK);
144
145 BuildGraphicalObject();
146}

164void StaticEntity::GroupTransform(std::vector<StaticEntity*>& objects, const Transform& centre, const Transform& transform)
165{
166 for(unsigned int i=0; i<objects.size(); ++i)
167 {
168 Transform Tw = objects[i]->getTransform();
169 Transform Tc = centre.inverse() * Tw;
170 Tc = transform * Tc;
171 Tw = centre * Tc;
172 objects[i]->setTransform(Tw);
173 }
174}
