79void SuctionCup::Engage(SimulationManager* sm)
80{
81 if(attach != nullptr && joint == nullptr && pump)
82 {
83 btDispatcher* dispatcher = sm->getDynamicsWorld()->getDispatcher();
84 int numManifolds = dispatcher->getNumManifolds();
85 for(int i=0; i<numManifolds; ++i)
86 {
87 btPersistentManifold* contactManifold = dispatcher->getManifoldByIndexInternal(i);
88 if(contactManifold->getNumContacts() == 0)
89 continue;
90
91 btCollisionObject* coA = (btCollisionObject*)contactManifold->getBody0();
92 btCollisionObject* coB = (btCollisionObject*)contactManifold->getBody1();
93 Entity* entA = (Entity*)coA->getUserPointer();
94 Entity* entB = (Entity*)coB->getUserPointer();
95
96 if(entA == attach && entB->getType() == EntityType::SOLID)
97 {
98 Transform jointFrame = ((SolidEntity*)entA)->getCG2CTransform();
99 jointFrame.setOrigin(contactManifold->getContactPoint(0).getPositionWorldOnA());
100 if(attachFe != nullptr)
101 {
102 joint = new SphericalJoint(getName() + "/P2P", (SolidEntity*)entB, attachFe, attachLinkId, jointFrame.getOrigin(), false);
103 joint->getSolidB()->getRigidBody()->setDamping(0.0, 0.5);
104 }
105 else
106 {
107 joint = new SpringJoint(getName() + "/Spring", (SolidEntity*)entA, (SolidEntity*)entB, jointFrame,
108 Vector3(10,10,10), Vector3(10,10,10), Vector3(0.5,0.5,0.5), Vector3(0.5,0.5,0.5));
109 }
110 }
111 else if(entB == attach && entA->getType() == EntityType::SOLID)
112 {
113 Transform jointFrame = ((SolidEntity*)entB)->getCG2CTransform();
114 jointFrame.setOrigin(contactManifold->getContactPoint(0).getPositionWorldOnB());
115 if(attachFe != nullptr)
116 {
117 joint = new SphericalJoint(getName() + "/P2P", (SolidEntity*)entA, attachFe, attachLinkId, jointFrame.getOrigin(), false);
118 joint->getSolidB()->getRigidBody()->setDamping(0.0, 0.5);
119 }
120 else
121 {
122 joint = new SpringJoint(getName() + "/Spring", (SolidEntity*)entB, (SolidEntity*)entA, jointFrame,
123 Vector3(10,10,10), Vector3(10,10,10), Vector3(0.5,0.5,0.5), Vector3(0.5,0.5,0.5));
124 }
125 }
126
127 if(joint != nullptr)
128 {
129 contactManifold->clearManifold();
130 sm->AddJoint(joint);
131 break;
132 }
133 }
134 }
135 else if(joint != nullptr && !pump)
136 {
137 if(joint->isMultibodyJoint())
138 joint->getSolidB()->getRigidBody()->setDamping(0.0, 0.0);
139 sm->RemoveJoint(joint);
140 joint = nullptr;
141 }
142}
