55void ThermalCamera::setNoise(GLfloat tempStdDev)
56{
57 if(tempStdDev >= 0.f && tempStdDev != noiseStdDev)
58 {
59 noiseStdDev = tempStdDev;
60 if(glCamera != nullptr)
61 glCamera->setNoise(noiseStdDev);
62 }
63}

65void ThermalCamera:: setDisplaySettings(ColorMap cm, Scalar minTemp, Scalar maxTemp)
66{
67 colorMap = cm;
68
69 if(minTemp > maxTemp)
70 displayRange = measurementRange;
71 else
72 {
73 glm::vec2 tempRange = glm::vec2((GLfloat)minTemp, (GLfloat)maxTemp);
74 displayRange = glm::clamp(tempRange, measurementRange.x, measurementRange.y);
75 }
76
77 if(glCamera != nullptr)
78 {
79 glCamera->setColorMap(colorMap);
80 glCamera->setDisplayRange(displayRange);
81 }
82}

104void ThermalCamera::InitGraphics()
105{
106 glCamera = new OpenGLThermalCamera(glm::vec3(0,0,0), glm::vec3(0,0,1.f), glm::vec3(0,-1.f,0), 0, 0, resX, resY, (GLfloat)fovH, measurementRange, depthRange, freq < Scalar(0));
107 glCamera->setNoise(noiseStdDev);
108 glCamera->setColorMap(colorMap);
109 glCamera->setDisplayRange(displayRange);
110 glCamera->setCamera(this);
111 UpdateTransform();
112 glCamera->UpdateTransform();
113 InternalUpdate(0);
114 ((GraphicalSimulationApp*)SimulationApp::getApp())->getGLPipeline()->getContent()->AddView(glCamera);
115
116 unsigned int w, h;
117 getResolution(w, h);
118 displayData = new GLubyte[w*h*3];
119}

121void ThermalCamera::SetupCamera(const Vector3& eye, const Vector3& dir, const Vector3& up)
122{
123 glm::vec3 eye_ = glm::vec3((GLfloat)eye.x(), (GLfloat)eye.y(), (GLfloat)eye.z());
124 glm::vec3 dir_ = glm::vec3((GLfloat)dir.x(), (GLfloat)dir.y(), (GLfloat)dir.z());
125 glm::vec3 up_ = glm::vec3((GLfloat)up.x(), (GLfloat)up.y(), (GLfloat)up.z());
126 glCamera->SetupCamera(eye_, dir_, up_);
127}

134void ThermalCamera::NewDataReady(void* data, unsigned int index)
135{
136 if(newDataCallback != nullptr)
137 {
138 if(index == 0)
139 {
140 unsigned int w, h;
141 getResolution(w, h);
142 memcpy(displayData, data, w*h*3);
143 }
144 else
145 {
146 temperatureData = (GLfloat*)data;
147 newDataCallback(this);
148 temperatureData = nullptr;
149 }
150 }
151}
