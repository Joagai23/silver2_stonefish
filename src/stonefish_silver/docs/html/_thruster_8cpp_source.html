58void Thruster::setSetpoint(Scalar s)
59{
60 if (normalized)
61 setpoint = btClamped(s, Scalar(-1), Scalar(1)) * setpointLimit;
62 else
63 setpoint = btClamped(s, -setpointLimit, setpointLimit);
64 if (inv) setpoint *= Scalar(-1);
65 ResetWatchdog();
66}

114void Thruster::Update(Scalar dt)
115{
116 Actuator::Update(dt);
117
118 if (attach == nullptr)
119 return; // No attachment, no action
120
121 // Update rotation & angular velocity
122 if (rotorModel->getType() == RotorDynamicsType::MECHANICAL_PI)
123 std::static_pointer_cast<MechanicalPI>(rotorModel)->setDampingTorque(btFabs(torque));
124 omega = rotorModel->Update(dt, setpoint);
125 theta += omega * dt; // Just for animation
126
127 // Check if thruster is sumberged and compute thrust model
128 Transform solidTrans = attach->getCGTransform();
129 Transform thrustTrans = attach->getOTransform() * o2a;
130 Ocean *ocn = SimulationApp::getApp()->getSimulationManager()->getOcean();
131
132 if (ocn != nullptr && ocn->IsInsideFluid(thrustTrans.getOrigin()))
133 {
134 // Update Thrust
135 if (thrustModel->getType() == ThrustModelType::FD)
136 {
137 Vector3 relPos = thrustTrans.getOrigin() - solidTrans.getOrigin();
138 Vector3 velocity = attach->getLinearVelocityInLocalPoint(relPos);
139 Scalar u = -thrustTrans.getBasis().getColumn(0).dot(ocn->GetFluidVelocity(thrustTrans.getOrigin()) - velocity);
140 std::static_pointer_cast<FDThrust>(thrustModel)->setIncomingFluidVelocity(u);
141 }
142 std::pair<Scalar, Scalar> out = thrustModel->Update(omega);
143 thrust = out.first;
144 torque = out.second;
145
146 // Account for handedness of the propeller
147 if (!RH && thrustModel->getType() != ThrustModelType::FD)
148 thrust = -thrust;
149
150 // Apply forces and torques
151 Vector3 thrustV(thrust, 0, 0);
152 Vector3 torqueV(torque, 0, 0);
153 attach->ApplyCentralForce(thrustTrans.getBasis() * thrustV);
154 attach->ApplyTorque((thrustTrans.getOrigin() - solidTrans.getOrigin()).cross(thrustTrans.getBasis() * thrustV));
155 attach->ApplyTorque(thrustTrans.getBasis() * torqueV);
156 }
157 else
158 {
159 thrust = Scalar(0);
160 torque = Scalar(0);
161 }
162}

164std::vector<Renderable> Thruster::Render()
165{
166 Transform thrustTrans = Transform::getIdentity();
167 if (attach != nullptr)
168 thrustTrans = attach->getOTransform() * o2a;
169 else
170 LinkActuator::Render();
171
172 // Rotate propeller
173 thrustTrans *= Transform(Quaternion(0, 0, theta), Vector3(0, 0, 0));
174
175 // Add renderable
176 std::vector<Renderable> items(0);
177 Renderable item;
178 item.type = RenderableType::SOLID;
179 item.materialName = propeller_->getMaterial().name;
180 item.objectId = propeller_->getGraphicalObject();
181 item.lookId = dm == DisplayMode::GRAPHICAL ? propeller_->getLook() : -1;
182 item.model = glMatrixFromTransform(thrustTrans);
183 items.push_back(item);
184
185 item.type = RenderableType::ACTUATOR_LINES;
186 item.points.push_back(glm::vec3(0, 0, 0));
187 item.points.push_back(glm::vec3(0.1f * thrust, 0, 0));
188 items.push_back(item);
189
190 return items;
191}
