70void Trajectory::Play(Scalar dt)
71{
72 //Update time
73 switch(playMode)
74 {
75 case PlaybackMode::ONETIME:
76 {
77 playTime += dt;
78 if(playTime > endTime)
79 playTime = endTime;
80 }
81 break;
82
83 case PlaybackMode::REPEAT:
84 {
85 playTime += dt;
86 if(playTime > endTime)
87 {
88 playTime -= endTime;
89 ++iteration;
90 }
91 }
92 break;
93
94 case PlaybackMode::BOOMERANG:
95 {
96 playTime += forward ? dt : -dt;
97 if(playTime > endTime)
98 {
99 playTime = endTime - (playTime - endTime);
100 forward = false;
101 ++iteration;
102 }
103 else if(playTime < Scalar(0))
104 {
105 playTime = -playTime;
106 forward = true;
107 ++iteration;
108 }
109 }
110 break;
111 }
112
113 //Compute new interpolants
114 Interpolate();
115
116 //Clear velocities when stopped
117 if(playTime == endTime && playMode == PlaybackMode::ONETIME)
118 {
119 interpVel.setZero();
120 interpAngVel.setZero();
121 interpAcc.setZero();
122 }
123}

125void Trajectory::calculateVelocityShortestPath(const Transform &transform0, const Transform &transform1, Scalar timeStep, Vector3 &linVel, Vector3 &angVel)
126{
127 linVel = (transform1.getOrigin() - transform0.getOrigin()) / timeStep;
128
129 // https://doi.org/10.48550/arXiv.1812.01537 Appendix B, SO(3) Logarithm (Lie Theory)
130 Matrix3 R0, R1;
131 R0 = transform0.getBasis();
132 R1 = transform1.getBasis();
133 Matrix3 dR = R0.transpose() * R1;
134
135 // trace of dR
136 Scalar tr = dR[0][0] + dR[1][1] + dR[2][2];
137
138 Scalar theta = btAcos((tr - 1.0) / 2.0);
139
140 // check for divide by zero
141 if (btFabs(theta) < SIMD_EPSILON)
142 {
143 angVel.setZero();
144 }
145 else
146 {
147 Matrix3 R3 = dR - dR.transpose();
148 Vector3 v(R3[2][1], R3[0][2], R3[1][0]);
149
150 angVel = R0 * (v * theta / (2.0 * btSin(theta)) / timeStep);
151 }
152}
153}
