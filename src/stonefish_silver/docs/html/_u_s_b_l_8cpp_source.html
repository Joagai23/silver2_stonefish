66void USBL::InternalUpdate(Scalar dt)
67{
68 AcousticModem::InternalUpdate(dt);
69
70 //Auto pinging with fixed rate
71 if(ping && pingRate > Scalar(0))
72 {
73 pingTime += dt;
74 Scalar invRate = Scalar(1)/pingRate;
75
76 if(pingTime >= invRate)
77 {
78 Comm::SendMessage("PING");
79 pingTime -= invRate;
80 }
81 }
82}
