38void USBLSimple::setNoise(Scalar rangeDev, Scalar horizontalAngleDevDeg, Scalar verticalAngleDevDeg)
39{
40 noiseRange = std::normal_distribution<Scalar>(Scalar(0), btFabs(rangeDev));
41 noiseHAngle = std::normal_distribution<Scalar>(Scalar(0), btFabs(horizontalAngleDevDeg)/Scalar(180)*M_PI);
42 noiseVAngle = std::normal_distribution<Scalar>(Scalar(0), btFabs(verticalAngleDevDeg)/Scalar(180)*M_PI);
43 noise = true;
44}

46void USBLSimple::setResolution(Scalar range, Scalar angleDeg)
47{
48 rangeRes = btFabs(range);
49 angleRes = btFabs(angleDeg)/Scalar(180)*M_PI;

52void USBLSimple::ProcessMessages()
53{
54 std::shared_ptr<AcousticDataFrame> msg;
55 std::string ack {"ACK"};
56 std::vector<uint8_t> ackData(ack.begin(), ack.end());
57
58 for(auto it = rxBuffer.begin(); it != rxBuffer.end(); ++it)
59 {
60 msg = std::static_pointer_cast<AcousticDataFrame>(*it);
61 if(msg->data == ackData)
62 {
63 //Get message data
64 AcousticModem* cNode = getNode(msg->source);
65 Vector3 cO = msg->txPosition;
66 Transform dT = getDeviceFrame();
67 Vector3 dO = dT.getOrigin();
68 Vector3 dir = getDeviceFrame().getBasis().inverse() * ((cO - dO).normalized()); //Direction in device frame
69 Scalar slantRange = msg->travelled/Scalar(2); //Distance to node is hald of the full travelled distance
70 Scalar t = msg->timeStamp + slantRange/SOUND_VELOCITY_WATER;
71
72 //Find ranging angles
73 Scalar d = Vector3(dir.getX(), dir.getY(), Scalar(0)).length();
74 Scalar hAngle = atan2(dir.getY(), dir.getX());
75 Scalar vAngle = atan2(d, dir.getZ());
76
77 //Apply noise and quantization
78 if(noise)
79 {
80 slantRange += noiseRange(randomGenerator);
81 hAngle += noiseHAngle(randomGenerator);
82 vAngle += noiseVAngle(randomGenerator);
83 }
84
85 if(rangeRes > Scalar(0))
86 slantRange -= btFmod(slantRange, rangeRes); //Quantization
87
88 if(angleRes > Scalar(0))
89 {
90 hAngle -= btFmod(hAngle, angleRes); //Quantization
91 vAngle -= btFmod(vAngle, angleRes); //Quantization
92 }
93
94 //Calculate receiver position
95 Vector3 pos;
96 dir.setX(btCos(hAngle) * btSin(vAngle));
97 dir.setY(btSin(hAngle) * btSin(vAngle));
98 dir.setZ(btCos(vAngle));
99 dir.normalize();
100 pos = dir * slantRange;
101
102 //Update position in the transponder and in the USBL
103 Vector3 worldPos = dT * pos;
104 cNode->UpdatePosition(worldPos, true);
105
106 //Populate beacon info structure
107 BeaconInfo b;
108 b.t = t;
109 b.relPos = pos;
110 b.azimuth = hAngle;
111 b.elevation = M_PI_2 - vAngle;
112 b.range = slantRange;
113 b.localDepth = dO.getZ();
114 b.localOri = dT.getRotation();
115 beacons[msg->source] = b;
116
117 newDataAvailable = true;
118 }
119 }
120
121 // Standard processing of messages and removal of "ACK and "PING" messages
122 AcousticModem::ProcessMessages();
123}
