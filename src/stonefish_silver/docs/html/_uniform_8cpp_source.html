51std::vector<Renderable> Uniform::Render(VelocityFieldUBO& ubo)
52{
53 std::vector<Renderable> items(0);
54 Scalar vel = v.length();
55 Vector3 dir = vel > Scalar(0) ? (v/vel) : Vector3(0,0,0);
56 ubo.posR = glm::vec4(0.f);
57 ubo.dirV = glm::vec4((GLfloat)dir.getX(), (GLfloat)dir.getY(), (GLfloat)dir.getZ(), (GLfloat)vel);
58 ubo.params= glm::vec3(0.f);
59 ubo.type = 0;
60 return items;
61}
