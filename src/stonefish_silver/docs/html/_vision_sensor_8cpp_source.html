54Transform VisionSensor::getSensorFrame() const
55{
56 if(attach != nullptr)
57 {
58 if(attach->getType() == EntityType::STATIC)
59 return ((StaticEntity*)attach)->getTransform() * o2s;
60 else
61 return ((MovingEntity*)attach)->getOTransform() * o2s;
62 }
63 else
64 return o2s;
65}

67void VisionSensor::getSensorVelocity(Vector3& linear, Vector3& angular) const
68{
69 if(attach != nullptr && attach->getType() != EntityType::STATIC)
70 {
71 linear = ((MovingEntity*)attach)->getLinearVelocity();
72 angular = ((MovingEntity*)attach)->getAngularVelocity();
73 }
74 else
75 {
76 linear = V0();
77 angular = V0();
78 }
79}
