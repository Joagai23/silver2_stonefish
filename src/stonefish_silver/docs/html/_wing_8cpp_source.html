34Wing::Wing(std::string uniqueName, BodyPhysicsSettings phy, Scalar baseChordLength, Scalar tipChordLength,
35 Scalar maxCamber, Scalar maxCamberPos, Scalar profileThickness, Scalar wingLength, const Transform& origin,
36 std::string material, std::string look, Scalar thickness)
37 : SolidEntity(uniqueName, phy, material, look, thickness)
38{
39 T_O2G = T_O2C = origin;
40 baseChordLength = baseChordLength <= Scalar(0) ? Scalar(1) : baseChordLength;
41 tipChordLength = tipChordLength < Scalar(0) ? baseChordLength : tipChordLength;
42 maxCamber = maxCamber < Scalar(0) ? Scalar(0) : (maxCamber > Scalar(100) ? Scalar(100) : maxCamber);
43 maxCamberPos = maxCamberPos < Scalar(0) ? Scalar(0) : (maxCamberPos > Scalar(100) ? Scalar(100) : maxCamberPos);
44 profileThickness = profileThickness < Scalar(1) ? Scalar(1) : (profileThickness > Scalar(40) ? Scalar(40) : profileThickness);
45 wingLength = wingLength < Scalar(0) ? Scalar(0) : wingLength;
46
47 //1. Build wing geometry
48 phyMesh = OpenGLContent::BuildWing((GLfloat)baseChordLength, (GLfloat)tipChordLength, (GLfloat)maxCamber, (GLfloat)maxCamberPos,
49 (GLfloat)profileThickness, (GLfloat)wingLength);
50
51 //2. Compute physical properties
52 Vector3 CG;
53 Matrix3 Irot;
54 ComputePhysicalProperties(phyMesh, thickness, mat.density, mass, CG, volume, surface, Ipri, Irot);
55 T_CG2C.setOrigin(-CG); //Set CG position
56 T_CG2C = Transform(Irot, Vector3(0,0,0)).inverse() * T_CG2C; //Align CG frame to principal axes of inertia
57 T_CG2O = T_CG2C * T_O2C.inverse();
58 T_CG2G = T_CG2O * T_O2G;
59
60 //3. Compute hydrodynamic properties
61 ComputeFluidDynamicsApprox(GeometryApproxType::ELLIPSOID);
62 T_O2H = T_CG2O.inverse() * T_CG2H;
63 P_CB = Vector3(0,0,0);
64}

66Wing::Wing(std::string uniqueName, BodyPhysicsSettings phy, Scalar baseChordLength, Scalar tipChordLength, std::string NACA, Scalar wingLength, const Transform& origin,
67 std::string material, std::string look, Scalar thickness)
68 : SolidEntity(uniqueName, phy, material, look, thickness)
69{
70 T_O2G = T_O2C = origin;
71 baseChordLength = baseChordLength <= Scalar(0) ? Scalar(1) : baseChordLength;
72 tipChordLength = tipChordLength < Scalar(0) ? baseChordLength : tipChordLength;
73 wingLength = wingLength < Scalar(0) ? Scalar(0) : wingLength;
74
75 Scalar maxCamber(0);
76 Scalar maxCamberPos(0);
77 Scalar profileThickness(20);
78
79 //1. Decode NACA
80 if(NACA.length() == 4)
81 {
82 int nacaCode[4];
83 int i;
84
85 for(i=0; i<4; ++i)
86 if(NACA[i] >= '0' && NACA[i] <= '9')
87 nacaCode[i] = NACA[i] - '0';
88 else
89 break;
90
91 if(i==4) //Decode
92 {
93 maxCamber = (Scalar)nacaCode[0];
94 maxCamberPos = (Scalar)nacaCode[1] * 10.0;
95 profileThickness = (Scalar)nacaCode[2] * 10.0 + (Scalar)nacaCode[3];
96 }
97 }
98
99 //2. Build wing geometry
100 phyMesh = OpenGLContent::BuildWing((GLfloat)baseChordLength, (GLfloat)tipChordLength, (GLfloat)maxCamber, (GLfloat)maxCamberPos,
101 (GLfloat)profileThickness, (GLfloat)wingLength);
102
103
104 //3. Compute physical properties
105 Vector3 CG;
106 Matrix3 Irot;
107 ComputePhysicalProperties(phyMesh, thickness, mat.density, mass, CG, volume, surface, Ipri, Irot);
108 T_CG2C.setOrigin(-CG); //Set CG position
109 T_CG2C = Transform(Irot, Vector3(0,0,0)).inverse() * T_CG2C; //Align CG frame to principal axes of inertia
110 T_CG2O = T_CG2C * T_O2C.inverse();
111 T_CG2G = T_CG2O * T_O2G;
112
113 //4. Compute hydrodynamic properties
114 ComputeFluidDynamicsApprox(GeometryApproxType::ELLIPSOID);
115 T_O2H = T_CG2O.inverse() * T_CG2H;
116 P_CB = Vector3(0,0,0);
117}

124btCollisionShape* Wing::BuildCollisionShape()
125{
126 btConvexHullShape* convex = new btConvexHullShape();
127 for(size_t i=0; i<phyMesh->getNumOfVertices(); ++i)
128 {
129 glm::vec3 pos = phyMesh->getVertexPos(i);
130 Vector3 v(pos.x, pos.y, pos.z);
131 convex->addPoint(v);
132 }
133 //convex->optimizeConvexHull();
134 convex->setMargin(0);
135 return convex;
136}
